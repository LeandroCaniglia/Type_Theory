\chapter{Induction}

\section{Intuition for Uniqueness}

The principles of induction and recursion do more than guarantee the existence of certain functions; they completely characterize them by providing a uniqueness principle. Here we present two examples: natural numbers and lists.

Recall that the type $\N$ of natural numbers has two constructors
\begin{itemize}
    \item $0\colon\N$
    \item $\fun{succ}\colon\N\to\N$.
\end{itemize}
A similar example is the type $\fun L_A$ (a.k.a.~$\fun{List}(A)$) of finite lists of elements of some type $A$, which has constructors
\begin{itemize}
    \item $\fun{nil}\colon\fun L_A$
    \item $\fun{cons}\colon A\to\fun L_A\to\fun L_A$,
\end{itemize}
where $\fun{nil}$ represents the empty list, and $\fun{cons}(a,\ell)$ the list with head $a$ and tail~$\ell$.

The induction principle for this type states that, given a family $E\colon\fun L_A\to\univ$, to construct a dependent function
\[
    f\colon\prod_{\ell\colon\fun L_A}E(\ell),
\]
it is sufficient to provide the following data:
\begin{itemize}
    \item\phantomsection\label{itm:list-induction} a term $e_{\fun{nil}}\colon E(\fun{nil})$ and
    \item a dependent function $e_{\fun{cons}}\colon\prod_{a\colon A}\prod_{\ell\colon\fun L_A}E(\ell)\to E(\fun{cons}(a,\ell))$.
\end{itemize}
This principle also specifies that the constructed function $f$ computes as expected on the constructors, meaning that we have the definitional equalities
\begin{equation}\label{eq:list-recurrences}
    f(\fun{nil})\equiv e_{\fun{nil}}
    \quad\text{and}\quad
    f(\fun{cons}(a,\ell))
        \equiv e_{\fun{cons}}(a,\ell,f(\ell)).
\end{equation}

\begin{xmpl}\label{xmpl:list-append}
    To define the function
    \[
        \fun{append}\colon\fun L_A\to\fun L_A\to\fun L_A
    \]
    that joins two lists, we define:
    \begin{itemize}
        \item the motive $E(\ell)\defeq\fun L_A\to\fun L_A$,
        \item $e_{\fun{nil}}\defeq\id_{\fun L_A}$, and
        \item $e_{\fun{cons}}(a,\ell,g)\defeq\lambda y.\,\fun{cons}(a,g(y))$.
    \end{itemize}
    Then $\fun{append}\defeq\fun{ind}_{\fun L_A}(E,e_{\fun{nil}},e_{\fun{cons}})$ satisfies
    \[
        \fun{append}(\fun{nil},y)\equiv y
    \]
    and
    \[
        \fun{append}(\fun{cons}(a,\ell),y)
        \equiv
        \fun{cons}(a,\fun{append}(\ell,y)).
    \]
\end{xmpl}

\begin{thm}\label{thm:N-induction-uniqueness}
    Let $f,g\colon\prod_{x\colon\N}E(x)$ be functions satisfying the recurrences
    \[
        e_z\colon E(0)
        \quad\text{and}\quad
        e_s\colon\prod_{n\colon\N}E(n)\to E(\fun{succ}(n))
    \]
    up to propositional equality. That is, assume that
    \[
        f(0)\eq{E(0)}e_z
        \quad\text{and}\quad
        g(0)\eq{E(0)}e_z,
    \]
    and that
    \begin{align*}
        \prod_{n\colon\N}
            f(\fun{succ}(n))
            &\eq{E(\fun{succ}(n))}
            e_s(n,f(n))
    \intertext{and}
        \prod_{n\colon\N}
            g(\fun{succ}(n))
            &\eq{E(\fun{succ}(n))}
            e_s(n,g(n)).
    \end{align*}
    Then $f\sim g$.
\end{thm}

\begin{proof}
    Let $C\colon\N\to\univ$ be the type family defined by $C(n)\defeq f(n)\eq{E(n)}g(n)$. To prove the theorem, it suffices to produce a dependent function in $\prod_{n\colon\N}C(n)$.

    \needspace{2\baselineskip}
    By the induction principle of $\N$, we need to provide the following data:
    \begin{itemize}
        \item a term $c_z\colon C(0)$ and
        \item a function $c_s\colon\prod_{n\colon\N}C(n)\to C(\fun{succ}(n))$.
    \end{itemize}
    The first part of our hypothesis gives us paths 
    \[
        p_0\colon f(0)\eq{E(0)}e_z
        \quad\text{and}\quad
        q_0\colon g(0)\eq{E(0)}e_z,
    \]
    which we can use to produce $c_z\defeq p_0\ct q_0^{-1}\colon C(0)$.

    To define $c_s$, given $n\colon\N$ and $h\colon C(n)$, we must construct $c_s(n,h)\colon C(\fun{succ}(n))$. Since $h\colon f(n)\eq{E(n)}g(n)$, we can use the action on paths to obtain
    \[
        \fun{ap}_{e_s(n,\dummy)}(h)\colon
            e_s(n,f(n))\eq{E(\fun{succ}(n))} e_s(n,g(n)).
    \]
    The second part of the hypothesis yields paths
    \[
        p_n\colon
            f(\fun{succ}(n))
            \eq{E(\fun{succ}(n))}
            e_s(n,f(n))
    \]
    and
    \[
        q_n\colon
            g(\fun{succ}(n))
            \eq{E(\fun{succ}(n))}
            e_s(n,g(n)).
    \]
    Concatenating these paths yields
    \[
        p_n\ct\fun{ap}_{e_s(n,\dummy)}(h)\ct q_n^{-1}\colon
            f(\fun{succ}(n))
            \eq{E(\fun{succ}(n))}
            g(\fun{succ}(n)),
    \]
    which is definitionally equal to $C(\fun{succ}(n))$. To complete the proof, we set $c_s(n,h)$ to be this concatenation.
\end{proof}

\begin{rem}\label{rem:List-induction-uniqueness}
    A similar theorem holds for lists: \textit{If $f,g\colon\prod_{x\colon\fun L_A}E(x)$ propositionally satisfy the recurrences \eqref{eq:list-recurrences}, then $f\sim g$}.

    The proof proceeds exactly as in the theorem, introducing
    \[
        C(\ell)\defeq f(\ell)\eq{E(\ell)}g(\ell),
    \]
    and then defining $c_{\fun{nil}}$ by concatenating the paths
    \[
        f(\fun{nil})\eq{E(\fun{nil})}e_{\fun{nil}}
        \quad\text{and}\quad
        g(\fun{nil})\eq{E(\fun{nil})}e_{\fun{nil}}.
    \]
    The definition of $c_{\fun{cons}}(a,\ell,\eta)$ comes from
    \[
        \fun{ap}_{e_{\fun{cons}}(a,\ell,\dummy)}(\eta)
            \colon e_{\fun{cons}}(a,\ell,f(\ell))
                \eq{E(\fun{cons}(a,\ell))}
                e_{\fun{cons}}(a,\ell,g(\ell))
    \]
    and the equalities
    \[
        f(\fun{cons}(a,\ell))\eq{E(\fun{cons}(a,\ell))}
        e_{\fun{cons}}(a,\ell,f(\ell))
    \]
    and
    \[
        g(\fun{cons}(a,\ell))\eq{E(\fun{cons}(a,\ell))}
        e_{\fun{cons}}(a,\ell,g(\ell)).
    \]
\end{rem}

Suppose we have a different construction of the natural numbers:
\begin{itemize}
    \item $0'\colon\N'$
    \item $\fun{succ}'\colon\N'\to\N'$,
\end{itemize}
along with an induction principle
\[
    \fun{ind}_{\N'}\colon\prod_{E\colon\N'\to\univ}
        E(0')\to\Bigl(\prod_{n'\colon\N'}E(n') \to E(\fun{succ}'(n'))\Bigr)
        \to \prod_{n'\colon\N'}E(n')
\]
and the definitional equalities
\begin{align*}
    \fun{ind}_{\N'}(E, e_0, e_s, 0') &\defeq e_0,\\
    \fun{ind}_{\N'}(E, e_0, e_s, \fun{succ}'(n')) &\defeq e_s(n', \fun{ind}_{\N'}(E, e_0, e_s, n')).
\end{align*}
We can use the constant families $E(n)\defeq\N'$ and $E'(n')\defeq\N$ to define
\begin{align*}
    f\colon\N\to\N'&\defeq\fun{ind}_\N
        (
            E,
            0',
            \lambda n.\,\fun{succ}'
        )
\intertext{and}
    g\colon\N'\to\N&\defeq\fun{ind}_{\N'}
        (
            E',
            0,
            \lambda n'.\,\fun{succ}
        ).
\end{align*}
First, observe that
\[
    g\circ f(0)\equiv g(0')\equiv 0\equiv\id_\N(0)
\]
and also
\[
    g\circ f(\fun{succ}(n))
        \equiv g(\fun{succ}'(f(n)))
        \equiv \fun{succ}(g(f(n)))
        \equiv \fun{succ}(g\circ f(n)).
\]
Since $\id_\N(\fun{succ}(n))\equiv\fun{succ}(\id_\N(n))$, by setting $e_s(x,y)\defeq \fun{succ}(y)$ we can rewrite the equivalences above as
\[
    g\circ f(\fun{succ}(n))\equiv e_s(n,g\circ f(n))
    \quad\text{and}\quad
    \id_\N(\fun{succ}(n))\equiv e_s(n,\id_\N(n)).
\]
Therefore, by Theorem~\ref{thm:N-induction-uniqueness}, we obtain a homotopy $\vartheta\colon g\circ f\sim\id_{\N}$.

A symmetric argument yields a homotopy $\eta\colon f\circ g\sim\id_{\N'}$. Consequently, $f$ and $g$ are quasi-inverses and so $\N\simeq\N'$. Thus, the univalence axiom constructs a path $\N\eq{\univ}\N'$.

\begin{xmpl}\label{xmpl:N-from-List(1)}
    A concrete example of an alternative construction of $\N$ is the following:
    \begin{itemize}
        \item $\N'\defeq\fun L_{\fun1}$,
        \item $0'\defeq\fun{nil}$, and
        \item $\fun{succ}'(\ell)\defeq\fun{cons}(\fstar,\ell)$,
    \end{itemize}
    with the induction principle
    \begin{align*}
        \fun{ind}_{\N'}&\colon\prod_{E\colon\N'\to\univ}
            E(\fun{nil})
            \to\Bigl(\prod_{\ell\colon\N'}E(\ell)
                \to E(\fun{succ}'(\ell))\Bigr)
            \to \prod_{y\colon\N'}E(y)\\
        \fun{ind}_{\N'}
            (
                E,
                e_{\fun{nil}},
                g
            )
        &\defeq
        \fun{ind}_{\fun{List}(\fun1)}
            (
                E,
                e_{\fun{nil}},
                \lambda z\colon\fun1.\,g_z
            ),
    \end{align*}
    which fulfills the requirements of $\fun{ind}_{\fun{List}(\fun1)}$ because
    \[
        (\lambda z\colon\fun1.\,g_z)\colon
            \prod_{z\colon\fun1}\;\prod_{\ell\colon\fun{List}(\fun1)}E(\ell)\to E(\fun{cons}(z,\ell))
    \]
    is defined as follows. For a given path $p_z\colon\fstar\eq{\fun1}z$, the action on paths yields
    \[
        \fun{ap}_{\fun{cons}(\dummy,\ell)}(p_z)\colon
        \fun{cons}(\fstar,\ell)\eq{\N'}\fun{cons}(z,\ell).
    \]
    Since $g(\ell)\colon E(\ell)\to E(\fun{cons}(\fstar,\ell))$, we can define $g_z(\ell)$ by composing $g(\ell)$ with the transport function along this path:
    \[
        g_z(\ell)\defeq
            \fun{transport}^E(\fun{ap}_{\fun{cons}(\dummy,\ell)}(p_z), \dummy)
                \circ g(\ell).
    \]
\end{xmpl}

\begin{rem}
    The same ideas can be applied to other inductive types. For instance, as shown in Exercise~\ref{exr:A+B-from-rec2}, the coproduct $A+B$ can be implemented as
    \[
        A\oplus B\defeq\sum_{x\colon\fun2}
            \fun{rec}_{\fun2}(\univ,A,B,x)
    \]
    with constructors
    \begin{itemize}
        \item $\fun{in}_A(x)\defeq\dep{0_{\fun2},x}$ and
        \item $\fun{in}_B(y)\defeq\dep{1_{\fun2},y}$,
    \end{itemize}
    which are well-defined because $\fun{rec}_{\fun2}(\univ,A,B,0_{\fun2})\equiv A$ and $\fun{rec}_{\fun2}(\univ,A,B,1_{\fun2})\equiv B$.
    
    In this case, the induction principle for $A\oplus B$ is given by
    \[
        \fun{ind}_{A\oplus B}(C,g_0,g_1)
        \defeq
        \fun{ind}_{\sum_{t\colon\fun2}\fun{rec}_{\fun2}(\univ,A,B,t)}
        \bigl(
            C,
            \fun{ind}_{\fun2}(D,g_0,g_1)
        \bigr),
    \]
    where the motive $D\colon\fun2\to\univ$ is defined as
    \[
        D(t)\defeq\prod_{z\colon\fun{rec}_{\fun2}(\univ,A,B,t)}C(\dep{t,z}).
    \]
    The associated computation rules evaluate as follows:
    \begin{align*}
        \fun{ind}_{A\oplus B}(C,g_0,g_1,\fun{in}_A(x)) &\equiv g_0(x),\\
        \fun{ind}_{A\oplus B}(C,g_0,g_1,\fun{in}_B(y)) &\equiv g_1(y).
    \end{align*}
    Since $A\oplus B$ satisfies the same induction principle as $A+B$, they are canonically equivalent.
\end{rem}

\section{Well-Founded Trees}

\textsl{Well-founded trees}, a.k.a.~\textsl{$\fun W$-trees}, generalize inductive data structures such as natural numbers and lists. The term ``well-founded'' means that there are no infinite downward paths in these trees.

A $\fun W$-tree has two basic ingredients: a \textsl{label} type $A$ and an \textsl{arity} type family $B\colon A\to\univ$. The $\fun W$-type models $\fun W$-trees. It is denoted $\fun W_{x\colon A}B(x)$, or simply $\fun W_AB$.

The term-construction process has two steps. First, we choose a label $a\colon A$ for the root. The type $A$ encodes the possible node constructors.

Second, the type family $B(a)$ dictates the branches hanging from that node. If $B(a)$ is the empty type, the node is a leaf: it has zero branches. If $B(a)$ is the boolean type, the node has two branches, and so forth. 

If the node has branches, a fully formed subtree hangs from each of them. As illustrated below, the branches are indexed by the elements $b_1,b_2,\dots,b$ of $B(a)$. The function $\phi\colon B(a)\to\fun W_{x\colon A}B(x)$ attaches a subtree $\phi(b)$ to each branch, represented by the dotted triangles.
\[
    \begin{tikzpicture}[>=stealth, thick]
        % Root node labeled by a in A
        \node[circle, draw, minimum size=8mm] (a) at (0,0) {$a$};
    
        % Apexes of the subtrees
        \coordinate (t1) at (-2.5, -2);
        \coordinate (t2) at (-0.7, -2);
        \coordinate (tb) at (2.5, -2);
    
        % Branches indexed by elements of B(a)
        \draw[->] (a) -- node[above left] {$b_1$} (t1);
        \draw[->] (a) -- node[left] {$b_2$} (t2);
        \draw[->] (a) -- node[above right] {$b$} (tb);
    
        % Ellipsis indicating B(a)-many arguments
        \node at (1, -2) {$\cdots$};
    
        % Dotted triangles representing the well-founded subtrees \phi(b_i)
        \draw[dotted, thick] (t1) -- ++(-0.8, -1.5) -- ++(1.6, 0) -- cycle;
        \node at (-2.5, -2.9) {$\scriptstyle\phi(b_1)$};
    
        \draw[dotted, thick] (t2) -- ++(-0.8, -1.5) -- ++(1.6, 0) -- cycle;
        \node at (-0.7, -2.9) {$\scriptstyle\phi(b_2)$};
    
        \draw[dotted, thick] (tb) -- ++(-0.8, -1.5) -- ++(1.6, 0) -- cycle;
        \node at (2.5, -2.9) {$\scriptstyle\phi(b)$};
    \end{tikzpicture}
\]
The induction principle of the $\fun W$-type ensures that each of these triangles is itself a well-founded tree. Thus, the branching process repeats inside each triangle and terminates at leaves after a finite number of steps.

Because $\fun W$-types serve as a universal generalization of inductive types, all the distinct constructors of a given type are unified into a single constructor, namely the function:
\begin{equation}\label{eq:sup-constructor}
    \fun{sup}\colon
        \prod_{a\colon A}\;
        \bigl(
            B(a)\to\fun W_AB
        \bigr)
            \to\fun W_AB.
\end{equation}
When a term is built using $\fun{sup}(a,\phi)$, the label $a\colon A$ acts as the indicator for which constructor is being applied, while the function $\phi\colon B(a)\to\fun W_AB$ supplies the subtrees required by that specific constructor.

In this way, the single constructor $\fun{sup}$ expresses the behavior of any number of individual constructors by delegating the choice of the operation to the label type~$A$.

\begin{rem}\label{rem:rec(sup)-is-P(W)->W}
    According to \nref{lpar:Sigma-type-rec-and-ind}, the constructor $\fun{sup}$ given by \eqref{eq:sup-constructor} determines a function
    \begin{equation}\label{eq:P(W)->W}
        \fun{rec}_{\sum_{a\colon A}B(a)\to\fun W_AB}
            (
                \fun W_AB,
                \fun{sup}
            )
            \colon
            \Bigl(
                \sum_{a\colon A}B(a)\to\fun W_AB
            \Bigr)
            \to\fun W_AB.
    \end{equation}
\end{rem}

This strictly separates the non-recursive data of a constructor from its recursive structure. The elements act as labels belonging to a type $A$, and the shape of the recursive arguments is dictated by an \textsl{arity} type family $B\colon A\to\univ$.

For $\fun{cons}$, we have two arguments: $x\colon X$ and $\ell\colon\fun L_X$. \section{{\normalfont\textsf W}-Types}

\begin{defn}
    The formal definition of the $\fun W$-type is given by its rules for formation, introduction, elimination, and computation. Let $A\colon\univ$ be a type and $B\colon A\to\univ$ be a type family.
    
    \textsc{Formation:} The type is formed by the binder $\fun W$, yielding a type in the universe:
    \[
        \fun W_{x\colon A}B(x)\colon\univ.
    \]
    
    \textsc{Introduction:} The elements of the $\fun W$-type are built using the constructor $\fun{sup}$. Given a label $a\colon A$ and a subtree-assigning function $f\colon B(a)\to \fun W_AB$, we can form a new tree $\fun{sup}(a,f)$, where
    \begin{equation}\label{eq:W-constructor}
        \fun{sup}\colon
            \prod_{a\colon A}\;
            \bigl(
                B(a)\to\fun W_AB
            \bigr)
                \to\fun W_AB.
    \end{equation}
    
    \textsc{Elimination:} To construct a dependent function over the $\fun W$-type, we use its induction principle. Given a motive $E\colon\fun W_AB\to\univ$, we must provide an inductive step function $e_s$ that constructs a term in $E(\fun{sup}(a,\phi))$ assuming we already have terms in $E(\phi(b))$ for all $b\colon B(a)$. The required signature for $e_s$ is
    \begin{equation}\label{eq:ind_W-step-function}
        e_s\colon
            \prod_{a\colon A}\;
            \prod_{\phi\colon B(a)\to \fun W_AB}
                \Bigl(\prod_{b\colon B(a)}E(\phi(b))\Bigr)
                \to E(\fun{sup}(a,\phi)).
    \end{equation}
    With this data, the induction principle produces a dependent function
    \begin{equation}\label{eq:W-induction-function}
        \fun{ind}_{\fun W}(E, e_s)
            \colon\prod_{w\colon \fun W_AB}E(w).
    \end{equation}
    As a special case, when the motive is a constant type family $E(w)\defeq C$ for some carrier type $C\colon\univ$, the inductive step function simplifies to a curried structure map $s_C\colon\prod_{a\colon A}(B(a)\to C)\to C$. The induction principle then reduces to the non-dependent \textsl{recursor}
    \begin{equation}\label{eq:rec_W}
        \fun{rec}_{\fun W}(C, s_C)
            \colon \fun W_AB\to C.
    \end{equation}
    To see how this simplification occurs, we can trace the definitional equalities. Fixing $a\colon A$, we have
    \begin{align*}
            \prod_{\phi\colon B(a)\to\fun W_AB}\,
            \Bigl(\prod_{b\colon B(a)}E(\phi(b))\Bigr)
            \to E(\fun{sup}(a,\phi))
        &\equiv
            \prod_{\phi\colon B(a)\to \fun W_AB}\,
            \Bigl(\prod_{b\colon B(a)}C\Bigr)
            \to C\\
        &\equiv
            \prod_{\phi\colon B(a)\to\fun W_AB}
            \!\!\!(B(a)\to C)\to C.
    \end{align*}
    The final expression provides access to both the recursive results $B(a)\to C$ and the original subtrees $\phi\colon B(a)\to\fun W_AB$. However, since $(B(a)\to C)\to C$ does not depend on the original trees in $\fun W_AB$, we can drop the unused parameter $\phi\colon B(a)\to\fun W_AB$ to obtain the curried structure map:
    \[
        s_C\colon\prod_{a\colon A}(B(a)\to C)\to C.
    \]

    \textsc{Computation:} The eliminators compute by evaluating the step function on the components of the $\fun{sup}$ constructor. For the induction principle, the computation rule is the definitional equality
    \begin{equation}\label{eq:ind_W-computation-rule}
        \fun{ind}_{\fun W}(E,e_s)(\fun{sup}(a,\phi)) 
            \equiv
            e_s(a,\phi,\fun{ind}_{\fun W}(E,e_s)\circ\phi).
    \end{equation}
    Accordingly, the computation rule for the recursor is
    \begin{equation}\label{eq:rec_W-computation-rule}
        \fun{rec}_{\fun W}(C,s_C)(\fun{sup}(a,\phi))
            \equiv s_C(a,\fun{rec}_{\fun W}
                (
                    C,s_C
                )\circ\phi).
    \end{equation}
\end{defn}

\begin{xmpl}\label{xmpl:List(X)-induction-from-W-induction}
    To see how this general framework operates in a concrete inductive type, let us continue with Example~\ref{xmpl:W-tree-for-List(X)}. The goal is to derive the induction principle for $\fun L_X$ from the induction principle for $\fun W$-types.

    We must construct the data required by $\fun W$-induction to yield standard list induction. This involves case analysis over the label $a\colon\fun1+X$.

    Suppose $a\equiv\fun{inl}(\fstar)$. In this case, we have $B(a)\equiv\fun0$. Consequently, the associated subtree function is the unique definitional equality $f\equiv\fun{rec}_{\fun0}(\fun W_AB)$. Moreover, $\fun{sup}(a,f)$ is definitionally equal to $\fun{nil}$.

    The inductive hypothesis is a dependent function $g\colon\prod_{b\colon\fun0}E(f(b))$. Because the product runs over the empty type, this type is equivalent to $\fun1$ and contains a unique, trivial inhabitant.

    For this case, the inductive step function of $\fun{ind}_{\fun W}$ needs to map this trivial inhabitant to a term in $E(\fun{sup}(a,f))\equiv E(\fun{nil})$. This is precisely the data required by \hyperref[itm:list-induction]{standard list induction$\,\scriptstyle[\uparrow]$}: a term $e_{\fun{nil}}\colon E(\fun{nil})$.

    In the case where $a\equiv\fun{inr}(x)$, we have $B(a)\equiv\fun1$. Thus, the subtree-assigning function has the signature $f\colon\fun1\to\fun W_{x\colon A}B(x)$. \hyperref[itm:list-from-W]{As established before$\,\scriptstyle[\uparrow]$}, the constructor application $\fun{sup}(a,f)$ is definitionally equal to $\fun{cons}(x,f(\fstar))$.

    The inductive hypothesis is a dependent function $g\colon\prod_{b\colon\fun1}E(f(b))$, which provides a single meaningful term: $g(\fstar)\colon E(f(\fstar))$.

    To complete the definition of $e_s$, we must construct a term in $E(\fun{sup}(a,f))$, which evaluates to $E(\fun{cons}(x,f(\fstar)))$. But this is exactly the data provided by the \hyperref[itm:list-induction]{standard list induction$\,\scriptstyle[\uparrow]$}: a dependent function $e_{\fun{cons}}$ that takes the head $x$, the tail $f(\fstar)$, and the inductive hypothesis $g(\fstar)$, and returns a term in $E(\fun{cons}(x,f(\fstar)))$.

    Consequently, we can fully define the inductive step function $e_s$ by pattern matching on the label $a\colon\fun1+X$:
    \begin{align*}
        e_s(\fun{inl}(\fstar), f, g) &\defeq e_{\fun{nil}}, \\
        e_s(\fun{inr}(x), f, g) &\defeq e_{\fun{cons}}(x, f(\fstar), g(\fstar)).
    \end{align*}
    It follows that the induction principle for lists is simply a specific instance of the induction principle for $\fun W$-types:
    \[
        \fun{ind}_{\fun{List}}(E, e_{\fun{nil}}, e_{\fun{cons}})
           \defeq\fun{ind}_{\fun W}(E, e_s).
    \]
\end{xmpl}

\begin{thm}\label{thm:W-type-induction-uniqueness}
    Let $g,h\colon\prod_{w\colon\fun W_AB}E(w)$ be two functions which satisfy the recurrence
    \[
        e\colon\prod_{a,f}\;
        \Bigl(
            \prod_{b\colon B(a)}E(f(b))
        \Bigr)
        \to E(\fun{sup}(a,f))
    \]
    propositionally, i.e., such that
    \begin{align*}
        \prod_{a,f} g(\fun{sup}(a,f))
            &\eq{E(\fun{sup}(a,f))}
                e(a,f,\lambda b.\,g(f(b))),\\
        \prod_{a,f} h(\fun{sup}(a,f))
            &\eq{E(\fun{sup}(a,f))}
                e(a,f,\lambda b.\,h(f(b))).
    \end{align*}
    Then $g\sim h$.
\end{thm}

\begin{proof}
    Let $C\colon\fun W_AB\to\univ$ be the type family defined by
    \[
        C(w)\defeq g(w)\eq{E(w)}h(w).
    \]
    We must construct a witness for $\prod_{w\colon\fun W_AB}C(w)$. By the induction principle of $\fun W$-types, this reduces to defining a step function \eqref{eq:ind_W-step-function}
    \[
        c_s(a,f)\colon
            (g\circ f\sim h\circ f)
            \to
                \bigl(
                    g(\fun{sup}(a,f))
                    \eq{E(\fun{sup}(a,f))}
                    h(\fun{sup}(a,f))
                \bigr)
    \]
    for $a\colon A$ and $f\colon B(a)\to\fun W_AB$.

    Let $\theta\colon g\circ f\sim h\circ f$. Applying the action on paths of the function
    \[
        e(a,f)\colon
            \Bigl(
                \prod_{b\colon B(a)}E(f(b))
            \Bigr)
            \to E(\fun{sup}(a,f))
    \]
    to the path
    \[
        \fun{funext}(\theta)\colon
            g\circ f
            \eq{\prod_{b\colon B(a)}E(f(b))}
            h\circ f
    \]
    yields
    \[
        \fun{ap}_{e(a,f)}(\fun{funext}(\theta))
            \colon e(a,f,g\circ f)
            \eq{E(\fun{sup}(a,f))}
            e(a,f,h\circ f).
    \]
    By hypothesis, we also have paths
    \begin{align*}
        u(a,f)&\colon g(\fun{sup}(a,f))
        \eq{E(\fun{sup}(a,f))}
        e(a,f,g\circ f)
    \intertext{and}
        v(a,f)&\colon h(\fun{sup}(a,f))
        \eq{E(\fun{sup}(a,f))}
        e(a,f,h\circ f).
    \end{align*}
    Therefore, we can define the target path by concatenation:
    \[
        c_s(a,f,\theta)\defeq u(a,f)
            \ct\fun{ap}_{e(a,f)}(\fun{funext}(\theta))
            \ct v(a,f)^{-1}.
    \]
\end{proof}

\section{Initial Algebras}

\begin{defns}\ 
    \begin{enumerate}[-]
            \item An \textsl{$\fun L_X$-algebra} is a type $C$ provided with an element $c_{\fun{nil}}\colon C$ and a function $c_{\fun{cons}}\colon X\to C\to C$. The type of such algebras is
            \[
                \fun L_X\fun{Alg}\defeq\sum_{C\colon\univ}C\times(X\to C\to C).
            \]
            \item Let $\mathcal C\defeq\dep{C,(c_{\fun{nil}},c_{\fun{cons}})}$ and $\mathcal D\defeq\dep{D,(d_{\fun{nil}}, d_{\fun{cons}})}$ be two $\fun L_X$-algebras. An \textsl{$\fun L_X$-morphism} between them is a map $h\colon C\to D$ such that $h(c_{\fun{nil}})\eq Dd_{\fun{nil}}$ and $h(c_{\fun{cons}}(x,c))\eq Dd_{\fun{cons}}(x,h(c))$ for all $x\colon X$ and $c\colon C$. The type of such morphisms is
        \[
            \fun L_X\Hom(\mathcal C,\mathcal D)\defeq
            \sum_{h}
                \bigl(
                    h(c_{\fun{nil}})
                    \eq D
                    d_{\fun{nil}}
                \bigr)
                \times
                \prod_{x,c}
                    h(c_{\fun{cons}}(x,c))
                    \eq D
                    d_{\fun{cons}}(x, h(c)).
        \]
    
        \item An $\fun L_X$-algebra $\mathcal I$ is \textsl{homotopy-initial} (a.k.a.~\textsl{h-initial}) if for any $\fun L_X$-algebra $\mathcal C$ the type of $\fun L_X$-morphisms from $\mathcal I$ to $\mathcal C$ is contractible, i.e.,
        \[
            \fun{isHinit}_{\fun L_X}(\mathcal I)
                \defeq
                \prod_{\mathcal C\colon\fun L_X\fun{Alg}}
                    \fun{isContr}(
                        \fun L_X\Hom(\mathcal I,\mathcal C)
                    )
        \]
        is inhabited.
    \end{enumerate}
\end{defns}

\begin{lem}\label{lem:L_X-morphisms-are-morphisms}
    Given $\fun L_X$-algebras $\mathcal{A}\defeq\dep{A,(a_{\fun{nil}}, a_{\fun{cons}})}$, $\mathcal{B}\defeq\dep{B,(b_{\fun{nil}},b_{\fun{cons}})}$, and $\mathcal{C}\defeq\dep{C,(c_{\fun{nil}}, c_{\fun{cons}})}$, the following hold:
    \begin{enumerate}[a),font=\upshape]
        \item The identity function $\id_A\colon A\to A$ is an $\fun L_X$-morphism from $\mathcal{A}$ to $\mathcal{A}$.
        \item If $f\colon A\to B$ and $g\colon B\to C$ are $\fun L_X$-morphisms, then the composition $g\circ f\colon A\to C$ is an $\fun L_X$-morphism from $\mathcal{A}$ to $\mathcal{C}$.
        \item If $f\colon A\to B$ is an $\fun L_X$-morphism and $g\colon B\to A$ is a quasi-inverse of $f$, then the quasi-inverse $g$ is an $\fun L_X$-morphism from $\mathcal{B}$ to $\mathcal{A}$.
    \end{enumerate}
\end{lem}

\begin{proof}\ 
    \begin{enumerate}[a)]
        \item We need to show that $\id_A(a_{\fun{nil}})
        \eq{}a_{\fun{nil}}$ and $\id_A(a_{\fun{cons}}(x,a))\eq{}a_{\fun{cons}}(x,\id_A(a))$ for all $x\colon X$ and $a\colon A$. Both equalities hold definitionally.
        
        \item For the base case, we have
        \[
            (g\circ f)(a_{\fun{nil}})
                \equiv g(f(a_{\fun{nil}}))
                \eq Cg(b_{\fun{nil}})
                \eq Cc_{\fun{nil}}.
        \]
        For the inductive step, let $x\colon X$ and $a\colon A$. Then,
        \begin{align*}
            (g\circ f)(a_{\fun{cons}}(x,a))
                &\equiv g(f(a_{\fun{cons}}(x,a)))\\
                &\eq Cg(b_{\fun{cons}}(x,f(a)))\\
                &\eq Cc_{\fun{cons}}(x,g(f(a)))\\
                &\equiv c_{\fun{cons}}(x,(g\circ f)(a)).
        \end{align*}
        
        \item We have homotopies $\eta\colon f\circ g\sim\id_B$ and $\vartheta\colon g\circ f\sim\id_A$. We must show that $g$ preserves the algebra constructors. For the base case, we apply $\fun{ap}_g$ to the equality $f(a_{\fun{nil}})\eq Bb_{\fun{nil}}$ to obtain $g(f(a_{\fun{nil}}))\eq Ag(b_{\fun{nil}})$. By the homotopy $\vartheta$, we know $g(f(a_{\fun{nil}}))\eq Aa_{\fun{nil}}$. Transitivity of equality thus yields $g(b_{\fun{nil}})\eq Aa_{\fun{nil}}$.
        
        For the inductive step, let $x\colon X$ and $b\colon B$. Since $f$ is an $\fun L_X$-morphism, we have $f(a_{\fun{cons}}(x,g(b)))\eq Bb_{\fun{cons}}(x, f(g(b)))$. The homotopy $\eta$ implies $f(g(b))\eq Bb$. Then substitution gives
        \[
            f(a_{\fun{cons}}(x,g(b)))
            \eq Bb_{\fun{cons}}(x,b).
        \]
        Applying $g$ to both sides yields
        \[
            g(f(a_{\fun{cons}}(x,g(b))))
            \eq Ag(b_{\fun{cons}}(x,b)).
        \]
        By the homotopy $\vartheta$, the \lhs is equal to $a_{\fun{cons}}(x,g(b))$. Therefore, we conclude
        \[
            a_{\fun{cons}}(x,g(b))
            \eq Ag(b_{\fun{cons}}(x,b)),
        \]
        which shows that $g$ is an $\fun L_X$-morphism.\qedhere
    \end{enumerate}
\end{proof}

\begin{thm}
    The type of $h$-initial $\fun L_X$-algebras is a mere proposition.
\end{thm}

\begin{proof}
    Let $\mathcal I$ and $\mathcal J$ be two $h$-initial $\fun L_X$-algebras. Then, $\fun L_X\Hom(\mathcal I,\mathcal J)$ is contractible, hence inhabited by, say, $f\colon\mathcal I\to\mathcal J$. By symmetry, there is also $g\colon\mathcal J\to\mathcal I$. Since  Lemma~\ref{lem:L_X-morphisms-are-morphisms} implies that $g\circ f$ and $\id_{\mathcal I}$ inhabit $\fun L_X\Hom(\mathcal I,\mathcal I)$, which is contractible, $g\circ f\eq{\fun L_X\Hom(\mathcal I,\mathcal I)}\id_{\mathcal I}$. The same reasoning proves that $f\circ g\eq{\fun L_X\Hom(\mathcal J,\mathcal J)}\id_{\mathcal J}$.

    It follows that $f$ and $g$ are quasi-inverses and so $\mathcal I\simeq\mathcal J$. The conclusion follows by the univalence axiom.
\end{proof}

\begin{thm}
    The $\fun L_X$-algebra $\dep{\fun L_X,(\fun{nil},\fun{cons})}$ is homotopy initial.
\end{thm}

\begin{proof}
    Let $\mathcal C\defeq\dep{C,(c_{\fun{nil}},c_{\fun{cons}})}$ be an $\fun L_X$-algebra.

    Define $u\colon\fun L_X\to C$ inductively by
    \begin{itemize}
        \item $u(\fun{nil})\defeq c_{\fun{nil}}$,
        \item $u(\fun{cons}(a,\ell))\defeq c_{\fun{cons}}(a,u(\ell))$.
    \end{itemize}
    By definition, $u$ is an $\fun L_X$-morphism.

    Given any other $\fun L_X$-morphism $v\colon\fun L_X\to C$, we have
    \begin{itemize}
        \item $v(\fun{nil})\eq{C} c_{\fun{nil}}$,
        \item $v(\fun{cons}(a,\ell))\eq{C} c_{\fun{cons}}(a,v(\ell))$.
    \end{itemize}
    Therefore, by Remark~\ref{rem:List-induction-uniqueness}, we deduce that $u\sim v$, hence $u\eq{\fun L_X\Hom(\fun L_X,C)}v$. Consequently, $\fun L_X\Hom(\fun L_X,C)$ is contractible with center $u$.
    %
    
\end{proof}

\begin{cor}
    The $\fun L_{\fun1}$-algebra $\dep{\N,(0, \fun{succ})}$ is homotopy initial.
\end{cor}

\begin{proof}
    By the theorem, the $\fun L_{\fun1}$-algebra $\dep{\fun L_{\fun1},(\fun{nil}, \fun{cons})}$ is $h$-initial. Since we have established in Example~\ref{xmpl:N-from-List(1)} the equality $\N\eq{\univ}\fun L_{\fun1}$, which identifies $0$ with $\fun{nil}$ and $\fun{succ}(n)$ with $\fun{cons}(\fstar,n)$, this yields a path
    \[
        q\colon
        \dep{\N,(0,\fun{succ})}
        \eq{\fun L_{\fun1}\fun{Alg}}
        \dep{\fun L_{\fun1},(\fun{nil},\fun{cons})}.
    \]
    If we define the type family $Q\colon\fun L_{\fun1}\fun{Alg}\to\univ$ by $Q(\mathcal{A})\defeq\fun{isHinit}_{\fun L_{\fun1}}(\mathcal{A})$, then
    \[
        \fun{transport}^Q(q^{-1}) \colon 
        \fun{isHinit}_{\fun L_{\fun1}}(\dep{\fun L_{\fun1},(\fun{nil},\fun{cons})}) 
        \to 
        \fun{isHinit}_{\fun L_{\fun1}}(\dep{\N,(0,\fun{succ})}).
    \]
    Applying this function to the fact that $\dep{\fun L_{\fun1},(\fun{nil},\fun{cons})}$ is $h$-initial provides the desired inhabitant.
\end{proof}

\begin{defn}
    Let $A\colon\univ$ be a type of labels and $B\colon A\to\univ$ an arity family. A \textsl{polynomial functor} (in normal form) is the type family
    \[
        P(X)\defeq\sum_{a\colon A}B(a)\to X
    \]
    for any $X\colon\univ$.
\end{defn}

\begin{rem}
    The notion of polynomial functor can be seen as a generalization of \eqref{eq:P(W)->W}. The polynomial aspect of $P(X)$ becomes clear when $B(a)\to X$ is written as $X^{B(a)}$.
\end{rem}

\begin{defn} 
    The type of \textsl{$\fun W$-algebras} is
    \[
        \fun W\fun{Alg}(A,B)
            \defeq\sum_{C\colon\univ}\;
                \prod_{a\colon A}(B(a)\to C)\to C.
    \]
    Its terms are dependent pairs $\dep{C,s_C}$, where $s_C\colon\prod_{a\colon A}(B(a)\to C)\to C$ is the \textsl{structure map} of the $\fun W$-algebra.
\end{defn}

\begin{rem}
    A $\fun W$-algebra $\dep{C,s_C}$ admits an \textsl{uncurried variant} $\dep{C,\bar{s}_C}$, where
    \[
        \bar{s}_C\defeq\fun{rec}_{P(C)}(C,s_C)
            \colon P(C)\to C.
    \]
    We will refer to this form as a \textsl{$P$-algebra}.
    
    These two views are interchangeable since $s_C$ can be recovered from $\bar{s}_C$ by the computation rule of the $\Sigma$-type.
\end{rem}

\needspace{2\baselineskip}
\begin{defns}\ 
    \begin{enumerate}[-]
        \item Given $P$-algebras $\dep{C,\bar{s}_C}$ and $\dep{D,\bar{s}_D}$, any map $f\colon C\to D$ induces a function $Pf\colon P(C)\to P(D)$ defined by
        \[
            Pf(\dep{a,\phi})\defeq\dep{a,f\circ\phi},
        \]
        for $a\colon A$ and $\phi\colon B(a)\to C$.
        
        \item A \textsl{$P$-morphism} from a $P$-algebra $\dep{C,\bar{s}_C}$ to another $\dep{D,\bar{s}_D}$ is a dependent pair $\dep{f,\eta_f}$, where $f\colon C\to D$ and $\eta_f\colon f\circ\bar{s}_C\sim\bar{s}_D\circ Pf$. This homotopy can be visualized in the square:
        \[
            \begin{tikzcd}[font=\small]
                P(C)\arrow[r,"Pf"]
                        \arrow[d,"\bar{s}_C"']
                        \arrow[dr,
                            phantom,
                            "\scriptstyle\eta_f"]
                    &P(D)
                        \arrow[d,"\bar{s}_D"]\\
                C
                        \arrow[r,"f"']
                    &D
            \end{tikzcd}
        \]
        Thus, the type of $P$-morphisms is
        \[
            \fun P\Hom(\dep{C,\bar{s}_C},\dep{D,\bar{s}_D})
                \defeq\sum_{f\colon C\to D}
                    f\circ\bar{s}_C\sim\bar{s}_D\circ Pf.
        \]
        Equivalently, using the curried variants $s_C$ and $s_D$, the homotopy condition is given by a family of identities, making the type of morphisms definitionally equivalent to
        \begin{equation}\label{eq:structure-coherence}
            \sum_{f\colon C\to D}\;
                \prod_{a\colon A}\;
                \prod_{\phi\colon B(a)\to C}
                    f(s_C(a,\phi))\eq Ds_D(a,f\circ\phi).
        \end{equation}
        We will denote the type of $\fun W$-morphisms by $\fun W\Hom(\dep{C,s_C},\dep{D,s_D})$, or $\fun W\Hom(C,D)$ for short, if the context permits.

        \item A $\fun W$-algebra $\dep{I,s_I}$ is \textsl{$h$-initial} if the type
        \[
            \fun{isHinit}_{\fun W}(A,B,\dep{I,s_I})
            \defeq
            \prod_{\dep{C,s_C}\colon\fun W\fun{Alg}(A,B)}
            \!\!\!\fun{isContr}
            \bigl(
                \fun W\Hom(I,C)
            \bigr)
        \]
        is inhabited.
    \end{enumerate}
\end{defns}

\begin{defn}
    For a $\fun W$-type $\fun W\defeq\fun W_{x\colon A}B(x)$, the \textsl{recursor} is the function
    \[
        \fun{rec}_{\fun W}\colon
            \prod_{C\colon\univ}\;
            \Bigl(
                \prod_{a\colon A}(B(a)\to C)\to C
            \Bigr)
            \to\fun W\to C.
    \]
    For any type $C\colon\univ$ and structure map $s_C\colon\prod_{a\colon A}(B(a)\to C)\to C$, the recursion map $\fun{rec}_{\fun W}(C, s_C)\colon\fun W\to C$ is defined inductively on constructors by the rule
    \[
        \fun{rec}_{\fun W}(C,s_C)(\fun{sup}(a,\phi))
        \defeq
        s_C(a,\fun{rec}_{\fun W}(C,s_C)\circ\phi),
    \]
    for any $a\colon A$ and $\phi\colon B(a)\to\fun W$.
\end{defn}

\begin{ntn}\label{ntn:recursor-commutativity}
    For $\fun W\defeq\fun W_AB$, a carrier type $C\colon\univ$, and a curried structure map $s_C\colon\prod_{a\colon A}(B(a)\to C)\to C$, the term
    \[
        \fun{comp}_{\fun W}(C,s_C,a,\phi)
        \defeq\fun{refl}_{f(\fun{sup}(a,\phi))}
    \]
    denotes the witness for the computation rule \eqref{eq:rec_W-computation-rule}. Specifically, it is an inhabitant of the identity type
    \begin{equation}\label{eq:recursor-commutativity}
        \fun{rec}_{\fun W}(C,s_C)(\fun{sup}(a,\phi))
        \eq C
        s_C(a,\fun{rec}_{\fun W}(C,s_C)\circ\phi).
    \end{equation}
\end{ntn}

\begin{lem}\label{lem:transport-identity-family}
    Let $H, C\colon\univ$ be types and $L, R\colon H \to C$ functions. Given a path $p\colon h_1\eq{H} h_2$ and a path $q\colon L(h_1)\eq CR(h_1)$, we have
    \[
        \fun{transport}^{h\mapsto L(h)\eq CR(h)}(p,q) 
        \eq{L(h_2)\eq CR(h_2)}
        \fun{ap}_L(p)^{-1}\ct q\ct\fun{ap}_R(p).
    \]
\end{lem}

\begin{proof}
    First observe that, in fact, both sides of the target identity inhabit the type $L(h_2)\eq CR(h_2)$. Hence, we can apply path induction on $p$. It suffices to consider the case where $h_2\equiv h_1$ and $p\equiv\fun{refl}_{h_1}$. 
    
    In this case, the \lhs becomes definitionally $q$:
    \[
        \fun{transport}^{h \mapsto L(h) \eq{C} R(h)}(\fun{refl}_{h_1}, q) \equiv q.
    \]
    Since $\fun{ap}_L(\fun{refl}_{h_1})\equiv \fun{refl}_{L(h_1)}$ and $\fun{ap}_R(\fun{refl}_{h_1})\equiv \fun{refl}_{R(h_1)}$, the \rhs reduces to
    \[
        \fun{refl}_{L(h_1)}^{-1}
            \ct q
            \ct\fun{refl}_{R(h_1)}.
    \]
    The conclusion follows by the unit laws for path concatenation.
\end{proof}

\begin{lem}\label{lem:WHom-equality}
    Let $\dep{D,s_D}$ and $\dep{C,s_C}$ be two $\fun W$-algebras. Let $\dep{f,\eta_f}$ and $\dep{g,\eta_g}$ be two $\fun W$-morphisms from $\dep{D,s_D}$ to $\dep{C,s_C}$. The identity type
    \[
        \dep{f,\eta_f}\eq{\fun W\Hom(D,C)}\dep{g,\eta_g}
    \]
    is equivalent to the type of dependent pairs $\dep{e,s_e}$, where $e\colon f\sim g$ is a homotopy and $s_e$ is a family of paths assigning to each $a\colon A$ and $\phi\colon B(a)\to D$ a path
    \[
        s_e(a,\phi)\colon
            e(s_D(a,\phi))
            \eq{}
            \eta_f(a,\phi)
            \ct\fun{ap}_{s_C(a,\dummy)}(
                \fun{funext}(e\rwct\phi))
            \ct\eta_g(a,\phi)^{-1}
    \]
    in $f(s_D(a,\phi)) \eq{C} g(s_D(a,\phi))$. In other words, the path diagram
    \[
        \begin{tikzcd}[row sep=large, column sep=3.2cm]
            f(s_D(a,\phi))
                    \arrow[r,"{e(s_D(a,\phi))}"]
                    \arrow[d,"{\eta_f(a,\phi)}"']
                    \arrow[rd,
                        "{\scriptstyle s_e(a,\phi)}",
                        phantom]
                &g(s_D(a,\phi))
                    \arrow[d,gray,"{\eta_g(a,\phi)}"]\\
            s_C(a,f\circ\phi)
                    \arrow[r,
                        "{\fun{ap}_{s_C(a,\dummy)}
                            (\fun{funext}(e\rwct\phi))
                        }",
                        swap]
                &s_C(a,g\circ\phi)
                    \arrow[u,
                        bend left=50,
                        "{\eta_g(a,\phi)^{-1}}"]
        \end{tikzcd}
    \]
    commutes propositionally if, and only if, $\dep{f,\eta_f}\eq{\fun W\Hom(D,C)}\dep{g,\eta_g}$.
\end{lem}

\begin{proof}
    Equating the dependent pairs $\dep{f,\eta_f}$ and $\dep{g,\eta_g}$ in the $\Sigma$-type of $\fun W$-morphisms $\fun W\Hom(D,C)$ is equivalent to providing a path $p\colon f\eq{D\to C}g$ and a path $q$ witnessing the identity between the transport of $\eta_f$ along $p$ and $\eta_g$.
    
    By function extensionality, the path $p$ is equivalent to a homotopy $e\colon f\sim g$, giving $p\defeq\fun{funext}(e)$.
    
    For any $a\colon A$ and $\phi\colon B(a)\to D$, let us define the local terms
    \[
        L_{a,\phi}(h)\defeq h(s_D(a,\phi))
            \quad\text{and}\quad
        R_{a,\phi}(h)\defeq s_C(a,h\circ\phi).
    \]
    This allows us to define the dependent type family $E$ as
    \[
        E(h)\defeq
            \prod_{a\colon A}\;
            \prod_{\phi\colon B(a)\to D}
                L_{a,\phi}(h)\eq CR_{a,\phi}(h).
    \]
    According to \eqref{eq:structure-coherence}, we have $\eta_f\colon E(f)$ and $\eta_g\colon E(g)$, and the transport condition mentioned above translates to
    \[
        \fun{transport}^E(p,\eta_f)\eq{}\eta_g.
    \]
    By applying Proposition~\ref{prop:transport-pointwise-evaluation} to the outer product over $A$, we obtain
    \[
        \fun{transport}^E(p,\eta_f)
        \eq{}
        \lambda a.\,\fun{transport}^{
            \lambda h.\,\prod_{\phi\colon B(a)\to D}
                L_{a,\phi}(h)\eq CR_{a,\phi}(h)
            }
            (p,\eta_f(a)).
    \]
    Applying the proposition a second time to the inner product over $B(a)\to D$ yields
    \[
        \fun{transport}^E(p,\eta_f)\eq{}
        \lambda a.\,\lambda \phi.\, 
            \fun{transport}^{
                \lambda h.\,L_{a,\phi}(h)
                \eq CR_{a,\phi}(h)
            }
            (p,\eta_f(a,\phi)).
    \]
    By Lemma~\ref{lem:transport-identity-family}, we obtain
    \[\tag{$\ast$}
        \fun{transport}^{h \mapsto L_{a,\phi}(h) \eq{C} R_{a,\phi}(h)}(p, \eta_f(a,\phi)) 
        \eq{} \fun{ap}_{L_{a,\phi}}(p)^{-1} \ct \eta_f(a,\phi) \ct \fun{ap}_{R_{a,\phi}}(p).
    \]
    \needspace{2\baselineskip}
    Computing the action of $\fun{ap}_{L_{a,\phi}}$ and $\fun{ap}_{R_{a,\phi}}$ for $p\defeq\fun{funext}(e)$ gives:
    \begin{enumerate}[a),font=\upshape]
        \item $\fun{ap}_{L_{a,\phi}}(\fun{funext}(e))\eq{} e(s_D(a,\phi))$.
        \item $\fun{ap}_{R_{a,\phi}}(\fun{funext}(e))\eq{} \fun{ap}_{s_C(a,\dummy)}(\fun{funext}(e\rwct\phi))$.
    \end{enumerate}
    Substituting these into the transport equation ($\ast$) yields
    \[
        e(s_D(a,\phi))^{-1}
        \ct\eta_f(a,\phi)
        \ct\fun{ap}_{s_C(a,\dummy)}
            (
                \fun{funext}(e\rwct\phi)
            ).
    \]
    The coherence condition requires this transported path to equal $\eta_g(a,\phi)$:
    \[
        e(s_D(a,\phi))^{-1} \ct \eta_f(a,\phi) \ct \fun{ap}_{s_C(a,\dummy)}(\fun{funext}(e\rwct\phi)) \eq{} \eta_g(a,\phi).
    \]
    Concatenating on the left by $e(s_D(a,\phi))$ and on the right by $\eta_g(a,\phi)^{-1}$ isolates $e(s_D(a,\phi))$, yielding the required coherence condition:
    \[
        e(s_D(a,\phi))
        \eq{}
        \eta_f(a,\phi)
            \ct\fun{ap}_{s_C(a,\dummy)}
                (
                    \fun{funext}(e\rwct\phi)
                )
            \ct\eta_g(a,\phi)^{-1}.
    \]
    This completes the proof of the \textit{if} part.
    
    For the \textit{only if} part, note that the equality $\dep{f,\eta_f}\eq{}\dep{g,\eta_g}$ means the existence of paths
    \begin{align*}
        p&\colon f\eq{D\to C}g,\\
        q&\colon
            \fun{transport}^E(p,\eta_f)
            \eq{E(g)}
            \eta_g.
    \end{align*}
    Thus, we can apply $\fun{happly}$ to $q$ to translate this global equality into the local path:
    \[
        \fun{happly}(q,a)\colon
            \fun{transport}^E(p,\eta_f)(a)
            \eq{B(a)\to D}
            \eta_g(a).
    \]
    By applying $\fun{happly}$ a second time we obtain
    \[
        \fun{happly}(\fun{happly}(q,a),\phi)\colon
        \fun{transport}^E(p,\eta_f)(a,\phi)
        \eq{}
        \eta_g(a,\phi).
    \]
    According to Proposition~\ref{prop:transport-pointwise-evaluation} and Lemma~\ref{lem:transport-identity-family}, the left-hand side reduces point-wise. Thus, this double application of $\fun{happly}$ yields the local path 
    \[
        e(s_D(a,\phi))^{-1}
        \ct\eta_f(a,\phi)
        \ct\fun{ap}_{s_C(a,\dummy)}
            (
                \fun{funext}(e\rwct\phi)
            )\eq{}\eta_g(a,\phi),
    \]
    which we can use to derive $s_e(a,\phi)$.
\end{proof}

\begin{thm}
    For any type $A\colon\univ$ and type family $B\colon A\to\univ$, the $\fun W$-algebra $\dep{\fun W_AB,\fun{sup}}$ is $h$-initial.
\end{thm}

\begin{proof}
    Let $\fun W\defeq\fun W_AB$. We must show that for any $\fun W$-algebra $\dep{C,s_C}$, the type $\fun W\Hom(\fun W,C)$ is contractible. 
    
    For the center of contraction $\dep{f,\eta_f}$, let $f\colon\fun W\to C$ be defined by \eqref{eq:rec_W}, i.e.,
    \[
        f\defeq\fun{rec}_{\fun W}(C,s_C).
    \]
    The homotopy condition \eqref{eq:structure-coherence} for $\eta_f$ requires that, for $a\colon A$ and $\phi\colon B(a)\to\fun W$, we have
    \[
        f(\fun{sup}(a,\phi))\eq Cs_C(a,f\circ\phi).
    \]
    This equality is exactly the computation rule \eqref{eq:recursor-commutativity} for the $\fun W$-recursor. Thus, using Notation~\ref{ntn:recursor-commutativity}, we define $\eta_f(a,\phi)\defeq\fun{comp}_{\fun W}(C,s_C,a,\phi)$.

    To prove contractibility, it remains to show that for any other $\fun W$-morphism $\dep{g,\eta_g}$, there is an identity $\dep{f,\eta_f}\eq{\fun W\Hom(\fun W,C)}\dep{g,\eta_g}$. 
    
    By Lemma~\ref{lem:WHom-equality}, providing such an identity is equivalent to providing a dependent pair $\dep{e,s_e}$, where $e\colon f\sim g$ and $s_e(a,\phi)$ witnesses the equality
    \[
        e(\fun{sup}(a,\phi))\eq{}
        \eta_f(a,\phi)
        \ct\fun{ap}_{s_C(a,\dummy)}
            (\fun{funext}(e\rwct\phi))
        \ct \eta_g(a,\phi)^{-1}.
    \]
    To construct the homotopy $e$, we proceed by $\fun W$-induction for the type family $E\colon\fun W\to\univ$ defined by
    \[
        E(w)\defeq f(w)\eq Cg(w).
    \]
    According to \eqref{eq:ind_W-step-function}, given
    \[
        \epsilon\colon\prod_{b\colon B(a)}
            f(\phi(b))\eq Cg(\phi(b)),
    \]
    we need to provide a path in $f(\fun{sup}(a,\phi))\eq Cg(\fun{sup}(a,\phi))$. We define
    \[
        e_s(a,\phi,\epsilon)\defeq
            \eta_f(a,\phi)
            \ct\fun{ap}_{s_C(a,\dummy)}
                (\fun{funext}(\epsilon))
            \ct\eta_g(a,\phi)^{-1}.
    \]
    By the computation rule \eqref{eq:ind_W-computation-rule}, the resulting homotopy $e\colon f\sim g$ satisfies
    \[
        e(\fun{sup}(a,\phi))\eq{}e_s(a,\phi,e\rwct\phi),
    \]
    since $e\rwct\phi\equiv e\circ\phi$.
    
    Thus, the required coherence condition $s_e$ of Lemma~\ref{lem:WHom-equality} is satisfied, yielding the required identity $\dep{f,\eta_f}\eq{\fun W\Hom(\fun W,C)}\dep{g,\eta_g}$.
    %
    
\end{proof}


\section{Homotopy-Inductive Types}

Let us get back to the alternative definition $\N'$ of the natural numbers given in Example~\ref{xmpl:W-tree-for-N}, namely
\begin{itemize}
    \item $\N'\defeq\fun W_{\fun2}\fun{rec}_{\fun2}(\univ,\fun0,\fun1)$,
    \item $0'\defeq\fun{sup}(0_{\fun2},\fun{rec}_{\fun0}(\N'))$, and
    \item $\fun{succ}'(n)\defeq\fun{sup}(1_{\fun2},\lambda{\fstar}.\,n)$.
\end{itemize}
To see if the standard computation rules for natural numbers can be derived from the rules for $\fun W$-types, we must define the step function $e_s$ as specified in \eqref{eq:ind_W-step-function}, i.e.,
\[
    e_s\colon
    \prod_{a\colon\fun2}\;
    \prod_{\phi\colon B(a)\to\N'}
    \Bigl(\prod_{b\colon B(a)}E(\phi(b))\Bigr)
    \to E(\fun{sup}(a,\phi)).
\]
Consequently, we assume we are given the standard inductive data for the natural numbers: a motive $E$, a base case $e_{0'}\colon E(0')$, and a successor step function
\[
    e_{\fun{succ}'}\colon\prod_{n\colon\N'}E(n)
    \to E(\fun{sup}(1_{\fun2},\lambda{\fstar}.\,n)).
\]
To derive the standard induction principle for $\N'$ from the $\fun W$-induction, we must define $\fun{ind}_{\N'}$ in terms of $\fun{ind}_{\fun W}$:
\[
    \fun{ind}_{\N'}(E,e_{0'},e_{\fun{succ}'})
        \defeq\fun{ind}_{\fun W}(E,e_s).
\]
Moreover, this definition must satisfy:
\[
    \fun{ind}_{\N'}(E,e_{0'},e_{\fun{succ}'})(\fun{succ}'(n))
    \equiv
        e_{\fun{succ}'}\bigl(
                n,
                \fun{ind}_{\N'}(
                    E,e_{0'},
                    e_{\fun{succ}'}
                    )(n)
            \bigr).
\]
If we substitute our definition of $\fun{ind}_{\N'}$ into this expected identity, we obtain:
\begin{equation}\label{eq:ind_W-requirement}
    \fun{ind}_{\fun W}(E,e_s)(\fun{succ}'(n))
    \equiv
        e_{\fun{succ}'}\bigl(
                n,
                \fun{ind}_{\fun W}(E,e_s)(n)
            \bigr).
\end{equation}
Given $\phi\colon\fun1\to\N'$ and $h\colon\prod_{b\colon\fun1}E(\phi(b))$, to define $e_s(1_{\fun2},\phi,h)$, we must construct a term of type $E(\fun{sup}(1_{\fun2},\phi))$.

Since the domain of both $\phi$ and $h$ is $\fun1$, we can apply them to $\fstar\colon\fun1$ obtaining
\begin{align*}
    \phi(\fstar)&\colon\N',\\
    h(\fstar)&\colon E(\phi(\fstar)).
\end{align*}
We can now pass these terms to $e_{\fun{succ}'}$, which yields
\[
    e_{\fun{succ}'}(\phi(\fstar),h(\fstar))
    \colon E(\fun{sup}(1_{\fun2},\lambda{\fstar}.\,\phi(\fstar))).
\]
This codomain is definitionally equal to $E(\fun{sup}(1_{\fun2},\phi))$. Since this matches the required type, we can define the step function as
\begin{equation}\label{eq:e_s-definition}
    e_s(1_{\fun2},\phi,h)\defeq e_{\fun{succ}'}
        (\phi(\fstar),h(\fstar)).
\end{equation}
Substituting this definition into the $\fun W$-type computation rule for $\fun{succ}'(n)$[cite: 1], we obtain
\begin{align*}
    \fun{ind}_{\fun W}(E,e_s)(\fun{succ}'(n))
        &\equiv
            e_s\bigl(
                    1_{\fun2},
                    \lambda{\fstar}.\,n,
                    \fun{ind}_{\fun W}(E,e_s)
                        \circ(\lambda{\fstar}.\,n)
                \bigr)\\
        &\equiv
            e_{\fun{succ}'}\bigl(
                    (\lambda{\fstar}.\,n)(\fstar),
                        (
                            \fun{ind}_{\fun W}(E,e_s)
                            \circ
                            (\lambda{\fstar}.\,n)
                        )(\fstar)
                \bigr)\\
        &\equiv
            e_{\fun{succ}'}\bigl(
                    n,
                    \fun{ind}_{\fun W}(E,e_s)(n)
                \bigr),
\end{align*}
which reconstructs the expected recurrence \eqref{eq:ind_W-requirement}.

However, the problem arises because the base case fails. This requirement translates into the definitional equality
\[
    \fun{ind}_{\N'}(E,e_{0'},e_{\fun{succ}'})(0')
        \equiv e_{0'}.
\]
In our setup, substituting the definition of $\fun{ind}_{\N'}$ translates this requirement into
\[
    \fun{ind}_{\fun W}(E,e_s)(0')
        \equiv e_{0'}.
\]
To see why this fails, let us evaluate the left-hand side. We have
\begin{align*}
    \fun{ind}_{\fun W}(E,e_s)(0')
        &\equiv \fun{ind}_{\fun W}(E,e_s)
            (\fun{sup}(0_{\fun2},\fun{rec}_{\fun0}(\N')))\\
        &\equiv e_s\bigl(
                0_{\fun2},
                \fun{rec}_{\fun0}(\N'),
                \fun{ind}_{\fun W}(E,e_s)
                    \circ\fun{rec}_{\fun0}(\N')
            \bigr)
            &&\text{; \eqref{eq:ind_W-computation-rule}}.
\end{align*}
The required signature for the base case is
\[
    e_s(0_{\fun2},\dummy,\dummy)\colon
    \prod_{\phi\colon\fun0\to\N'}
    \Bigl(\prod_{b\colon\fun0}E(\phi(b))\Bigr)
    \to E(\fun{sup}(0_{\fun2},\phi)).
\]
We are given the standard base case data
\[
    e_{0'}\colon E(0')
        \equiv E(\fun{sup}(0_{\fun2},\fun{rec}_{\fun0}(\N'))).
\]
To define $e_s(0_{\fun2},\phi,h)\defeq e_{0'}$, we would need the strict syntactic match between codomains:
\[
    E(\fun{sup}(0_{\fun2},\phi))
    \equiv
    E(\fun{sup}(0_{\fun2},\fun{rec}_{\fun0}(\N'))).
\]
But this would require
\[
    \phi\equiv\fun{rec}_{\fun0}(\N'),
\]
which does not hold because not all functions out of $\fun0$ are judgmentally equal.

As a result, the direct assignment fails to typecheck natively.

\subsection{Bridging the Gap}

Of course, given $\phi\colon\fun0\to\N'$ there is a trivial homotopy $\phi\sim\fun{rec}_{\fun0}(\N')$. By function extensionality, this gives a path $p_\phi\colon\phi\eq{\fun0\to\N'}\fun{rec}_{\fun0}(\N')$.

Thus, to bridge the gap described above, we must transport the given term $e_{0'}$ along this path:
\[
    e_s(0_{\fun2},\phi,h)
        \defeq\fun{transport}^{E(\fun{sup}(0_{\fun2},\dummy))}
            (p_\phi^{-1},e_{0'}).
\]
When we compute the $\fun W$-type induction for $0'$, we obtain
\[
    \fun{ind}_{\fun W}(E,e_s)(0')
        \equiv\fun{transport}^{E(\fun{sup}(0_{\fun2},\dummy))}(p_\phi^{-1},e_{0'}).
\]
Because $p_\phi$ is constructed using the function extensionality axiom, which is not a constructor that can be evaluated, the transport cannot reduce any further. Therefore, the judgmental equality $\fun{ind}_{\fun W}(E,e_s)(0')\equiv e_{0'}$ fails, holding only propositionally.

\medskip

\textbf{Note.} Although relaxing these computation rules to hold only propositionally might seem like a viable workaround, the resulting loss of judgmental equality is precisely why modern proof assistants abandon the\/ $\fun W$-type encoding altogether. By elevating \textit{all} inductive types to primitive status, these systems ensure that base constructors like\/ $\fun{nil}$ and\/ $\fun{zero}$ remain judgmentally unique and perfectly behaved.

As Chapter~6 demonstrates, this foundational shift is also a structural necessity for accommodating higher-dimensional path constructors.

Nevertheless, it is important to emphasize that while the\/ $\fun W$-type encoding is a pragmatic dead-end for language designers, it provides a crucial foundational framework for homotopy type theorists.

\section{Strict Positivity}

The aim of this section is to establish the requirements that constructors of inductive types must comply with in order to ensure valid definitions of new types.

The main idea is that a proper constructor may admit arguments that are functions, but the type being defined cannot be the domain of those functional arguments.

A good way to illustrate the necessity of this restriction is by showing how ignoring it leads to the Russell paradox.

Suppose we define an inductive type $X$ by means of a single constructor that violates our restriction:
\[
    \fun c\colon(X\to\fun2)\to X.
\]
Because $X$ is an inductive type, we should be able to define, for any $C\colon\univ$, a function $u\colon X\to C$ by specifying how $u$ operates on points of $X$ constructed from functions $f\colon X\to\fun2$ via~$\fun c$.

For example, in the case $C\defeq X\to\fun2$, we could define
\[
    u\colon X\to(X\to\fun2)
\]
by establishing
\[\tag{$1$}
    u(\fun c(f))\defeq f.
\]

Following Russell's paradox, this would allow us to build a diagonal function $d\colon X\to\fun2$ as
\[\tag{$2$}
    d(x)\defeq\fun{not}(u(x)(x)).
\]
In consequence, we could define
\[\tag{$3$}
    x_0\defeq\fun c(d).
\]
The contradiction explodes when we try to evaluate the function $d$ on the element $x_0$. Indeed,
\[
    d(x_0)
        \stackrel{(3)}\equiv d(\fun c(d))
        \stackrel{(2)}\equiv\fun{not}(u(\fun c(d))(x_0))
        \stackrel{(1)}\equiv\fun{not}(d(x_0)),
\]
which is impossible in the type $\fun2$ [cf.~Exercise~\ref{exr:2-simeq-2-simeq-2}].

\subsection{Free Variables}

\begin{defn}
    Let $\Phi$ be a type expression, and let $\mathcal F(\Phi)$ denote the set of \textsl{free variables} in $\Phi$, defined recursively on the syntactic structure of the expression as follows:
    \begin{itemize}
        \item $\mathcal F(V)\defeq\set V$ for any variable $V$.

        \item $\mathcal F(C)\defeq\emptyset$ for any constant type $C$.
        
        \item $\mathcal F(A+B)\defeq\mathcal F(A)\cup\mathcal F(B)$.

        \item $\mathcal F\bigl(\prod_{y\colon A}B(y)\bigr)\defeq\mathcal F(A)\cup(\mathcal F(B(y))\setminus\set y)$.
        
        \item $\mathcal F\bigl(\sum_{y\colon A}B(y)\bigr)\defeq\mathcal F(A)\cup(\mathcal F(B(y))\setminus\set y)$.
    \end{itemize}
\end{defn}

\begin{rems}\ 
    \begin{enumerate}[-]
        \item The variable $y$ in the last two parts of the definition is excluded from the set of free variables because it is free in $B(y)$ but gets \textsl{bound} in the entire expression.

        \item The last two parts of the definition imply that $\mathcal F(A\to B)$ and $\mathcal F(A\times B)$ equal $\mathcal F(A)\cup\mathcal F(B)$.
    
        \item Using this precise syntactic extraction, the informal statement ``$X$ does not occur in $\Phi$'' translates directly to the set-theoretic condition:
        \[
            X\notin\mathcal F(\Phi).
        \]
        Conversely, ``$X$ occurs in $\Phi$'' simply means $X\in\mathcal F(\Phi)$.
    \end{enumerate}
    
\end{rems}

\begin{defn}
    Let $X$ be a type variable and $\Phi$ a type expression. The predicate $\fun{Pos}_X(\Phi)$, denoting that $X$ occurs \textsl{strictly positively} in $\Phi$, holds when one of the following conditions is met:
    \begin{enumerate}[a),font=\upshape]
        \item $X\notin\mathcal F(\Phi)$,
        
        \item $\Phi\equiv X$,
        
        \item $\Phi\equiv \prod_{y\colon A}B(y)$, $X\notin\mathcal F(A)$, and $\fun{Pos}_X(B(y))$ holds for all $y\colon A$,
        
        \item $\Phi\equiv \sum_{y\colon A}B(y)$, $X \notin\mathcal F(A)$, and $\fun{Pos}_X(B(y))$ holds for all $y\colon A$,
        
        \item $\Phi\equiv A+B$, and $\fun{Pos}_X(A)$ and $\fun{Pos}_X(B)$ both hold.
    \end{enumerate}
\end{defn}

\begin{rem}
    Clauses c) and d) require $X\notin\mathcal F(A)$ to prevent~$X$ from appearing anywhere inside the domain $A$, while recursively checking $\fun{Pos}_X(B(y))$ to allow $X$ to occur strictly positively in the codomain.
\end{rem}

\begin{thm}
    Let $X$ be a type variable and let $\Phi(X)$ be a type expression such that $\fun{Pos}_X(\Phi(X))$ holds. Given any two types $U$ and $V$, and a function $f\colon U\to V$, there exists a canonically defined function
    \[
        \fun{map}_\Phi(f)\colon \Phi(U)\to \Phi(V).
    \]
    Furthermore, this mapping preserves identities and composition, making $\Phi$ a covariant functor.
\end{thm}

\begin{proof}
    We construct the function $\fun{map}_\Phi(f)$ by structural induction on the derivation of $\fun{Pos}_X(\Phi(X))$.

    \begin{enumerate}[a),font=\upshape]
        \item \textsc{Non-occurrence:} If $X \notin \mathcal F(\Phi)$, then $\Phi(U) \equiv \Phi(V) \equiv \Phi$. We define $\fun{map}_\Phi(f)\defeq\fun{id}_\Phi$, the identity function on $\Phi$.
        
        \item \textsc{Base variable:} If $\Phi(X) \equiv X$, then $\Phi(U) \equiv U$ and $\Phi(V) \equiv V$. We define $\fun{map}_\Phi(f) \defeq f$.

        \item \textsc{Coproduct:} Suppose $\Phi(X) \equiv A(X) + B(X)$, where both $\fun{Pos}_X(A(X))$ and $\fun{Pos}_X(B(X))$ hold. By the inductive hypothesis, we have 
        \[
            \fun{map}_A(f)\colon A(U)\to A(V)
            \quad\text{and}\quad
            \fun{map}_B(f)\colon B(U)\to B(V).
        \]
        We define $\fun{map}_\Phi(f)$ by pattern matching on the coproduct:
        \begin{align*}
            \fun{map}_\Phi(f)(\fun{inl}(a)) &\defeq \fun{inl}(\fun{map}_A(f)(a)),\\
            \fun{map}_\Phi(f)(\fun{inr}(b)) &\defeq \fun{inr}(\fun{map}_B(f)(b)).
        \end{align*}
        
        \item \textsc{Dependent product:} Suppose $\Phi(X) \equiv \prod_{y\colon A} B(X,y)$, where $X \notin \mathcal F(A)$ and $\fun{Pos}_X(B(X,y))$ holds for all $y\colon A$. By the inductive hypothesis, for any $y\colon A$, we have a function $\fun{map}_{B(y)}(f)\colon B(U,y)\to B(V,y)$. Given a dependent function $g\colon \prod_{y\colon A} B(U,y)$, we define
        \[
            \fun{map}_\Phi(f)(g)\defeq\lambda y.\,\fun{map}_{B(y)}(f)(g(y)).
        \]
        
        \item \textsc{Dependent sum:} Suppose $\Phi(X) \equiv \sum_{y\colon A} B(X,y)$, where $X \notin \mathcal F(A)$ and $\fun{Pos}_X(B(X,y))$ holds for all $y\colon A$. By the inductive hypothesis, we again have $\fun{map}_{B(y)}(f)$ for any $y\colon A$. Given a pair $p\colon \sum_{y\colon A} B(U,y)$, we define
        \[
            \fun{map}_\Phi(f)(p)\defeq\bigl(\pi_1(p), \fun{map}_{B(\pi_1(p))}(f)(\pi_2(p))\bigr).
        \]
    \end{enumerate}
    This completes the construction. To complete the proof of functoriality, it remains to show that $\fun{map}_\Phi(\fun{id}_U)$ is propositionally equal to $\fun{id}_{\Phi(U)}$ and $\fun{map}_\Phi(g \circ f)$ is propositionally equal to $\fun{map}_\Phi(g) \circ \fun{map}_\Phi(f)$. We omit the details because these properties follow by structural induction under the same case analysis as above.
\end{proof}

\begin{xmpl}
    The converse statement ---that functoriality implies strict positivity--- is false. Consider the double continuation type former, using the boolean type~$\fun2$:
    \[
        \Phi(X)\defeq(X\to\fun2)\to\fun2.
    \]
    This type expression is \textit{not} strictly positive. According to our definition, for $\fun{Pos}_X((X \to \fun2) \to \fun2)$ to hold, we would need $X$ not to occur in the domain $X \to \fun2$. Because $X \in \mathcal F(X \to \fun2)$, the strict positivity check fails. In this expression, $X$ is nested inside two negative positions (left sides of arrows), which makes it an ``ordinarily positive'' position, but not a strictly positive one.

    However, $\Phi$ natively defines a covariant functor. Given any function $f\colon U \to V$, we can define the mapping $\fun{map}_\Phi(f)\colon \Phi(U) \to \Phi(V)$ as follows:
    \[
        \fun{map}_\Phi(f)(g)\defeq\lambda h.\,g(h \circ f),
    \]
    where $g\colon (U \to \fun2) \to \fun2$ and $h\colon V \to \fun2$. This mapping also preserves identities and compositions.
\end{xmpl}

\begin{rem}
    The example highlights why type theory specifically demands \textit{strict} positivity rather than just ordinary positivity. If we have $\Phi(X)\defeq X\to A$, which is contravariant in the sense that it maps $f\colon U\to V$ to $f^*\colon\Phi(V)\to\Phi(U)$, then composing it with a second contravariant expression $\Phi'(X)\defeq X\to A'$, results by double inversion of the arrows in the covariant behavior $f\mapsto f^{**}$.
\end{rem}

\section{Inductive Type Families}

In this section we will see how a collection of types indexed by some other type can be simultaneously defined by induction. Instead of defining a single type $X$, we define a family of types $X\colon I\to\univ$, where $I$ is the index type.

For a standard inductive type $X$, every constructor must output a value of type~$X$. For an inductive type family $X\colon I\to\univ$, the constructors can target a specific index $i\colon I$ to produce a value in~$X(i)$.

\begin{xmpl}\label{xmpl:Vec_A}
    Let $A$ be a type. The inductive type family of \textsl{vectors over~$A$}, denoted $\fun{Vec}_A \colon \N \to \univ$, is generated by the following two constructors:
    \begin{itemize}
        \item $\fun{nil} \colon \fun{Vec}_A(0)$
        \item $\fun{cons} \colon \prod_{n\colon\N} A \to \fun{Vec}_A(n) \to \fun{Vec}_A(\fun{succ}(n))$
    \end{itemize}
\end{xmpl}

For a standard type $X$, the motive is a family of types $P\colon X\to\univ$. For a type family $X\colon I\to\univ$, the elements are distributed across different indices. Therefore, our motive must depend on both the index and the element. It becomes $P\colon\prod_{i\colon I}X(i)\to\univ$.

\begin{xmpl}\label{xmpl:Vec_A-ind}
    Following with Example~\ref{xmpl:Vec_A}, let $P \colon \prod_{n\colon\N}\fun{Vec}_A(n)\to\univ$ be a family of types.
    
    The eliminator for the vector family $\fun{ind}_{\fun{Vec}_A}$ takes two arguments corresponding to the constructors:
    \begin{itemize}
        \item a base case: $p_{\fun{nil}}\colon P(0,\fun{nil})$,

        \item an inductive step:
        \[
            p_{\fun{cons}}\colon
                \prod_{n\colon\N}\;
                \prod_{a\colon A}
                \prod_{v\colon\fun{Vec}_A(n)}
                    P(n,v)
                    \to
                    P(\fun{succ}(n),\fun{cons}(n,a,v)).
        \]
    \end{itemize}
    Given these, the eliminator produces a dependent function for all vectors at all indices:
    \[
        \fun{ind}_{\fun{Vec}_A}(
                P,
                p_{\fun{nil}},
                p_{\fun{cons}}
            )
            \colon
                \prod_{n\colon\N}\;
                \prod_{v\colon\fun{Vec}_A(n)}
                    P(n,v),
    \]
    with computation rules:{\small
    \begin{align*}
        \fun{ind}_{\fun{Vec}_A}(
                P,
                p_{\fun{nil}},
                p_{\fun{cons}},
                0,
                \fun{nil}
            )
            &\equiv p_{\fun{nil}}\\
        \fun{ind}_{\fun{Vec}_A}\bigl(
                P,
                p_{\fun{nil}},
                p_{\fun{cons}},
                \fun{succ}(n),
                \fun{cons}(n,a,v)
            \bigr)
            &\equiv p_{\fun{cons}}
                \bigl(
                    n,
                    a,
                    v,
                    \fun{ind}_{\fun{Vec}_A}(
                            P,
                            p_{\fun{nil}},
                            p_{\fun{cons}},
                            n,
                            v
                        )
                \bigr).
    \end{align*}}
\end{xmpl}

\paragraph{Eliminator Methods.}
The eliminator $\fun{ind}_X$ is a machine that builds proofs by induction. To operate this machine, we have to supply it with the \textsl{eliminator methods}.

Consider the natural numbers $\N$ with constructors $\fun{zero}\colon\N$ and $\fun{succ}\colon\N\to\N$. Let our motive be a property $P\colon\N\to\univ$ that we want to prove for all natural numbers.

The eliminator $\fun{ind}_{\N}$ requires two methods, one for each constructor:
\begin{itemize}
    \item \textit{The method for $\fun{zero}$:} 
    Since the $\fun{zero}$ constructor takes no arguments, this method reduces to proving the base case:
    \[
        p_{\fun{zero}}\colon P(0).
    \]
    
    \item \textit{The method for $\fun{succ}$:} 
    The constructor $\fun{succ}$ takes a number $n$ and produces $n+1$. The method must show that \textsl{if} the property holds for $n$, it holds for $n+1$. This is the inductive step:
    \[
        p_{\fun{succ}}\colon\prod_{n\colon\N}
            P(n)\to P(\fun{succ}(n)).
    \]
\end{itemize}
The eliminator consumes these two methods and outputs the global proof for all natural numbers:
\[
    \fun{ind}_{\N}(P,p_{\fun{zero}},p_{\fun{succ}})
        \colon\prod_{n\colon\N}P(n).
\]

\begin{rem}
    From a programming perspective, the eliminator is a switch statement that branches based on how the data was constructed. The eliminator method is simply the block of code of one particular branch. If the branch involves recursion, the method receives as an argument the result of the recursive call.
\end{rem}

\begin{defn}\label{defn:no-recursive-args}
    Let $I$ be an index type and $X\colon I\to\univ$ an inductive type family. A constructor $\fun c$ with no recursive arguments (a \textsl{base constructor}) is specified by two components:
    \begin{itemize}
        \item \textsc{Context:} A type $C$ for all non-recursive arguments.
        
        \item \textsc{Target Index:} A function $g\colon C\to I$ that determines the index of the constructed element.
    \end{itemize}
    The signature of the constructor is:
    \begin{equation}\label{eq:no-recursive-args}
        \fun c\colon\prod_{z\colon C}X(g(z)).
    \end{equation}
    Given a motive $P\colon\prod_{i\colon I}X(i)\to\univ$, the eliminator method $p_{\fun c}$ corresponding to this base constructor must satisfy:
    \[
        p_{\fun c}\colon\prod_{z\colon C}
            P(g(z),\fun c(z)).
    \]
\end{defn}


\begin{xmpl}
    For the $\fun{nil}$ constructor of the vector family $\fun{Vec}_A\colon\N\to\univ$, there are no non-recursive arguments either. Therefore, 
    \begin{itemize}
        \item \textsc{Context:} The context type is the unit type $\fun1$.
        
        \item \textsc{Target Index:} The empty vector has length $0$, so $g(\fstar)\defeq0$.
    \end{itemize}
    This gives the signature $\fun c\colon\prod_{z\colon\fun1}\fun{Vec}_A(0)$, which is equivalent to giving the constant element:
    \[
        \fun{nil}\colon\fun{Vec}_A(0).
    \]
\end{xmpl}

\begin{defn}\label{defn:one-recursive-arg}
    Let $I$ be an index type and $X\colon I\to\univ$ be an inductive type family. A \textsl{constructor} $\fun c$ with exactly one recursive argument is specified by three components:
    \begin{itemize}
        \item \textsc{Context:} A type $C$ for all non-recursive arguments.
        
        \item \textsc{Source Index:} A function $h\colon C\to I$ that determines the index required for the recursive premise.
        
        \item \textsc{Target Index:} A function $g\colon C\to I$ that determines the index of the constructed element.
    \end{itemize}
    The signature of the constructor is:
    \begin{equation}\label{eq:one-recursive-arg}
        \fun c\colon\prod_{z\colon C}X(h(z))\to X(g(z)).
    \end{equation}
    Given a motive $P\colon\prod_{i\colon I}X(i)\to\univ$, the \textsl{eliminator method} $p_{\fun c}$ corresponding to this constructor must provide a proof for the new element:
    \[
        p_{\fun c}\colon
            \prod_{z\colon C\mathstrut}\;
            \prod_{x\colon X(h(z))\mathstrut}
                P(h(z),x)\to P(g(z),\fun c(z,x))
    \]
\end{defn}

\begin{xmpl}
    Applying this to the $\fun{cons}$ constructor of the vector family $\fun{Vec}_A\colon\N\to\univ$, we have:
    \begin{itemize}
        \item \textsc{Context:} The non-recursive data is a length $n\colon\N$ and a coordinate $a\colon A$. Hence, we define $C\defeq\N\times A$.
        
        \item \textsc{Source Index:} Since the recursive vector must have length $n$, we have $h(n,a)\defeq n$.
        
        \item \textsc{Target Index:} The resulting vector has length $n+1$. Thus, the target index function is $g(n,a)\defeq\fun{succ}(n)$.
    \end{itemize}
    Plugging these into the signature of the generic constructor yields:
    \[
        \fun c\colon\prod_{(n,a)\colon\N\times A}
            \fun{Vec}_A(n)\to\fun{Vec}_A(\fun{succ}(n))
    \]
    By currying the dependent pair into separate arguments, we recover the standard signature:
    \[
        \fun{cons}\colon
            \prod_{n\colon\N}\;
            \prod_{a\colon A}
                \fun{Vec}_A(n)
                \to
                \fun{Vec}_A(\fun{succ}(n)).
    \]
\end{xmpl}

\begin{xmpl}
    Consider the inductive predicate defining even numbers, represented as a family $\fun{IsEven}\colon\N\to\univ$. It has a step constructor that says ``if $n$ is even, then $n+2$ is even''.
    \begin{itemize}
        \item \textsc{Context:} We only need the number $n$, so $C\defeq\N$.
        
        \item \textsc{Source Index:} The premise is about $n$, so $h(n)\defeq n$.
        
        \item \textsc{Target Index:} The conclusion is about $n+2$, so $g(n)\defeq\fun{succ}(\fun{succ}(n))$.
    \end{itemize}
    This directly produces the signature:
    \[
        \fun{stepEven}\colon\prod_{n\colon\N}
            \fun{IsEven}(n)
            \to
            \fun{IsEven}(\fun{succ}(\fun{succ}(n))).
    \]
\end{xmpl}


\begin{defn}
    Let $I$ be an index type and $X\colon I\to\univ$ an inductive type family. A general constructor $\fun c$ is specified by four components:
    \begin{itemize}
        \item \textsc{Context:} The type $C$ of all non-recursive arguments.
        \item \textsc{Branching Type:} A family of types $W\colon C\to\univ$ where, for any context element $z\colon C$, the type $W(z)$ indexes the recursive arguments.
        
        \item \textsc{Source Index:} A function $h\colon\prod_{z\colon C}W(z)\to I$ that determines the index required for the recursive premise at branch $w$.
        
        \item \textsc{Target Index:} A function $g \colon C \to I$ that determines the index of the final constructed element.
    \end{itemize}
    The signature of the general constructor is:
    \begin{equation}\label{eq:general-inductive-family-constructor}
        \fun c\colon
            \prod_{z\colon C}
            \Bigl(
                \prod_{w\colon W(z)}X(h(z,w))
            \Bigr)
            \to X(g(z)).
    \end{equation}
    Let $P\colon\prod_{i\colon I}X(i)\to\univ$ be a motive. The eliminator method $p_{\fun c}$ corresponding to this general constructor must provide a proof for the newly constructed element $\fun c(u,z)$. 
    
    To do this, the method receives three pieces of data:
    \begin{itemize}
        \item The non-recursive context data $z\colon C$.

        \item The bundle of recursive elements $x\colon\prod_{w\colon W(z)} X(h(z,w))$.
        
        \item The bundle of inductive hypotheses, which proves the property $P$ for every recursive element across all branches. This is a function that, for any branch $w\colon W(z)$, returns a proof of $P(h(z,w),x(w))$.
    \end{itemize}
    The signature of the general eliminator method is therefore:
    \[
        p_{\fun c}\colon
            \prod_{z\colon C}\;
            \prod_{x\colon\prod_{w\colon W(z)}X(h(z,w))}
                \Bigl(
                    \prod_{w\colon W(z)}P(h(z,w),x(w))
                \Bigr)
                \to P(g(z),\fun c(z,x)).
    \]
\end{defn}

\begin{rem}
    When we evaluate the global eliminator $\fun{ind}_X$ on the constructed term $\fun c(z,x)$, the computation rule establishes that
    \[
        \fun{ind}_X(P,\dots,\fun c(z,x))
            \equiv p_c
                \bigl(
                    z,
                    x,
                    \lambda w.\,\fun{ind}_X
                        \bigl(
                            P,
                            \dots,
                            x(w)
                        \bigr)
                \bigr).
    \]
\end{rem}

\subsection*{The Accessibility Predicate (Well-Foundedness)}

In classical mathematics, a relation $R$ on a set $A$ is \textsl{well-founded} if every non-empty subset has an $R$-minimal element, or equivalently, if there are no infinite descending chains:
\[
    x_0\leftarrow x_1\leftarrow x_2\leftarrow\cdots,
\]
where $x\leftarrow y$ stands for $R(x,y)$.

Constructive type theory defines well-foundedness differently, using an inductive approach. An element is \textsl{accessible} if it is built up solely from other accessible elements.

\begin{defn}
    Given a type $A$ and a relation $R\colon A\times A\to\univ$, the \textsl{accessibility predicate} is an inductive type family $\fun{Acc}\colon A\to\univ$ generated by a single constructor:
    \[
       \fun{accIntro}\colon\prod_{x\colon A}
        \Bigl(
            \prod_{y\colon A}R(y,x)\to\fun{Acc}(y)
        \Bigr)
        \to\fun{Acc}(x)
    \]
\end{defn}

This constructor embodies a single rule: an element $x$ is \textsl{accessible} if every strict predecessor of $x$ is already accessible.

\begin{rem}
    The base cases are naturally captured by \textsl{minimal} elements.
    
    We say that an element $x_0$ is \textsl{minimal} if, for every $y\colon A$, there is a witness $e_y\colon R(y,x_0)\to\fun0$. Consequently, the composition $\fun{rec}_{\fun0}(\fun{Acc}(y))\circ e_y$ is a function of type $R(y,x_0)\to\fun{Acc}(y)$. Applying $\fun{accIntro}(x_0)$ to this family of functions yields a proof of $\fun{Acc}(x_0)$ without requiring any actual predecessors.
\end{rem}


\subsection*{Well-Founded Recursion}

The idea behind the accessibility predicate is to provide a structural foundation for well-founded recursion. A termination checker usually performs \textsl{structural} recursion: it calls a function on a syntactically smaller piece of an inductive datatype.

To write a recursive function where the argument decreases according to some arbitrary relation $R$, it suffices to prove that the relation is well-founded by showing that every element in $A$ is accessible:
\[
   \fun{isWellFounded}\colon\prod_{x\colon A}\fun{Acc}(x)
\]
Constructing the function $\fun{isWellFounded}$ proves that recursive functions that decrease according to $R$ will always hit a base case and terminate.

\bigskip

This is perhaps the most important example of an indexed family requiring the full schema. To prove that a relation $R \colon A \times A \to \univ$ is well-founded (allowing us to write terminating recursive functions over it), we define the inductive family $\fun{Acc} \colon A \to \univ$.

The constructor says: an element $x$ is ``accessible'' if every $y$ strictly smaller than $x$ is accessible.
\begin{itemize}
    \item \textsc{Context:} We define $C \defeq A$ and let $z\equiv x$, the element we want to prove accessible.
    
    \item \textsc{Branching Type:} We need one recursive premise for every $y$ that is smaller than $x$. The ``shape'' of the recursion depends entirely on the context $x$. We define the branching type as the type of all elements related to $x$:
    \[
        W(x)\defeq\sum_{y\colon A} R(y, x)
    \]
    
    \item \textsc{Source Index:} For a given branch $w \equiv (y, r) \colon W(x)$, the recursive premise must be about $y$. So,  $h(x,(y,r))\defeq y$.
    
    \item \textsc{Target Index:} Since the constructor proves that $x$ is accessible, we define $g(x)\defeq x$.
\end{itemize}
Plugging these into the general schema:
\[
    \fun{accIntro}\colon
        \prod_{x\colon A}
            \Bigl(
                \prod_{
                    \dep{y,r}\colon
                        \sum_{y\colon A} R(y,x)
                }
                \fun{Acc}(y)
            \Bigr)
            \to\fun{Acc}(x)
\]
Currying the inner product gives the standard signature:
\[
    \fun{accIntro}\colon\prod_{x\colon A}
        \Bigl(
            \prod_{y\colon A}R(y,x)\to\fun{Acc}(y) 
        \Bigr)
        \to\fun{Acc}(x)
\]
Here, the number of recursive arguments dynamically changes based on how many elements relate to the specific $x$ we provide.

\textbf{Note.} Suppose we have an algorithm that calculates integer division, producing a remainder $y$ from a number $x$. The compiler knows nothing about integer arithmetic, so to prove termination we cannot rely on the mathematical fact that $y$ is smaller than $x$. Instead, the strategy is to structurally traverse a proof tree $a$ of $\fun{Acc}(x)$.

When the algorithm computes $y$, it simultaneously computes a witness $r$ of $R(y,x)$. Since the proof $a$ is constructed by means of $\fun{accIntro}$, its argument is a function of type $\prod_{z\colon A}R(z,x)\to\fun{Acc}(z)$. By applying this function to $y$ and~$r$, we extract a proof $a_y$ of $\fun{Acc}(y)$. Since $a_y$ was extracted directly from the structure of $a$, the compiler recognizes it as syntactically smaller, ensuring that successive recursive calls will terminate.

\begin{rem}
    To derive the definition of a base constructor from the general constructor schema, we formalize the absence of recursive arguments by setting the branching type to the empty type.

    \textsc{Step 1.}
    For a constructor with no recursive arguments, there are no recursive premises to evaluate for any context $z\colon C$. Therefore, we define the branching type as the empty type:
    \[
        W(z)\defeq\fun0.
    \]
    Consequently, the source index function $h\colon\prod_{z\colon C}\fun0\to I$ is vacuously defined by the induction principle of the empty type.
    
    \textsc{Step 2.}
    We substitute $W(z)\defeq\fun0$ into the general signature given in Equation~\eqref{eq:general-inductive-family-constructor}:
    \[
        \fun c\colon\prod_{z\colon C}
            \Bigl(
                \prod_{w\colon\fun0}X(h(z,w))
            \Bigr)
            \to X(g(z)).
    \]
    The dependent product over the empty type is inhabited by the induction function $\fun{ind}_{\fun0}$, making any $\Pi$-type over $\fun0$ equivalent to $\fun1$. Thus, the signature simplifies to:
    \[
        \fun c\colon\prod_{z\colon C}\fun1\to X(g(z)).
    \]
    Since $(\fun1\to A)\simeq A$, we arrive at the exact signature for a base constructor:
    \[
        \fun c\colon\prod_{z\colon C} X(g(z)).
    \]
    
    \textsc{Step 3.}
    Invoking the same reasons as above, the general eliminator method $p_{\fun c}$ reduces to:
    \[
        p_{\fun c}\colon\prod_{z\colon C}
            P(g(z),\fun c(z)).
    \]
\end{rem}

\begin{xmpl}\label{xmpl:Fin}
    The type $\fun{Fin}(n)$ represents the canonical type with~$n$ elements.\footnote{See also Exercise~\ref{exr:Fin}.} Rather than being a primitive notion, it is formally defined as an indexed inductive type family over the natural numbers, $\fun{Fin}\colon\N\to\univ$, generated by two constructors:
    \begin{itemize}
    \item \textsl{Finite zero:} $\fun{fz}\colon\prod_{k\colon\N}\fun{Fin}(\fun{succ}(k))$
    
    \item \textsl{Finite successor:} $\fun{fs}\colon\prod_{k\colon\N}\fun{Fin}(k)\to\fun{Fin}(\fun{succ}(k))$
    \end{itemize}
    The constructor $\fun{fz}$ (finite zero) provides the ``first'' element of any non-empty finite type, and $\fun{fs}$ (finite successor) injects the $k$ elements of $\fun{Fin}(k)$ into the ``next'' $k$ slots of $\fun{Fin}(\fun{succ}(k))$.

    \begin{enumerate}
        \item \textit{The Base Constructor.} The constructor $\fun{fz}$ takes a natural number as a parameter, and has no recursive premises. Therefore, we can apply Definition~\ref{defn:no-recursive-args} directly:
        \begin{itemize}
        \item \textsc{Context:} The non-recursive argument is a natural number $k$. Thus, $C\defeq\N$.
        
        \item \textsc{Target Index:} The constructed element has type $\fun{Fin}(\fun{succ}(k))$, so $g(k)\defeq\fun{succ}(k)$.
        \end{itemize}
        Applying the base constructor signature from Equation~\eqref{eq:no-recursive-args} yields:
        \[
            \fun{fz}\colon\prod_{k\colon\N}\fun{Fin}(g(k)),
        \]
        which evaluates to:
        \[
            \fun{fz}\colon\prod_{k\colon\N}
                \fun{Fin}(\fun{succ}(k)).
        \]
    
        \item \textit{The Recursive Constructor.} The constructor $\fun{fs}$ takes a natural number parameter, followed by one recursive argument from the $\fun{Fin}$ family.
        \begin{itemize}
            \item \textsc{Context:} The non-recursive argument is a natural number $k$. Thus, $C\defeq\N$.
            
            \item \textsc{Branching Type:} There is one recursive argument, so $W(k)\defeq\fun1$.
            
            \item \textsc{Source Index:} The single recursive premise must belong to $\fun{Fin}(k)$. Thus, evaluated at the single branch $\fstar$, we have $h(k,\fstar)\defeq k$.
            
            \item \textsc{Target Index:} The constructed element belongs to $\fun{Fin}(\fun{succ}(k))$, so $g(k)\defeq\fun{succ}(k)$.
        \end{itemize}
        Applying the general schema yields:
        \[
            \fun{fs}\colon \prod_{k\colon\N} \Bigl( \prod_{w\colon\fun1} \fun{Fin}(k) \Bigr) \to \fun{Fin}(\fun{succ}(k)),
        \]
        which reduces to:
        \[
            \fun{fs}\colon \prod_{k\colon\N} \fun{Fin}(k) \to \fun{Fin}(\fun{succ}(k)).
        \]
    \end{enumerate}
\end{xmpl}


\section{Mutual Inductive Types}

Suppose we want to define two types, $\fun{Even}$ and $\fun{Odd}$, representing the even and odd natural numbers. We would like to think of them as simultaneously defined by three constructors:
\begin{itemize}
    \item $\fun{zeroEven}\colon\fun{Even}$
    \item $\fun{succEven}\colon\fun{Odd}\to\fun{Even}$
    \item $\fun{succOdd}\colon\fun{Even}\to\fun{Odd}$
\end{itemize}
However, that would require introducing a new primitive notion of mutual induction. A simpler way to achieve the same result is to define an index type~$I$ with two tags and use it to obtain an inductive type family.

The index type $I$ can be introduced as the inductive type generated by:
\begin{itemize}
    \item $I\defeq\fun2$,
    \item $\fun{even}\defeq 0_{\fun2}$,
    \item $\fun{odd}\defeq 1_{\fun2}$.
\end{itemize}
We then define a single inductive type family $X\colon I\to\univ$, where $X(\fun{even})$ represents $\fun{Even}$, and $X(\fun{odd})$ represents $\fun{Odd}$.
\begin{enumerate}
    \item \textit{The Base Constructor.} This constructor defines the base case for even numbers. It has no recursive arguments, so we apply Definition~\ref{defn:no-recursive-args}:
    \begin{itemize}
        \item \textsc{Context:} $C\defeq\fun1$.
        
        \item \textsc{Target Index:} The function $g\colon\fun1\to I$ is given by $g(\fstar)\defeq\fun{even}$.
    \end{itemize}
    Applying the base constructor signature, we obtain:
    \[
        \fun{c_1}\colon\prod_{z\colon\fun1}X(\fun{even}),
    \]
    which is equivalent to $X(\fun{even})$.

    \item \textit{The Recursive Constructor for Even.}
    This constructor builds an even number from a predecessor, which must be odd. We apply the general constructor schema:
    \begin{itemize}
        \item \textsc{Context:} $C\defeq\fun1$.
        
        \item \textsc{Branching Type:} There is one recursive argument, so $W(\fstar)\defeq \fun1$.
        
        \item \textsc{Source Index:} The single recursive premise comes from the odd index. Thus, $h(\fstar,\fstar)\defeq\fun{odd}$.
        
        \item \textsc{Target Index:} The resulting element is even, so $g(\fstar)\defeq\fun{even}$.
    \end{itemize}
    Plugging these into the general schema yields:
    \[
        \fun{c_2}\colon\prod_{z\colon\fun1}
            \Bigl(
                \prod_{w\colon\fun1}X(\fun{odd})
            \Bigr)
            \to X(\fun{even}),
    \]
    which reduces to:
    \[
        \fun c_2\colon X(\fun{odd})\to X(\fun{even}).
    \]

    \item \textit{The Recursive Constructor for Odd.} Analogously, this constructor is derived as:
    \[
        \fun{c_3}\colon X(\fun{even})\to X(\fun{odd}).
    \]
\end{enumerate}

\subsection*{The General Approach}
Encoding mutual induction is fully captured by the general definition of an inductive type family.

To define a collection of mutually inductive types we  generalize the Even/Odd approach in the following way:

\begin{enumerate}
    \item \textit{The Index Type:} Instead of a two-element type, we define an index type~$I$ that contains a unique tag for every type we wish to define in our mutual block. For instance, if we are defining $n$ mutual types, we define an index type with $n$ base constructors, which we can label $I\defeq\fun{Fin}(n)$. 
    
    \item \textit{The Unified Family:} We define a single inductive type family $X\colon I\to\univ$. For each $i\colon I$, the application $X(i)$ represents the $i$th type in our mutual collection.
    
    \item \textit{The Constructors:} In the formal family $X$, a constructor targeting the $i$th type is simply a constructor whose target index function is the constant function $g(z)\defeq i$. 
    
    \item \textit{The Cross-References:} The mutual dependencies are entirely handled by the source index function $h\colon\prod_{z\colon C}W(z)\to I$. Whenever a constructor requires a recursive premise from the $j$th type in the collection, we simply set $h(z,w)\defeq j$ for that specific branch $w$.
\end{enumerate}

Because the general constructor schema
\[
    \fun c\colon\prod_{z\colon C}
        \Bigl(
            \prod_{w\colon W(z)}X(h(z,w))
        \Bigr)
        \to X(g(z))
\]
allows the source index $h$ to output any element of $I$ independently for each branch $w$, a single constructor can require premises from arbitrarily many different types in the mutual collection. Thus, mutual inductive types are a direct consequence of indexed inductive families.

\section{Inductive Type Families \&~{\normalfont\textsf W}-Types}

Inductive type families generalize $\fun W$-types in such a way that $\fun W$-types can be viewed as their unindexed variants.

To show that a $\fun W$-type $\fun W_AB$ is exactly an inductive type family indexed by the unit type, we set $I\defeq\fun1$ and define our family $X\colon\fun1\to\univ$ that consists of a single type $X(\fstar)$.

\begin{enumerate}[a)]
    \item \textit{Components.} A $\fun W$-type is specified by a type of shapes $A\colon\univ$ and an arity type family $B\colon A\to\univ$. We map these directly to the four components of a general constructor:
    \begin{itemize}
        \item \textsc{Context:} The non-recursive arguments determine the ``shape'' of the node. Thus, $C\defeq A$. Let $x\colon A$.
    
        \item \textsc{Branching Type:} The recursive premises for $x$ correspond to the ``positions'' where subtrees can be attached. Thus, $W(x)\defeq B(x)$.
    
        \item \textsc{Source Index:} Since there is only the $\fstar$ index, $h(x,w)\defeq\fstar$.
    
        \item \textsc{Target Index:} $g(x)\defeq\fstar$.
    \end{itemize}

    \item \textit{Constructor.} The signature of the general inductive family constructor is:
    \[
        \fun c\colon\prod_{z\colon C}
            \Bigl(
                \prod_{w\colon W(z)}X(h(z,w))
            \Bigr)
            \to X(g(z)),
    \]
    which becomes
    \begin{align*}
        \fun c&\colon\prod_{x\colon A}
            \Bigl(
                \prod_{w\colon B(x)}X(\fstar)
            \Bigr)
            \to X(\fstar)\\
            &\equiv\prod_{x\colon A}(B(x)\to X(\fstar))
            \to X(\fstar),
    \end{align*}
    where $W\defeq X(\fstar)$. If we set $\fun W\defeq X(\fstar)$, the constructor signature becomes:
    \[
        \prod_{x\colon A}(B(x)\to\fun W)\to\fun W,
    \]
    which reproduces the canonical introduction rule \eqref{eq:sup-constructor} for $\fun W_AB$.
\end{enumerate}

\section{Inductive-Inductive Definitions}

In standard mutual induction, we define two (or more) types simultaneously. However, while their constructors depend on one another, their existence does not, and so their formation rules are independent.

In an \textsl{inductive-inductive} definition, we define a type $A$ and a type family $B$ that depends on $A$. The formation rules of these entities look like this:
\begin{align*}
    A&\colon\univ\\
    B&\colon A\to\univ
\end{align*}
The issue is that we cannot even write down the type signature for $B$ without having defined $A$ first as a well-formed type. In addition, because they are defined together, the constructors that build elements of $A$ might need to use elements of $B$.

\begin{xmpl} {\upshape[Contexts \& Types]}
    As we saw in \nref{lpar:assumptions}, whenever we write a mathematical statement, we are always working under some set of assumptions or available variables.
    
    The inductive-inductive definition of $\fun{Con}$ and $\fun{Ty}$ is a way to formalize this process inside type theory itself.
    
    A \textsl{context}, modeled by the type $\fun{Con}$, represents the space of assumptions. It can be regarded as a list of variable declarations. An example would be:
    \[
        \Gamma\defeq
            [
                x\colon\fun{Nat},\;
                y\colon \fun{Vector}(x)
            ]
    \]
    We build contexts one step at a time. We start with the empty context, represented by the $\fun{nil}$ constructor, which has no variables. Then, we extend it by adding one variable at a time, represented by the $\fun{ext}$ constructor.

    \needspace{3\baselineskip}
    The family $\fun{Ty}(\Gamma)$ models all the valid type expressions that make sense under the specific assumptions of the context $\Gamma$. The double dependency arises because:
    \begin{itemize}
        \item To add a new variable to a context, we must specify its type, which must already be known to be valid in the current context. Thus, building an element of $\fun{Con}$ requires an element of $\fun{Ty}(\Gamma)$.
            
        \item To know if a dependent type is valid, we need to check those variables against our current environment. Thus, building an element of $\fun{Ty}(\Gamma)$ requires a context $\Gamma\colon\fun{Con}$.
    \end{itemize}
    
    To model this, we need an inductive-inductive definition. The idea works as follows:
    \begin{enumerate}[a),font=\upshape]
        \item First, we define the base case for contexts. The empty context requires no prerequisites and does not depend on any types:
        \[
            \fun{nil}\colon\fun{Con}.
        \]
        
        \item Now, the intertwining begins. To extend a context with a new variable, we need a valid type to assign to that variable. However, because of the formation rules, a type is only valid with respect to an existing context. So, the context-extension constructor requires both a prior context $\Gamma$ and a type valid in $\Gamma$:
        \[
            \fun{ext}\colon
                \prod_{\Gamma\colon\fun{Con}}
                    \fun{Ty}(\Gamma)\to\fun{Con}
        \]
        Notice how a constructor for the base type $\fun{Con}$ consumes an element of the dependent family $\fun{Ty}$.
    \end{enumerate}
    
    Conversely, the constructors for $\fun{Ty}$ must take elements of $\fun{Con}$ as arguments, because every element of $\fun{Ty}$ must be indexed by a context. 

    The circularity is saved by the existence of the \textsl{primitive types}. These are precisely those types that form the base cases of the recursive structure. They require only a context and no prior types or variables to be well-formed.
    
    Primitive types are usually restricted to $\fun0$, $\fun1$, $\fun2$, $\N$, and the universe $\univ$. More precisely, we have
    \[
        \fun{empty},
        \fun{unit},
        \fun{bool},
        \fun{nat},
        \fun{univ}
            \colon\prod_{\Gamma\colon\fun{Con}}
                \fun{Ty}(\Gamma),
    \]
    which establishes that these types are valid in any context.

    In standard type theory, this is often written as an inference rule stating that if $\Gamma$ is a valid context, then $\N$ is a valid type under $\Gamma$:
    \[
        \frac{\vdash\Gamma\text{ ctx}}
            {\Gamma\vdash\N\text{ type}}
    \]
\end{xmpl}

\subsection*{The General Schema}

In general, an inductive-inductive definition consists of the simultaneous specification of a type $A$ and a type family $B$ indexed by $A$:
\begin{align*}
    A &\colon \univ \\
    B &\colon A \to \univ
\end{align*}
along with a set of constructors for $A$ and a set of constructors for $B$. 

The constructors for $A$ specify how to build elements of $A$, and they may take arguments of type $A$ as well as arguments of type $B(a)$ for previously constructed elements $a \colon A$.

The constructors for $B$ specify how to build elements of $B(a)$ for a given $a \colon A$. These constructors may depend on elements of $A$ and elements of $B(a')$ for previously constructed elements $a' \colon A$.

The elimination principle for such a definition must be mutually defined: we simultaneously specify a motive for $A$ and a motive for $B$, and provide methods for all constructors of both $A$ and $B$.

To understand the formal structure, let us look at the four standard components of this definition:

\textbf{Formation Rules:}
\[
    A\colon\univ
    \quad\text{and}\quad
    B\colon A\to\univ.
\]

\textbf{Constructors:}
\begin{align*}
    \fun c_A
        &\colon\prod_{x\colon A}B(x)\to A,\\
    \fun c_B
        &\colon
            \prod_{x\colon A}\;
            \prod_{y\colon B(x)}
                B(c_A(x, y)).
\end{align*}

\textbf{Induction Principle:} We must specify two motives:

The motive for $A$ is standard:
\begin{align*}
    P_A
        &\colon A\to\univ.
\intertext{However, the motive for $B$ is specialized:}
    P_B
        &\colon
            \prod_{a\colon A}\;
            \prod_{b\colon B(a)}
                P_A(a)\to\univ.
\end{align*}
Given these motives, the elimination principle states that if we provide a method for every constructor of $A$ and every constructor of $B$, we obtain a mutually defined pair of functions:
\begin{align*}
    f_A &\colon\prod_{a\colon A}P_A(a)\\
    f_B &\colon
        \prod_{a\colon A}\;
        \prod_{b\colon B(a)}
            P_B(a,b,f_A(a))
\end{align*}
To write the computation rules explicitly, we need to define two additional eliminator methods:
\[
    m_A\colon
        \prod_{x\colon A}\;
        \prod_{y\colon B(x)}\;
        \prod_{p_x\colon P_A(x)}\;
        \prod_{p_y\colon P_B(x,y,p_x)}
            P_A(\fun c_A(x,y))
\]
and
\[
    m_B\colon
        \prod_{x\colon A}\;
        \prod_{y\colon B(x)}\;
        \prod_{p_x\colon P_A(x)}\;
        \prod_{p_y\colon P_B(x,y,p_x)} 
            P_B\bigl(
                \fun c_A(x,y),
                \fun c_B(x,y),
                m_A(x,y,p_x,p_y)
            \bigr)
\]

\textbf{Computation Rules:}
\begin{align*}
    f_A(\fun c_A(x,y)) 
        &\equiv m_A\bigl(
            x,y,f_A(x),f_B(x,y)
        \bigr)\\
    f_B(\fun c_A(x,y),\fun c_B(x, y)) 
        &\equiv m_B\bigl(
            x,y,f_A(x),f_B(x,y)
        \bigr).
\end{align*}
Notice how the computation of $f_B$ on the constructor $\fun c_B$ is synchronized with the computation of $f_A$ on the constructor~$\fun c_A$.

\section{Identity Types and Identity Systems}

\paragraph{Path Induction}\ 

Let $A$ be a type. The identity type family
\[
    \dummy\eq A\dummy\colon A\to A\to\univ 
\]
can be seen as an inductive family with two indices. Following Definition~\ref{defn:no-recursive-args} for a constructor with no recursive arguments, we map the components as follows:

\begin{itemize}
    \item \textsc{Index Type:} Uncurrying gives $I\defeq A\times A$.
    
    \item \textsc{Family:} The type family $X\colon A\times A\to\univ$ is defined by $X(a,b)\defeq a\eq Ab$.
    
    \item \textsc{Context:} The constructor $\fun{refl}$ requires one non-recursive argument in $A$. Thus, $C\defeq A$.
    
    \item \textsc{Target Index:} The constructor $\fun{refl}$ yields an element of $a\eq A a$; hence, the function $g\colon C\to I$ is given by $g(a)\defeq(a,a)$.
\end{itemize}

By the general signature \eqref{eq:no-recursive-args}, the constructor is:
\[
    \fun{refl}\colon\prod_{z\colon C}X(g(z))
        \equiv\prod_{a\colon A}a\eq Aa.
\]

To derive the induction principle, consider a motive of type:
\[
    \prod_{(a,b)\colon A\times A}X(a,b)\to\univ.
\]
By currying, this is equivalent to
\[
    P\colon\prod_{a,b\colon A}(a\eq Ab)\to\univ.
\]
According to Definition~\ref{defn:no-recursive-args}, the eliminator method for our constructor has the signature
\[
    p_{\fun{refl}}\colon\prod_{a\colon A}P(a,a,\fun{refl}_a).
\]
The general eliminator $\fun{ind}_X$ produces the dependent function:
\[
    f\colon
        \prod_{a,b\colon A}\;
        \prod_{p\colon a\eq Ab}
        P(a,b,p),
\]
which satisfies the computation rule
\[
    f(a,a,\fun{refl}_a)\equiv p_{\fun{refl}}(a).
\]

\paragraph{Based Path Induction}\ 

Let us now fix a parameter $a_0\colon A$. We can define the based identity type family
\[
    a_0\eq A\dummy\colon A\to\univ
\]
as an inductive family with one index. Following Definition~\ref{defn:no-recursive-args} again we have:
\begin{itemize}
    \item \textsc{Index Type:} $I\defeq A$.
    
    \item \textsc{Family:} The type family $X\colon A\to\univ$ is defined by $X(b)\defeq a_0\eq Ab$.
    
    \item \textsc{Context:} Since the constructor has no arguments, the context is $\fun1$.
    
    \item \textsc{Target Index:} The constructor yields an element of $a_0\eq Aa_0$, so the target index function $g\colon\fun1\to I$ is given by $g(\fstar)\defeq a_0$.
\end{itemize}

By the general signature \eqref{eq:no-recursive-args}, the constructor is:
\[
    \fun{refl}_{a_0}\colon\prod_{z\colon\fun1}X(g(z))
        \equiv a_0\eq A a_0.
\]

To derive the based path induction principle, consider a motive of type:
\[
    P\colon\prod_{b\colon A}(a_0\eq Ab)\to\univ.
\]
According to Definition~\ref{defn:no-recursive-args}, the eliminator method for our constructor has the signature
\[
    p_{\fun{refl}}\colon\prod_{z\colon\fun1}
        P(g(z),\fun{refl}(a_0))
        \equiv P(a_0,\fun{refl}_{a_0}).
\]
Thus, the method reduces to providing an element $d\colon P(a_0,\fun{refl}_{a_0})$.

The general eliminator $\fun{ind}_X$ produces the dependent function:
\[
    f\colon
        \prod_{b\colon A}\;
        \prod_{p\colon a_0\eq Ab}
        P(b,p),
\]
which satisfies the computation rule
\[
    f(a_0,\fun{refl}_{a_0})\equiv d.
\]

{\small
\textbf{Note.} For decades, long before the introduction of HoTT, type theorists believed that $\fun{refl}$ could only construct trivial paths. This intuition originated in the behavior of standard inductive types, where, for instance, every natural number is zero or a successor,  every boolean term is $0_{\fun2}$ or $1_{\fun2}$, etc. However, in 1993, Martin Hofmann and Thomas Streicher constructed a mathematical model in which types are interpreted as groupoids and identity types as the spaces of morphisms between objects. In their model, non-trivial automorphisms existed within the strict syntactic rules of Martin-Löf Type Theory. This showed that the type theory was already describing homotopy spaces.

The example given in Exercise~\ref{exr:not}, which shows that $\fun{not}\colon\fun2\to\fun2$ is an involution, allows us to conclude via the univalence axiom that the identity type $\fun2\eq\univ\fun2$ is not trivial. Specifically, it contains at least two distinct paths: the constant path $\fun{refl}_{\fun2}$ and the path $\fun{ua}(\fun{not})$ induced by negation.

More generally, while the inductive definition of the identity family provides the architectural scaffolding, the structural richness of identity paths is inextricably tied to univalence. Without this axiom, the identity type acts merely as a formal equality symbol; with it, the type acquires the full expressiveness of symmetries and transformations.
}

\begin{rem}\label{rem:yoneda-path-induction-correspondence}
    The based path induction principle is the type-theoretic incarnation of the Yoneda lemma~\ref{thm:yoneda-general} from category theory. 
    
    As described in \S\,\ref{types-as-categories}, types can be viewed as categories and identity paths as morphisms between them. Under this analogy, a type family $C\colon A\to\univ$ acts as a functor from this category to the universe of spaces. For this correspondence to hold precisely, it is crucial to let $\univ$ represent the category whose objects are types (as usual) and whose morphisms are non-dependent functions (rather than identity paths).
    
    Under this interpretation, every type family $P\colon A\to\univ$ is a functor. Specifically, for every $x,y\colon A$ and $p\colon x\eq Ay$, its action on morphisms is given by
    \[
        P(p)\defeq
            \fun{transport}^P(p,\dummy)
            \colon P(x)\to P(y).
    \]
    The functoriality of $P$ follows from Theorem~\ref{thm:transportP} and Lemma~\ref{lem:transport-composition}. Although path concatenation is written left-to-right, $P$ is covariant. 
    
    Consider two type families $R,S\colon A\to\univ$. Given a dependent function
    \[
        g\colon\prod_{b\colon A}R(b)\to S(b),
    \]
    for any path $p\colon x\eq{A}y$, we obtain two maps of type $R(x)\to S(y)$:
    \[
        g(y,\fun{transport}^R(p,\dummy))
        \quad\text{and}\quad
        \fun{transport}^S(p,g(x,\dummy)).
    \]
    By Lemma~\ref{lem:dependent-post-composition-transport}, for $r\colon R(x)$, these maps satisfy the propositional equality
    \[
        g(y,\fun{transport}^R(p,r))
        \eq{S(y)}
        \fun{transport}^S(p,g(x,r)).
    \]
    This yields the following propositionally commutative diagram:
    \[
        \begin{tikzcd}[row sep=large,column sep=2cm]
            R(x)
                \arrow[r,
                    "{\fun{transport}^R(p,\dummy)}"
                    ]
                \arrow[d,"g(x)"']
            &R(y)
                \arrow[d,"g(y)"]\\
            S(x)
                \arrow[r,
                    "{\fun{transport}^S(p,\dummy)}"'
                ]
            &S(y)
        \end{tikzcd}
    \]
    Thus, every dependent function $g\colon\prod_{b\colon A}R(b)\to S(b)$ can be seen as a natural transformation from $R$ to $S$. 
    
    In particular, a natural transformation from the representable functor $a_0\eq A\dummy$ to a type family $C$ corresponds to a homotopy $\eta\colon(a_0\eq A\dummy)\sim C$, which is judgmentally equal to a dependent function
    \[
        \eta\colon\prod_{b\colon A}(a_0\eq Ab)\to C(b).
    \]
    The following table summarizes the analogy:
    {\small
    \[
        \begin{tabular}{ll}
            \textbf{Category Theory}
            &\textbf{Type Theory}\\
            \hline
            \rule{0mm}{3mm}%
            Categories $\cat A$, $\cat{Set}$
            &Types $A$, $\univ$\\
            Objects $a,b$
            &Points $a,b\colon A$\\
            Morphisms $f\colon a\to b$
            &Paths $p\colon a\eq Ab$, or\\
            &maps $g\colon A\to B$ in $\univ$\\
            Functor $C\colon\cat A\to\cat{Set}$
            &Type family $C\colon A\to\univ$\\
            Representable functor $\Hom(a_0,\dummy)$
            &Type family $a_0\eq A\dummy$\\
            Natural transformation $\eta\colon\Hom(a_0,\dummy)\to C$
            &Homotopy $\eta\colon(a_0\eq A\dummy)\sim C$\\
            $\id_{a_0}\in\Hom(a_0,a_0)$
            &$\fun{refl}_{a_0}\colon a_0\eq Aa_0$\\
            $\eta_{a_0}(\id_{a_0})\in C(a_0)$
            &$\eta(a_0,\fun{refl}_{a_0})\colon C(a_0)$\\
            Yoneda bijection
            &Based path induction\\
            %\hline
        \end{tabular}
    \]
    }
\end{rem}

\begin{defns}
    Let $A$ be a type and $a_0 \colon A$ an element.
    \begin{enumerate}[-]
        \item A \textsl{pointed predicate} over $\dep{A,a_0}\colon\puniv$ is a family $R\colon A\to\univ$ equipped with a \textsl{basepoint} $r_0\colon R(a_0)$.
        
        \item Given pointed predicates $\dep{R,r_0}$ and $\dep{S,s_0}$, a family of maps
        \[
            g\colon\prod_{b\colon A}R(b)\to S(b)
        \]
        is said to be \textsl{pointed} if $g(a_0,r_0)\eq{S(a_0)}s_0$. The type of such pointed maps is defined as
        \[
            \fun{ppmap}(R,S)\defeq
                \sum_{
                    g\colon\prod_{b\colon A}R(b)\to S(b)
                }
                g(a_0,r_0)\eq{S(a_0)}s_0.
        \]
        
        \item An \textsl{identity system at $a_0$} is a pointed predicate $\dep{R,r_0}$ such that for any type family
        \[
            D\colon\prod_{b\colon A}R(b)\to\univ
        \]
        and any element $d\colon D(a_0,r_0)$, there exists a dependent function
        \[
            f\colon
                \prod_{b\colon A}\;
                \prod_{r\colon R(b)}
                    D(b,r)
        \]
        satisfying $f(a_0,r_0)\eq{D(a_0,r_0)}d$.
    \end{enumerate}
\end{defns}

\begin{rems}\label{rem:ppmap-and-fibers}\ 
    \begin{enumerate}[-]
        \item The canonical example of a pointed predicate over $\dep{A,a_0}$ is the identity family $(a_0\eq A\dummy)$ equipped with the basepoint $\fun{refl}_{a_0}$.

        An arbitrary pointed predicate $\dep{R,r_0}$ is an identity system if it satisfies the exact same elimination principle (based path induction) as the identity type. In this sense, identity systems generalize the role representable functors play in the Yoneda lemma.

        \item An alternative way to interpret the type $\fun{ppmap}(R,S)$ is by recognizing it as the fiber of the evaluation map at $s_0$, i.e.,
        \begin{equation}\label{eq:ppmap-and-fibers}
            \fun{ppmap}(R,S)\equiv
                \fun{fib}_{\ev{(a_0,r_0)}}(s_0).
        \end{equation}
    \end{enumerate}
\end{rems}

\begin{thm}\label{thm:identity-systems-are-representable}
    Let\/ $A$ be a type and\/ $a_0\colon A$. If\/ $\dep{R,r_0}$ is an identity system at\/ $a_0$, then for all\/ $b\colon A$, there is an equivalence\/ $R(b)\simeq(a_0\eq Ab)$.
\end{thm}

\begin{proof}
    Because $\dep{a_0\eq A\dummy,\fun{refl}_{a_0}}$ is an identity system, we can define a dependent function
    \[
        f\colon\prod_{b\colon A}(a_0\eq Ab)\to R(b)
    \]
    by taking $R$ as our motive and $r_0\colon R(a_0)$ as the base case. The computation rule for based path induction yields the definitional equality $f(a_0,\fun{refl}_{a_0})\equiv r_0$.
    
    Conversely, because $\dep{R,r_0}$ is an identity system, we can define a dependent function 
    \[
        g\colon\prod_{b\colon A}R(b)\to (a_0\eq Ab)
    \]
    by taking $a_0\eq A\dummy$ as our motive and $\fun{refl}_{a_0}\colon a_0\eq Aa_0$ as the base case. By the definition of an identity system, we obtain a propositional equality
    \[
        p\colon g(a_0,r_0)
        \eq{a_0\eq Aa_0}
        \fun{refl}_{a_0}.
    \]
    
    To show that $f$ and $g$ are quasi-inverses, we must construct the homotopies
    \[
        g(b,f(b,\dummy))\sim\id_{a_0\eq Ab}
    \quad\text{and}\quad
        f(b,g(b,\dummy))\sim\id_{R(b)}.
    \]    
    
    For the first homotopy, we proceed by based path induction on $q\colon a_0\eq Ab$. It suffices to consider the case $b\equiv a_0$ and $q\equiv\fun{refl}_{a_0}$. This requires a path
    \[
        g(a_0,f(a_0,\fun{refl}_{a_0}))
        \eq{a_0\eq Aa_0}
        \fun{refl}_{a_0}.
    \]
    Since $f(a_0,\fun{refl}_{a_0})\equiv r_0$, the goal reduces to $g(a_0,r_0)\eq{a_0\eq Aa_0}\fun{refl}_{a_0}$, which is exactly the path $p$.
    
    For the second homotopy, we proceed by the induction principle of the identity system $R$ on $r\colon R(b)$. It suffices to consider the case $b\equiv a_0$ and $r\equiv r_0$. This requires a path 
    \[
        f(a_0,g(a_0,r_0))\eq{R(a_0)}r_0.
    \]
    Applying the action on paths of $f(a_0,\dummy)$ to $p$, we obtain a path 
    \[
        f(a_0,g(a_0,r_0))
        \eq{R(a_0)}
        f(a_0,\fun{refl}_{a_0}).
    \]
    Since $f(a_0,\fun{refl}_{a_0})\equiv r_0$, the required path is reached.
    
    Therefore, $f$ and $g$ are quasi-inverses, completing the proof.
    %
    
\end{proof}

\begin{rems}\label{rem:f-is-transport}\ 
    \begin{enumerate}[-]
        \item The forward map $f\colon\prod_{b\colon A}(a_0\eq Ab)\to R(b)$ constructed via based path induction in the theorem is definitionally equal to the transport operation applied to the base point $r_0$. In other words
        \[
            f(b,p)\equiv\fun{transport}^R(p,r_0).
        \]
        In this sense, an identity system $\dep{R,r_0}$ can be viewed as a space where transporting the root $r_0$ along paths completely and uniquely traces out the entirety of the family $R$. To geometrically visualize this, imagine $R$ as a fibration over $A$, with a point $r_0$ in the fiber over $a_0$. As $r_0$ is dragged along all possible paths in~$A$ with origin at $a_0$, the point $r_0$ transports onto a subspace of $R$. The theorem establishes that this transporting action reaches the entirety of $R$, without ever overlapping itself.

        \item The Yoneda lemma establishes a bijection between natural transformations $\eta\colon\Hom(a_0,\dummy)\to F$ and elements of $F(a_0)$, given by evaluating the component $\eta_{a_0}$ at the identity morphism $\id_{a_0}$. In the proof above, $r_0\colon R(a_0)$ is precisely the selection for this image of the identity morphism. Just as in the Yoneda lemma, this choice of $r_0$ uniquely determines the forward transformation, which is exactly how we defined the dependent function $f$ via based path induction.
    \end{enumerate}
\end{rems}

\begin{thm}\label{thm:ppmap-contractibility}
    A pointed predicate\/ $\dep{R,r_0}$ over\/ $\dep{A,a_0}$ is an identity system if, and only if, for any pointed predicate\/ $\dep{S,s_0}$, the type\/ $\fun{ppmap}(R,S)$ is contractible.
\end{thm}

\begin{proof}
    Suppose that $\dep{R,r_0}$ is an identity system. The induction principle of $\dep{R,r_0}$ applied to the motive $D(b,r)\defeq S(b)$ and the base element $s_0\colon D(a_0,r_0)$ yields a function
    \[
        f\colon\prod_{b\colon A}\;
            \prod_{r\colon R(b)}S(b)
    \]
    and a path $p\colon f(a_0,r_0)\eq{S(a_0)}s_0$. Thus, the dependent pair $\dep{f,p}$ is an element of $\fun{ppmap}(R,S)$. We claim that this pair is a center of contraction.
    
    Given $\dep{g,q}\colon\fun{ppmap}(R,S)$, we must construct a path
    \[
        \dep{f,p}\eq{\fun{ppmap}(R,S)}\dep{g,q}
    \]
    consisting of two parts: a path $\epsilon\colon f\eq{}g$, and a path $\zeta\colon\fun{transport}^E(\epsilon,p)\eq{}q$, where $E(h)\defeq h(a_0,r_0)\eq{S(a_0)}s_0$.

    To construct $\epsilon$, it suffices to provide a homotopy
    \[
        \eta\colon
            \prod_{b\colon A}\;
            \prod_{r\colon R(b)}
                f(b,r)\eq{S(b)}g(b,r)
    \]
    and define $\epsilon\defeq\fun{funext}(\eta)$.

    To construct $\eta$, we again use the induction principle of $\dep{R,r_0}$. We define the motive as $M(b,r)\defeq f(b,r)\eq{S(b)}g(b,r)$ and use the path $p\ct q^{-1}\colon M(a_0,r_0)$ for the base case.

    By the computation rule, $\eta$ yields a path
    \[
        \beta\colon\eta(a_0,r_0)\eq{}p\ct q^{-1}.
    \]
    To construct $\zeta$, observing that $E\equiv(\lambda u.\,u\eq{}s_0)\circ\ev{(a_0,r_0)}$, we obtain
    \begin{align*}
        \fun{transport}^E(\epsilon,p)
            &\eq{}\fun{transport}^{u\mapsto u\eq{}s_0}(
                \fun{ap}_{\ev{(a_0,r_0)}}(\epsilon),p
                )
                &&\text{; Lem.~\ref{lem:post-composition-transport}}\\
            &\eq{}\fun{ap}_{\ev{(a_0,r_0)}}(\epsilon)^{-1}\ct p
                &&\text{; Prop.~\ref{prop:transport-decomposition-for-paths}}\\
            &\eq{}\fun{happly}(\epsilon,(a_0,r_0))^{-1}\ct p
                &&\text{; Eq.~\ref{eq:happly-is-ap-ev}}\\
            &\equiv\fun{happly}(\fun{funext}(\eta),(a_0,r_0))^{-1}\ct p\\
            &\eq{}\eta(a_0,r_0)^{-1}\ct p\\
            &\eq{}(p\ct q^{-1})^{-1}\ct p
                &&\text{; }\beta\\
            &\eq{}q.
    \end{align*}
    This completes the construction of $\zeta\colon\fun{transport}^E(\epsilon,p)\eq{}q$.

    Conversely, suppose that $\fun{ppmap}(R,S)$ is contractible for any pointed predicate $\dep{S,s_0}$. We must show that $\dep{R,r_0}$ is an identity system. Let $D$ be an arbitrary type family over $R$, and let $d_0\colon D(a_0,r_0)$ be a base point. The pointed predicate $\dep{D,d_0}$ fulfills our hypothesis; thus, according to equation~\eqref{eq:ppmap-and-fibers}, the type
    \[
        \fun{ppmap}(R,D)\equiv\fun{fib}_{\ev{(a_0,r_0)}}(d_0)
    \]
    is contractible. Since this holds for all base points $d_0$, Remark~\ref{rem:equivalence-fib-is-contractible} ensures that the evaluation map $\ev{(a_0,r_0)}$ is an equivalence. Therefore, $\dep{R,r_0}$ is an identity system.
    %
    
\end{proof}


\begin{thm}\label{thm:local=identity-system-equivalences}
    Let\/ $\dep{R,r_0}$ be a pointed predicate over\/ $\dep{A,a_0}$. Then, the following conditions are logically equivalent:
    \begin{enumerate}[a),font=\upshape]
        \item The pointed predicate\/ $\dep{R,r_0}$ is an identity system at\/ $a_0$.
        
        \item For any\/ $b\colon A$, the function\/ $\fun{transport}^R(\dummy,r_0)\colon(a_0\eq{A}b)\to R(b)$ is an equivalence.
        
        \item The type\/ $\Sigma_AR$ is contractible.

        \item For any pointed predicate\/ $\dep{S,s_0}$, the type\/ $\fun{ppmap}(R,S)$ is contractible.
    \end{enumerate}
\end{thm}

\begin{proof}\ 
    \begin{enumerate}[a),font=\upshape]
        \item $\Rightarrow$~b) This is a direct consequence of Theorem~\ref{thm:identity-systems-are-representable} and Remark~\ref{rem:f-is-transport}.

        \item $\Rightarrow$~c) The equivalence $(a_0\eq{A}b)\simeq R(b)$ induces an equivalence
        \[
            \Bigl(\sum_{b\colon A}a_0\eq{A}b\Bigr)
            \simeq
            \Bigl(\sum_{b\colon A}R(b)\Bigr).
        \]
        Hence, the conclusion follows from Lemma~\ref{lem:Sigma-identity-is-contractible}.

        \item $\Rightarrow$~a) Consider the dependent pair $\dep{a_0,r_0}$ in the total space $\Sigma_AR$. Given a type family
        \[
            D\colon\prod_{b\colon A}R(b)\to\univ
        \]
        and a term $d\colon D(a_0,r_0)$, we must construct a function
        \[
            f\colon\prod_{b\colon A}\;
                \prod_{r\colon R(b)}D(b,r)
        \]
        satisfying $f(a_0,r_0)\eq{D(a_0,r_0)}d$.
        
        First, observe that uncurrying produces the family $\bar{D}\colon\Sigma_AR\to\univ$, defined by $\bar{D}(\dep{b,r})\defeq D(b,r)$. 
        
        Given $b\colon A$ and $r\colon R(b)$, we obtain the dependent pair $\dep{b,r}\colon\Sigma_AR$. By contractibility, there is a path $\omega\colon\dep{a_0,r_0}\eq{\Sigma_AR}\dep{b,r}$. Thus, we can define
        \[
            f(b,r)\defeq
                \fun{transport}^{\bar{D}}(\omega,d)
                \colon D(b,r).
        \]
        This map satisfies the computation rule because $\omega$ reduces definitionally to $\fun{refl}_{\dep{a_0,r_0}}$, making the transport definitionally equal to the identity function.

        \item[a) $\Leftrightarrow$~d)] This equivalence is established in Theorem~\ref{thm:ppmap-contractibility}.
    \end{enumerate}
\end{proof}

\begin{defns}
    Let $A$ be a type.
    \begin{enumerate}[-]
        \item A \textsl{reflexive relation} over $A$ is a binary type family $R\colon A\to A\to\univ$ equipped with a \textsl{reflexivity function} $r_0\colon\prod_{a\colon A}R(a,a)$.
        
        \item Given two reflexive relations $\dep{R,r_0}$ and $\dep{S,s_0}$ over $A$, a \textsl{morphism of reflexive relations} is a family of maps
        \[
            g\colon\prod_{a,b\colon A}R(a,b)\to S(a,b)
        \]
        that propositionally preserves the reflexivity elements, meaning there is a family of paths 
        \[
            p\colon\prod_{a\colon A}
                g(a,a,r_0(a))\eq{S(a,a)}s_0(a).
        \]
        
        \item The type of such morphisms is denoted by
        \[
            \fun{rrmap}(R,S)\defeq
            \sum_{g\colon\prod_{a,b\colon A}R(a,b)\to S(a,b)}\;
                \prod_{a\colon A}g(a,a,r_0(a))\eq{S(a,a)}s_0(a).
        \]
        
        \item An \textsl{identity system over $A$} is a reflexive relation $\dep{R,r_0}$
        such that for any type family 
        \[
            D\colon\prod_{a,b\colon A}R(a,b)\to\univ
        \]
        and any
        \[
            d\colon\prod_{a\colon A}D(a,a,r_0(a)),
        \]
        there exists a function
        \[
            f\colon
                \prod_{a,b\colon A}\;
                \prod_{r\colon R(a,b)}
                    D(a,b,r)
        \]
        satisfying $f(a,a,r_0(a))\eq{D(a,a,r_0(a))}d(a)$ for all $a\colon A$.
    \end{enumerate}
\end{defns}

\begin{rem}
    Identity systems generalize the concept of based path induction to a global relation over the entire type $A$. Instead of considering the paths departing from a specific base point, the binary family $R\colon A\to A\to\univ$ characterizes paths between any two arbitrary points.
    
    As the following theorem shows, this definition axiomatizes equality geometrically. If a binary relation $R$ and a reflexivity term $r_0$ satisfy this induction principle, then the space $R(a,b)$ is equivalent to the built-in identity type $(a\eq{A}b)$. 
    
    To formally establish that ``isomorphic structures are equal,'' instead of constructing (potentially complex) paths in the universe, it suffices to define $R(a,b)$ to mean that $a$ and $b$ are isomorphic, and then prove that $\dep{R,r_0}$ forms a global identity system. The induction principle then promotes those isomorphisms to propositional equalities.
\end{rem}

\begin{thm}\label{thm:global-identity-system-equivalences}
    Let\/ $R\colon A\to A\to\univ$ be a type family and\/ $r_0\colon\prod_{a\colon A}R(a,a)$. Then, the following conditions are logically equivalent:
    \begin{enumerate}[a),font=\upshape]
        \item The pair\/ $\dep{R,r_0}$ is an identity system over\/ $A$.
        
        \item For all\/ $a\colon A$, the pointed predicate\/ $\dep{R(a),r_0(a)}$ is an identity system at\/~$a$.
        
        \item For any\/ $a\colon A$, the type\/ $\sum_{b\colon A}R(a,b)$ is contractible.
        
        \item For any\/ $a,b\colon A$, the map\/ $\fun{transport}^{R(a)}(\dummy,r_0(a))\colon(a\eq{A}b)\to R(a,b)$ is an equivalence.
        
        \item For any reflexive relation\/ $\dep{S,s_0}$ over\/ $A$, the type of morphisms of reflexive relations\/ $\fun{rrmap}(R,S)$ is contractible.
    \end{enumerate}
\end{thm}

\begin{proof}\ 
    \begin{enumerate}[a),font=\upshape]
        \item $\Rightarrow$~b) Fix $a\colon A$. Given a local motive $C\colon\prod_{y\colon A}R(a,y)\to\univ$ and a base case $c_0\colon C(a,r_0(a))$, we must construct a dependent function of type
        \[
            \prod_{y\colon A}\;
            \prod_{r\colon R(a,y)}
                C(y,r).
        \]
        To apply the global induction principle, we define a motive
        \begin{align*}
            D&\colon\prod_{x,y\colon A}R(x,y)\to\univ\\
            D(x,y,r)&\defeq
                \prod_{p\colon a\eq{A}x}
                    C\bigl(
                        y,
                        \fun{transport}^{R(\dummy,y)}
                            (p^{-1},r)
                    \bigr),
        \end{align*}
        which type-checks because $\fun{transport}^{R(\dummy,y)}(p^{-1})
                \colon R(x,y)\to R(a,y)$.
        
        To provide a global base case $d\colon\prod_{x\colon A}D(x,x,r_0(x))$, we construct $d(x)$ for a given $x\colon A$. By path induction on $p\colon a\eq{A}x$, it suffices to consider the case where $x\equiv a$ and $p\equiv\fun{refl}_{a}$. In this case, the required type reduces to $C(a,r_0(a))$, so we may define
        \begin{equation}\label{eq:global-base-case}
            d(a,\fun{refl}_{a})\defeq c_0.
        \end{equation}
        By the global induction principle of $\dep{R,r_0}$, there exists a function
        \[
            f\colon\prod_{x,y\colon A}\;
                \prod_{r\colon R(x,y)}D(x,y,r)
        \]
        such that $f(x,x,r_0(x))\eq{}d(x)$ for all $x\colon A$.
        
        Thus, for the required local section we have
        \begin{align}\label{eq:g(y,r)-defn}
            g&\colon
                \prod_{y\colon A}\;
                \prod_{r\colon R(a,y)}C(y,r)
                    \nonumber\\
            g(y,r)&\defeq f(a,y,r,\fun{refl}_{a}).
        \end{align}
        This is well-defined because the type of the application is
        \[
            f(a,y,r,\fun{refl}_{a})\colon 
                C\bigl(y,\fun{transport}^{R(\dummy,y)}
                    (\fun{refl}_{a}^{-1},r)\bigr),
        \]
        which reduces definitionally to $C(y,r)$ since $\fun{transport}^{R(\dummy,y)}(\fun{refl}_{a}^{-1},r)\equiv r$.
        
        For the computation rule of $g$, we have
        \begin{align*}
            g(a,r_0(a))
                &\equiv
                    f(a,a,r_0(a),\fun{refl}_{a})
                &&\text{; \eqref{eq:g(y,r)-defn}}\\
                &\eq{}d(a,\fun{refl}_{a})
                &&\text{; comp.~rule for $f$}\\
                &\equiv c_0
                &&\text{; \eqref{eq:global-base-case}.}
        \end{align*}
        Thus, $\dep{R(a),r_0(a)}$ is an identity system at $a$.

        \item $\Leftrightarrow$~c) This is a direct consequence of the logical equivalence a)~$\Leftrightarrow$~c) of Theorem~\ref{thm:local=identity-system-equivalences}.

        \item $\Leftrightarrow$~d) This is a direct consequence of the logical equivalence c)~$\Leftrightarrow$~b) of Theorem~\ref{thm:local=identity-system-equivalences}.

        \item $\Rightarrow$~e) Since d)~$\Leftrightarrow$~c)~$\Leftrightarrow$~b), we can assume that for all $a\colon A$, the pointed predicate $\dep{R(a),r_0(a)}$ is an identity system at $a$.
    
        Now, consider the type from condition e):
        \begin{align*}
            T&\defeq
                \sum_{g\colon\prod_{a,b\colon A}
                    R(a,b)\to S(a,b)}\;
                    \prod_{a\colon A}
                        g(a,a,r_0(a))
                        \eq{S(a,a)}
                        s_0(a)
        \end{align*}
        By currying the domain of $g$ and applying \nref{lpar:axiom-of-choice}, $T$ is equivalent to the type
        \begin{align*}
            T'&\defeq
                \prod_{a\colon A}\;
                \sum_{
                    h\colon\prod_{b\colon A}
                        R(a,b)\to S(a,b)
                    }h(a,r_0(a))\eq{S(a,a)}s_0(a)\\
                &\equiv\prod_{a\colon A}\fun{ppmap}
                    (R(a),S(a))   
        \end{align*}
        According to part~d) of Theorem~\ref{thm:local=identity-system-equivalences}, $\fun{ppmap}(R(a),S(a))$ is contractible. By Theorem~\ref{thm:contractible-types}, it follows that $T'$, and hence $T$, is contractible.

        \item $\Rightarrow$~a) We will show that condition a) holds by establishing the chain of implications
        \[
        \text{e)}
            \Rightarrow\text{ c)}
            \stackrel{\mathstrut\checkmark}\Leftrightarrow\text{ b) }
            \Rightarrow\text{ a)}.
        \]
        To prove c), we apply condition e) to the type family
        \[
            S(a,b)\defeq\Sigma_AR(a,\dummy)
        \]
        and the base points $s_0(a)\defeq\dep{a,r_0(a)}$. By part~e), the type
        \[
            T\defeq\sum_{
                g\colon\prod_{a,b\colon A}
                    R(a,b)\to\Sigma_AR(a,\dummy)
                }\;
                \prod_{a\colon A}
                    g(a,a,r_0(a))
                    \eq{S(a,a)}
                    \dep{a,r_0(a)}
        \]
        is contractible.
        
        Consider the elements
        $\dep{g_1,p_1}, \dep{g_2,p_2}\colon T$, defined by
        \begin{align*}
            g_1(a,b,r)&\defeq\dep{b,r}
            &\text{and}&&
            p_1(a)&\defeq\fun{refl}_{\dep{a,r_0(a)}},\\
            g_2(a,b,r)&\defeq\dep{a,r_0(a)}
            &\text{and}&&
            p_2(a)&\defeq\fun{refl}_{\dep{a,r_0(a)}}.
        \end{align*}
        These elements are well-defined because
        \[
            g_1(a,a,r_0(a)),\;g_2(a,a,r_0(a))
                \equiv\dep{a,r_0(a)}.
        \]
        Since $T$ is contractible, both pairs are equal, which implies $g_1\eq{}g_2$. Hence, for any $a,b\colon A$ and $r\colon R(a,b)$, we have $\dep{b,r}\eq{}\dep{a,r_0(a)}$. 
        
        This means that for any $a\colon A$, every element in the total space $\Sigma_AR(a,\dummy)$ is equal to the center $\dep{a,r_0(a)}$. Thus, $\Sigma_AR(a,\dummy)$ is contractible, establishing condition~c).
    
        Having established condition b), we can now derive condition a).
        
        Let
        \[
            D\colon\prod_{x,y\colon A}R(x,y)\to\univ
        \]
        be a global motive with base case
        \[
            d_0\colon\prod_{x\colon A}D(x,x,r_0(x)).
        \]
        For any fixed $a\colon A$, condition b) provides the local induction principle for $R(a)$. We define the local motive $C_a(y,r)\defeq D(a,y,r)$ and the local base case $c_a\defeq d_0(a)\colon C_a(a,r_0(a))$. The local eliminator yields a dependent function
        \[
            f_a\colon\prod_{y\colon A}\;\prod_{r\colon R(a,y)}D(a,y,r)
        \]
        such that $f_a(a,r_0(a))\eq{}d_0(a)$. Thus, we may define the global section $f(x,y,r)\defeq f_x(y,r)$. The computation rule holds by definition:
        \[
            f(x,x,r_0(x))\equiv f_x(x,r_0(x))\eq{}d_0(x)
        \]
        for all $x\colon A$. Therefore, $\dep{R,r_0}$ is a global identity system over $A$, which concludes the proof.
    \end{enumerate}
    
\end{proof}

\begin{cor} {\upshape[Equivalence induction]}
    Given a type family
    \[
        D\colon\prod_{A,B\colon\univ}(A\simeq B)\to\univ
    \]
    and function
    \[
        d\colon\prod_{A\colon\univ}D(A,A,\fun{id}_A),
    \]
    there exists
    \[
        f\colon
            \prod_{A,B\colon\univ}\;
            \prod_{e\colon A\simeq B}D(A,B,e)
    \]
    such that\/ $f(A,A,\fun{id}_A)=d(A)$ for all\/ $A\colon\univ$.
\end{cor}

\begin{proof}
    Define $R(A,B)\defeq A\simeq B$ and $r_0(A)\defeq\fun{id}_A$. By the univalence axiom, for any types $A,B\colon\univ$, the canonical map
    \[
        \fun{idtoeqv}_{A,B}\colon(A\eq{\univ}B)\to(A\simeq B)
    \]
    is an equivalence. 
    
    We claim that $\fun{idtoeqv}_{A,B}(p)$ coincides with $\fun{transport}^{R(A)}(p,\fun{id}_A)$. By path induction on $p\colon A\eq{\univ}B$, it suffices to consider the case when $B\equiv A$ and $p\equiv\fun{refl}_A$. In that case, $\fun{transport}^{R(A)}(\fun{refl}_A,\id_A)\equiv \id_A$ and $\fun{idtoeqv}_{A,A}(\fun{refl}_A)\equiv\id_A$ as well. Consequently, part~d) of Theorem~\ref{thm:global-identity-system-equivalences} is met.
    
    The conclusion is a direct consequence of part~a) of said theorem.
\end{proof}

\begin{cor} {\upshape[Homotopy induction]}
    Given 
    \[
        D\colon\prod_{f,g\colon\prod_{a\colon A}B(a)}(f\sim g)\to\univ
    \]
    and
    \[
        d\colon\prod_{f\colon\prod_{a\colon A}B(a)}
            D(f,f,\lambda x.\,\fun{refl}_{f(x)}),
    \]
    there exists
    \[
        k\colon
            \prod_{f,g\colon\prod_{a\colon A}B(a)}\;
            \prod_{h\colon f\sim g}
                D(f,g,h)
    \]
    such that
    \[
        k(f,f,\lambda x.\,\fun{refl}_{f(x)})=d(f)
    \]
    for all\/ $f$.
\end{cor}

\begin{proof}
    Let $\Pi_AB\defeq\prod_{a\colon A}B(a)$. Define
    \[
        R(f,g)\defeq f\sim g
        \quad\text{and}\quad
        r_0(f)\defeq\lambda x.\,\fun{refl}_{f(x)}.
    \]
    By function extensionality, for any $f,g\colon \Pi_AB$, the canonical map
    \[
        \fun{happly}_{f,g}\colon(f\eq{\Pi_AB}g)\to(f\sim g)
    \]
    is an equivalence.

    We claim that $\fun{happly}_{f,g}(p)$ coincides with $\fun{transport}^{R(f)}(p,r_0(f))$. To see this, we proceed by path induction on $p\colon f\eq{\Pi_AB}g$, which reduces the goal to the case where $g\equiv f$ and $p\equiv\fun{refl}_f$. In this case, the claim holds trivially because both $\fun{happly}_{f,f}(\fun{refl}_f)$ and $\fun{transport}^{R(f)}(\fun{refl}_f,r_0(f))$ are definitionally equal to $r_0(f)$.
    
    Therefore, the function
    \[
        \fun{transport}^{R(f)}(\dummy,r_0(f))\colon(f\eq{\Pi_AB}g)\to R(f,g)
    \]
    is an equivalence for all $f,g\colon \Pi_AB$.

    This establishes condition d) of Theorem~\ref{thm:global-identity-system-equivalences} for the type $\Pi_AB$. By the logical equivalence of that theorem, condition~a) holds, meaning that $\dep{R,r_0}$ is an identity system over $\Pi_AB$. Expanding this definition yields exactly what the corollary states.
\end{proof}
