\section{Initial Algebras Revisited}

\begin{thm}{\upshape[Map Lifting Property]}\label{thm:map-lift}
    Let\/ $P\colon A\to\univ$ be a type family over\/~$A$ and\/ $E\defeq\sum_{x\colon A}P(x)$ its total space. Given a type\/ $B$ and a map\/ $f\colon B\to A$, there exists a function
    \[
        \fun{maplift}\colon
            \Bigl(
                \prod_{y\colon B}P(f(y))
            \Bigr)
            \to\sum_{\hat f\colon B\to E}
                \pi_1^E\circ\hat f\sim f
    \]
    satisfying, for every section\/ $s\colon\prod_{y\colon B}P(f(y))$, the judgmental identities
    \[
        \pi_1(\fun{maplift}(s))\equiv\lambda y.\,\dep{f(y),s(y)}
        \quad\text{and}\quad
        \pi_2(\fun{maplift}(s))\equiv\lambda y.\,\fun{refl}_{f(y)},
    \]
    where\/ $\pi_1^E\colon E\to A$ is the first projection of the total space, and $\pi_1$ and\/ $\pi_2$ denote the canonical projections of the target type of\/~$\fun{maplift}$.
\end{thm}

\begin{proof}
    Given $s\colon\prod_{y\colon B}P(f(y))$, define
    \begin{align*}
        \hat f\colon B&\to E\\
        Y&\mapsto\dep{f(y),s(y)}.
    \end{align*}
    This is well-typed because, for $y\colon B$, we have $f(y)\colon A$ and $s(y)\colon P(f(y))$, and produces the judgmentally commutative diagram
    \[
        \begin{tikzcd}
                &E
                    \arrow[d,"\pi_1"]\\
            B
                    \arrow[ru,"\hat f"]
                    \arrow[r,"f"']
                &A
        \end{tikzcd}
    \]
    By the definitional equality of the first projection on a dependent pair, we have
    \[
        \pi_1^E(\hat{f}(y))\equiv\pi_1^E(\dep{f(y),s(y)})\equiv f(y).
    \]
    Because these are judgmentally equal, the required homotopy $\eta$ between $\pi_1^E\circ\hat{f}$ and $f$ can be given by reflexivity at each point:
    \[
        \eta(y)\defeq\fun{refl}_{f(y)}.
    \]
    Thus, we can define
    \[
        \fun{maplift}(s)\defeq\dep{\hat{f},\eta}. 
    \]
    By construction, we have
    \[
        \pi_1(\fun{maplift}(s))\equiv\hat{f}\equiv\lambda y.\,\dep{f(y),s(y)}
    \]
    and
    \[
        \pi_2(\fun{maplift}(s))\equiv \eta\equiv\lambda y.\,\fun{refl}_{f(y)}.
    \]
\end{proof}

\begin{rem}
    Since the theorem's proof actually establishes a strict, judgmental equality $\pi_1^E\circ\hat{f}\equiv f$ rather than a simple homotopy, one might wonder why the statement claims a weaker conclusion. 
    
    This is necessary because judgmental equalities are meta-theoretic relations used by the formal system; they do not produce any type in the universe $\univ$. 
    
    An expression such as $\sum_{\hat f\colon B\to E}\pi_1^E\circ\hat{f}\equiv f$ is syntactically ill-formed. The homotopy instead acts as a certificate of commutativity that is visible to the internal logic of the theory and can be invoked in subsequent theorems.
\end{rem}

\begin{xmpl}
    The Map Lifting Property bridges the type-theoretic presentation of the induction principle with its categorical interpretation as an initial algebra.
    
    The induction principle for the natural numbers $\N$ with respect to a type family $P\colon\N\to\univ$ requires two pieces of data:
    \begin{enumerate}[a),font=\upshape]
        \item A base case element $b\colon P(0)$.
        \item A step function $s\colon\prod_{n\colon\N} P(n)\to P(\fun{succ}(n))$.
    \end{enumerate}
    We can reinterpret this data by lifting the constructor maps to the total space $\Sigma_\N P$. Following the theorem, we define the base map
    \begin{align*}
        f\colon\explain[B]{\Sigma_\N P}&\to\explain[A]\N\\
        \dep{n,u}&\mapsto\fun{succ}(n). 
    \end{align*}
    The uncurried step function $s$ assigns to every pair $\dep{n,u}\colon\Sigma_\N P$ an element in the fiber $P(\fun{succ}(n))$, which is exactly $P(f(\dep{n,u}))$. Therefore, the step function can be seen as a section
    \[
        s\colon\prod_{e\colon\Sigma_\N P}P(f(e)).
    \]
    By applying the Map Lifting Property, this section $s$ lifts the map $f$ to an endomorphism of the total space bundle, yielding a new map $\hat{f}\colon\Sigma_\N P\to\Sigma_\N P$. Denoting $\hat{f}$ by $\hat{\fun{succ}}$ to stress the categorical interpretation, we obtain the propositionally commutative triangle
    \[
        \begin{tikzcd}
                &\Sigma_\N P
                    \arrow[d,"\pi_1"]\\
            \Sigma_\N P
                    \arrow[ru,"\hat{\fun{succ}}"]
                    \arrow[r,"f"']
                &\N
        \end{tikzcd}
    \]
    where the lifted map sends every pair $\dep{n,u}$ to the pair $\bigdep{\fun{succ}(n),s(\dep{n,u})}$. 
    
    Similarly, we can apply the Map Lifting Property to the base constructor: the constant map $0\colon\fun1\to\N$. The base element $b\colon P(0)$ acts as a section over this map. Lifting it yields a point in the total space bundle given by the map $\hat{0}\colon\fun1\to\Sigma_\N P$, which selects the pair $\dep{0,b}$.
    \[
        \begin{tikzcd}
                &\Sigma_\N P
                    \arrow[d,"\pi_1"]\\
            \fun1
                    \arrow[ru,"\hat{0}"]
                    \arrow[r,"0"']
                &\N
        \end{tikzcd}
    \]
    Categorically, this demonstrates that both the base space $\N$ and the total space $\Sigma_\N P$ are algebras for the polynomial functor $F(X)\defeq\fun1+X$.
    
    The induction principle of $\N$ is equivalent to the statement that $\N$ is the initial algebra in this category: there exists a unique algebra homomorphism $\N\to\Sigma_\N P$ commuting with the constructors. This unique homomorphism is exactly the global dependent section
    \[
        \fun{ind}_\N(P)\colon\prod_{n\colon\N}P(n).
    \]
    Here is the commutative diagram illustrating this initial algebra homomorphism:
    \[
        \begin{tikzcd}[column sep=3.3cm,row sep=huge]
            \fun1+\N
                    \arrow[d,"{[0,\fun{succ}]}"']
                    \arrow[r,
                        "{
                            \id_{\fun1}+
                                \pi_1(
                                    \fun{maplift}(
                                        \fun{ind}_\N(P)
                                    )
                                )
                        }",
                        dashed]
                &\fun1+\Sigma_\N P
                    \arrow[d,"{[\hat{0},\hat{\fun{succ}}]}"]\\
            \N
                    \arrow[r,
                        "{\pi_1(\fun{maplift}(\fun{ind}_\N(P)))}"', dashed]
                &\Sigma_\N P
        \end{tikzcd}
    \]
    The commutativity of this square translates categorical properties into computation rules for the induction principle.
\end{xmpl}

\section{Induction Principles and Dependent Paths}

Like the general inductive types, higher inductive types provide a schema for defining new types generated by constructors. The difference is that higher inductive types may have constructors that produce paths and higher paths. For instance, the constructors of the circle~$\sphere^1$ are:
\begin{itemize}
    \item $\fun{base}\colon\sphere^1$,
    \item $\fun{loop}\colon\fun{base}\eq{\sphere^1}\fun{base}$.
\end{itemize}
This is analogous to the definition of $\N$, whose constructors are $0\colon\N$ and $\fun{succ}\colon\N\to\N$. In the circle, however, the generation extends to higher dimensions, which is natural when types are seen as higher groupoids.

Ordinary constructors, such as $\fun{base}$, are referred as \textsl{point constructors}; the others, such as $\fun{loop}$, as \textsl{path} or \textsl{higher constructors}.

Each path constructor must specify its source and target; for $\fun{loop}$, both are $\fun{base}$. A path constructor generates a new inhabitant of an identity type, which is not \textit{a priori\/} equal to any existing inhabitant. This distinguishes $\sphere^1$ from the inductive type~$\fun1$.

There are some important points regarding this generalization:
\begin{itemize}
    \item The operations may create paths that do not come directly from the constructors. For example, in $\sphere^1$, we also obtain nontrivial paths like $\fun{loop}\ct\fun{loop}$ and $\fun{loop}^{-1}$.
    
    \item To make two paths equal up to homotopy, the inductive definition must add a new $2$-dimensional constructor relating them. This ensures that the propositional equality between paths comes equipped with the necessary elimination and computation rules.
        
    \item Constructors without higher types in their inputs can generate higher paths. For instance, the $2$-dimensional sphere $\sphere^2$, generated by a point $\fun{base}\colon\sphere^2$ and a $2$-dimensional path $\fun{surf}\colon\fun{refl}_{\fun{base}}\eq{}\fun{refl}_{\fun{base}}$, contains a nontrivial $3$-dimensional path corresponding to the Hopf fibration.
\end{itemize}

\section{Elimination Rules}

To define operations on the circle $\sphere^1$, we must specify their behavior on the fibers above each constructor. We begin by defining the notion of a dependent path, which formalizes what it means for a path to lie over the loop constructor.

\begin{defn}\label{defn:eqt}
    Let $A$ be a type and $P\colon A\to\univ$ a type family. A dependent path from $u\colon P(x)$ to $v\colon P(y)$ lying over a path $p\colon x\eq{A}y$ is defined as
    \[
        u\eqt pPv\defeq\fun{transport}^P(p,u)\eq{P(y)}v.
    \]
\end{defn}

\begin{prop}\label{prop:eqt}
    Let\/ $P\colon A\to\univ$ be a type family. The following properties of dependent paths hold.

    \begin{enumerate}[a),font=\upshape]
        \item \emph{Reduction:} For any\/ $x\colon A$ and\/ $u,w\colon P(x)$,
        \[
            (u\eqt{\fun{refl}_x}{P}w)
            \equiv
            (u\eq{P(x)}w).
        \]

        \item \emph{Reflexivity:} For any\/ $x\colon A$ and\/ $u\colon P(x)$,
        \[
            \fun{refl}_u\colon u\eqt{\fun{refl}_x}{P}u.
        \]

        \item \emph{Symmetry:} For any\/ $p\colon x\eq{A}y$,\/ $u\colon P(x)$,\/ $v\colon P(y)$, and\/
        \[
            \ell\colon u\eqt{p}{P}v,
        \]
        there is a dependent path
        \[
            \ell^{-1}\colon v\eqt{p^{-1}}{P}u.
        \]

        \item \emph{Transitivity:} For any\/ $p\colon x\eq{A}y$ and\/ $q\colon y\eq{A}z$,\/ if
        \[
            \ell_1\colon u\eqt{p}{P}v
            \qquad\text{and}\qquad
            \ell_2\colon v\eqt{q}{P}w,
        \]
        then
        \[
            \ell_1\ct\ell_2\colon u\eqt{p\ct q}{P}w.
        \]

        \item \emph{Fiberwise concatenation:} For any\/ $p\colon x\eq{A}y$,\/ $u\colon P(x)$,\/ $v,w\colon P(y)$,\/ if
        \[
            \ell\colon u\eqt{p}{P}v
            \qquad\text{and}\qquad
            r\colon v\eq{P(y)}w,
        \]
        then
        \[
            \ell\ct r\colon u\eqt{p}{P}w.
        \]
    \end{enumerate}
\end{prop}

\begin{proof}\ 
    \begin{enumerate}[a),font=\upshape]
        \item \emph{Reduction:}
        By definition,
        \[
            (u\eqt{\fun{refl}_x}{P}w)
            \equiv
            \Bigl(
                \fun{transport}^P(\fun{refl}_x,u)
                \eq{P(x)}w
            \Bigr).
        \]
        Since
        \[
            \fun{transport}^P(\fun{refl}_x,u)\equiv u,
        \]
        the desired definitional equality follows.

        \item \emph{Reflexivity:}
        By (a), the type
        \[
            u\eqt{\fun{refl}_x}{P}u
        \]
        is definitionally equal to\/ $u\eq{P(x)}u$. Hence ordinary reflexivity gives
        \[
            \fun{refl}_u\colon u\eqt{\fun{refl}_x}{P}u.
        \]

        \item \emph{Symmetry:}
        By path induction on\/ $p$, we may assume that
        \[
            y\equiv x
            \qquad\text{and}\qquad
            p\equiv\fun{refl}_x.
        \]
        Thus\/ $v\colon P(x)$, and\/
        \[
            \ell\colon u\eqt{\fun{refl}_x}{P}v.
        \]
        By (a), this type is definitionally equal to\/ $u\eq{P(x)}v$. Moreover,
        \[
            p^{-1}\equiv\fun{refl}_x^{-1}\equiv\fun{refl}_x.
        \]
        Hence the required type
        \[
            v\eqt{p^{-1}}{P}u
        \]
        is definitionally equal to\/ $v\eq{P(x)}u$, which is inhabited by the ordinary inverse\/ $\ell^{-1}$.

        \item \emph{Transitivity:}
        By path induction on\/ $p$ and then on\/ $q$, we may assume that
        \[
            y\equiv x,\qquad z\equiv x,\qquad
            p\equiv\fun{refl}_x,\qquad
            q\equiv\fun{refl}_x.
        \]
        Consequently,
        \[
            \ell_1\colon u\eq{P(x)}v
            \qquad\text{and}\qquad
            \ell_2\colon v\eq{P(x)}w
        \]
        by (a). Also,
        \[
            p\ct q\equiv\fun{refl}_x.
        \]
        Therefore the target
        \[
            u\eqt{p\ct q}{P}w
        \]
        reduces, by (a), to\/ $u\eq{P(x)}w$, which is inhabited by the ordinary concatenation\/ $\ell_1\ct\ell_2$.

        \item \emph{Fiberwise concatenation:}
        By path induction on\/ $p$, we may assume that
        \[
            y\equiv x
            \qquad\text{and}\qquad
            p\equiv\fun{refl}_x.
        \]
        In this case, both\/ $v$ and\/ $w$ are elements of\/ $P(x)$, and by (a),
        \[
            \ell\colon u\eq{P(x)}v.
        \]
        Thus\/ the ordinary concatenation\/ $\ell\ct r$ has type\/ $u\eq{P(x)}w$. By (a), this is definitionally equal to
        \[
            u\eqt{\fun{refl}_x}{P}w,
        \]
        which is the required target.
    \end{enumerate}
\end{proof}

With Definition~\ref{defn:eqt}, the fundamental elimination rule for the circle is given by its induction principle:

\begin{description}[font=\normalfont\scshape]
    \item[Induction:] 
    Given a type family $P\colon\sphere^1\to\univ$, along with the following data:
    \begin{itemize}
        \item $b\colon P(\fun{base})$.
        \item $\ell\colon b\eqt{\fun{loop}}Pb$,
    \end{itemize}
    there exists a dependent function
    \[
        \fun{ind}_{\sphere^1}(P)\colon
            \prod_{x\colon\sphere^1}P(x)
    \]satisfying the computation rules:\footnote{See Lemma \ref{lem:dependent-map}}
    \[
        \fun{ind}_{\sphere^1}(P,\fun{base})\equiv b
        \quad\text{and}\quad
        \fun{apd}_{\fun{ind}_{\sphere^1}(P)}(\fun{loop})\coloneqq\ell.
    \]
\end{description}

\begin{rem}
    In the computation rule above, the evaluation at the point constructor is a strict judgmental equality. In contrast, the evaluation at the path constructor is a propositional equality. This \textsl{propositional equality by definition} is denoted using the hybridized symbol `$\coloneqq$'.
\end{rem}

The recursion principle is the non-dependent version of the induction principle when the type family is constant:

\begin{description}[font=\normalfont\scshape]
    \item[Recursion:] 
    Given a type $B$ equipped with a point $b\colon B$ and a path $\ell\colon b\eq Bb$, there exists a function $\fun{rec}_{\sphere^1}(B,b,\ell)\colon\sphere^1\to B$ satisfying the computation rules:
    \[
        \fun{rec}_{\sphere^1}(B,b,\ell)(\fun{base})\equiv b
        \quad\text{and}\quad
        \fun{ap}_{\fun{rec}_{\sphere^1}(B,b,\ell)}(\fun{loop})\coloneqq\ell.
    \] 
\end{description}

The following lemma formalizes the derivation of the recursion principle directly from the induction principle.

\begin{lem}\label{lem:S^1-rec-from-ind}
    If\/ $B$ is a type together with\/ $b\colon B$ and\/ $p\colon b\eq Bb$, then there is a function\/ $f\colon\sphere^1\to B$ with
    \begin{enumerate}[a),font=\upshape]
        \item $f(\fun{base})\equiv b$, and
        \item $\fun{ap}_f(\fun{loop})\coloneqq p$.
    \end{enumerate}
\end{lem}

\begin{proof}
    By Lemma~\ref{lem:constant-fibration}, we have
    \begin{align*}
        \fun{transportconst}^B(\fun{loop},b)
            &\colon
                \fun{transport}^{x\mapsto B}(\fun{loop},b)
                    \eq Bb\\
            &\equiv b\eqt{\fun{loop}}{x\mapsto B}b.
    \end{align*}
    Let
    \[
        \ell\defeq\fun{transportconst}^B(\fun{loop},b).
    \]
    From Proposition~\ref{prop:eqt}~e) we deduce
    \[
        \ell\ct p\colon
            b\eqt{\fun{loop}}{x\mapsto B}b.
    \]
    Applying the induction principle of $\sphere^1$ to the constant type family $x\mapsto B$, we obtain a function
    \[
        f\colon\sphere^1\to B
    \]
    such that
    \[
        f(\fun{base})\equiv b
        \quad\text{and}\quad
        \fun{apd}_f(\fun{loop})\coloneqq\ell\ct p.
    \]
    Furthermore,
    \begin{align*}
        \fun{apd}_f(\fun{loop})
            &\eq{}
                \fun{transportconst}^B(
                        \fun{loop},
                        f(\fun{base})
                    )
                \ct\fun{ap}_f(\fun{loop})
                &&\text{; Lem~\ref{lem:apd-constant-fibration}}\\
            &\eq{}
                \fun{transportconst}^B(\fun{loop},b)
                \ct\fun{ap}_f(\fun{loop})
                &&\text{; }f(\fun{base})\equiv b\\
            &\equiv
                \ell\ct\fun{ap}_f(\fun{loop}).
    \end{align*}
    Hence,
    \[
        \ell\ct\fun{ap}_f(\fun{loop})\eq{}\ell\ct p.
    \]
    The conclusion follows by left cancellation of $\ell$.
    %
    
\end{proof}

\begin{rem}\label{rem:eqt-h=k}
    Suppose that $h,k\colon A\to B$ are two functions. Then, if $P\colon A\to\univ$ is defined as $P(x)\defeq h(x)\eq{B}k(x)$ for $x\colon A$, Lemmas~\ref{lem:transport-in-identity-types} and~\ref{lem:path-equivalences}~b)~ii) (or Exercise~\ref{exr:path-equivalence}) imply
    \[
        (u\eqt\gamma Pv)\simeq
            \bigl(
                \fun{ap}_h(\gamma)^{-1}
                \ct u
                \ct\fun{ap}_k(\gamma)
                \eq{B}v
            \bigr).
    \]
\end{rem}

\begin{lem}\label{lem:S^1-f=g}
    If\/ $A$ is a type and\/ $f, g \colon \sphere^1 \to A$ are two maps together with two equalities\/ $p, q$:
    \begin{align*}
        p&\colon f(\fun{base})\eq Ag(\fun{base})\\
        q&\colon\fun{ap}_f(\fun{loop})
            \eqt{p}{x\mapsto x\eq Ax}
            \fun{ap}_g(\fun{loop}),
    \end{align*}
    then for all\/ $z\colon\sphere^1$ we have\/ $f(z)\eq Ag(z)$.
\end{lem}

\begin{proof}
    Let $P\colon\sphere^1\to\univ$ be the type family defined by $P(z)\defeq f(z)\eq Ag(z)$. Then, $p\colon P(\fun{base})$. On the other hand, by Remark~\ref{rem:eqt-h=k}, we have
    \begin{equation}\label{eq:p=p}
        (p\eqt{\fun{loop}}Pp)
            \eq{}
                \bigl(
                    \fun{ap}_f(\fun{loop})^{-1}
                        \ct p
                        \ct\fun{ap}_g(\fun{loop})
                        \eq{}p
                \big)
    \end{equation}
    By Proposition~\ref{prop:transport-decomposition-for-paths}, we have
    \[
        \fun{transport}^{x\mapsto(x\eq Ax)}(p,\fun{ap}_f(\fun{loop}))
            \eq{}p^{-1}\ct \fun{ap}_f(\fun{loop})\ct p.
    \]
    By definition, $q$ witnesses the equality
    \[
        q\colon\fun{transport}^{x\mapsto(x\eq{A}x)}(p,\fun{ap}_f(\fun{loop}))\eq{}\fun{ap}_g(\fun{loop}),
    \]
    producing an inhabitant of
    \[
        \fun{ap}_g(\fun{loop})
            \eq{}p^{-1}\ct \fun{ap}_f(\fun{loop})\ct p.
    \]
    Hence,
    \[
        p\ct \fun{ap}_g(\fun{loop})
            \eq{}\fun{ap}_f(\fun{loop})\ct p.
    \]
    Left-concatenating with $\fun{ap}_f(\fun{loop})^{-1}$ gives a proof of
    \[
        \fun{ap}_f(\fun{loop})^{-1}
        \ct p
        \ct \fun{ap}_g(\fun{loop})
        \eq{}p.
    \]
    Plugging this proof into \eqref{eq:p=p}, we obtain an element of $p\eqt{\fun{loop}}{P}p$. Then, the induction principle of the circle yields a section of $P$, proving that $f(z)\eq{A}g(z)$ for all $z\colon\sphere^1$.
    %
    
\end{proof}

\begin{thm}
    For any type\/ $A$, we have a natural equivalence
    \[
        (\sphere^1\to A)\simeq\sum_{x\colon A}x\eq Ax.
    \]
\end{thm}

\begin{proof}
    We have two canonical maps
    \begin{align*}
        f\colon(\sphere^1\to A)
            &\to\sum_{x\colon A}x\eq Ax\\
        \rho
            &\mapsto
                \dep{
                    \rho(\fun{base}),
                    \fun{ap}_\rho(\fun{loop})
                }\\[4mm]
        g\colon\Bigl(\sum_{x\colon A}x\eq Ax\Bigr)
            &\to(\sphere^1\to A)\\
        \dep{a,\ell}
            &\mapsto\fun{rec}_{\sphere^1}(A,a,\ell).
    \end{align*}
    To prove that they define an equivalence, we evaluate both compositions:
    \begin{align*}
        f\circ g(\dep{a,\ell})
            &\equiv
                f\bigl(
                    \fun{rec}_{\sphere^1}(A,a,\ell)
                \bigr)\\
            &\equiv
                \bigdep{
                    \fun{rec}_{\sphere^1}(A,a,\ell)
                        (\fun{base}),
                    \fun{ap}_{
                        \fun{rec}_{\sphere^1}(A,a,\ell)
                        }(\fun{loop})
                }\\
            &\eq{}\dep{a,\ell},
    \intertext{by the computation rules of $\fun{rec}_{\sphere^1}$.}
        g\circ f(\rho)
            &\equiv g(
                \dep{
                    \rho(\fun{base}),
                    \fun{ap}_\rho(\fun{loop})
                }
            )\\
            &\equiv\fun{rec}_{\sphere^1}(
                A,
                \rho(\fun{base}),
                \fun{ap}_\rho(\fun{loop})
                )\\
            &\sim\rho,
    \end{align*}
    by Lemma~\ref{lem:S^1-f=g}. The equality follows from function extensionality.

    Regarding naturality, suppose that $\phi\colon A\to B$. Then, $\phi$ induces a commutative diagram because, by Proposition~\ref{prop:ap-properties}, $\fun{ap}_\phi\circ\fun{ap}_\rho\eq{}\fun{ap}_{\phi\circ\rho}$:
    \[
        \begin{tikzcd}
            \sphere^1\to A
                    \arrow[r]
                    \arrow[d]
                &\sum_{x\colon A}x\eq Ax
                    \arrow[d]
                &\rho
                    \arrow[r,mapsto]
                    \arrow[d,mapsto]
                &\dep{
                    \rho(\fun{base}),
                    \fun{ap}_\rho(\fun{loop})
                    }
                    \arrow[d,mapsto]\\
            \sphere^1\to B
                    \arrow[r]
                &\sum_{y\colon B}y\eq By
                &\phi\circ\rho
                    \arrow[r,mapsto]
                &\dep{
                    \phi(\rho(\fun{base})),
                    \fun{ap}_{\phi}(
                        \fun{ap}_\rho(\fun{loop})
                        )
                    }
        \end{tikzcd}
    \]
\end{proof}

\section{The Interval}

The type \textsl{interval}, denoted by $\op I$, is defined inductively as
\begin{itemize}
    \item $0_{\op I}\colon\op I$,
    \item $1_{\op I}\colon\op I$,
    \item $\fun{seg}\colon 0_{\op I}\eq{\op I}1_{\op I}$.
\end{itemize}

The recursion principle for the interval says that given a type $B$ along with
\begin{itemize}
    \item a point $b_0\colon B$,
    \item a point $b_1\colon B$, and
    \item a path $s\colon b_0\eq Bb_1$,
\end{itemize}
there is a function
\[
    \fun{rec}_{\op I}(B)\colon\op I\to B
\]
such that
\[
    f(0_{\op I})\equiv b_0,\quad
    f(1_{\op I})\equiv b_1,\quad\text{and}\quad
    \fun{ap}_f(\fun{seg})\coloneqq s.
\]
The induction principle says that given $P\colon\op I\to\univ$ along with
\begin{itemize}
    \item a point $b_0\colon P(0_{\op I})$,
    \item a point $b_1\colon P(1_{\op I})$, and
    \item a path $s\colon b_0\eqt{\fun{seg}}{P} b_1$,
\end{itemize}
there is a function
\[
    \fun{ind}_{\op I}(P)\colon\prod_{x\colon\op I}P(x)
\]
such that
\[
    \fun{ind}_{\op I}(P)(0_{\op I})\equiv b_0,\quad
    \fun{ind}_{\op I}(P)(1_{\op I})\equiv b_1,\quad\text{and}\quad
    \fun{apd}_{\fun{ind}_{\op I}(P)}(\fun{seg})\coloneqq s.
\]

\begin{prop}
    The type\/ $\op I$ is contractible.
\end{prop}

\begin{proof}
    We choose $1_{\op I}$ as the center of contraction. We must find a contraction map $f\colon\prod_{x\colon\op I}x\eq{\op I}1_{\op I}$. To define this by induction on $x$ we must provide:
    \begin{itemize}
        \item a point $b_0\colon 0_{\op I}\eq{\op I} 1_{\op I}$,
        
        \item a point $b_1\colon 1_{\op I}\eq{\op I}1_{\op I}$, and
        
        \item a dependent path $s\colon\fun{seg}\eqt{\fun{seg}}{x\mapsto x\eq{\op I}1_{\op I}}\fun{refl}_{1_{\op I}}$.
    \end{itemize}
    We can choose $b_0\defeq\fun{seg}$ and $b_1\defeq\fun{refl}_{1_{\op I}}$. Therefore, by Proposition~\ref{prop:transport-decomposition-for-paths}, the type of $s$ is equivalent to
    \[
        \fun{seg}^{-1}\ct\fun{seg}
        \eq{1_{\op I}\eq{\op I}1_{\op I}}
        \fun{refl}_{1_{\op I}},
    \]
    which is trivially inhabited after path cancellation. Therefore, the required dependent path $s$ exists, completing the definition of $f$.
\end{proof}

\begin{thm} {\upshape[Function Extensionality]}
    The recursion principle for the interval implies function extensionality.
\end{thm}

\begin{proof}
    Let $f,g\colon A\to B$ be two functions such that $f(x)\eq Bg(x)$ for every $x\colon A$. We have to prove that $f\eq{A\to B}g$.
    
    Let $\eta\colon\prod_{x\colon A}f(x)\eq Bg(x)$ be the given pointwise equality. For each $x\colon A$, we apply the recursion principle of the interval to the type $B$, the points $f(x)$ and $g(x)$, and the path $\eta(x)$. This yields a function $h_x\colon\op I\to B$ such that $h_x(0_{\op I})\equiv f(x)$, $h_x(1_{\op I})\equiv g(x)$, and $\fun{ap}_{h_x}(\fun{seg})\eq{f(x)\eq{B} g(x)}\eta(x)$.
    
    Let $\epsilon\colon\op I\to(A\to B)$ be defined by
    \[
        \epsilon(i)\defeq\lambda x.\,h_x(i).
    \]
    Evaluating $\epsilon$ at the endpoints of $\op I$, we obtain
    \[
        \epsilon(0_{\op I})
            \equiv(\lambda x.\,h_x(0_{\op I}))
            \equiv(\lambda x.\,f(x))
            \equiv f
    \]
    and
    \[
        \epsilon(1_{\op I})
            \equiv(\lambda x.\,h_x(1_{\op I}))
            \equiv(\lambda x.\,g(x))
            \equiv g.
    \]
    Consequently,
    \[
        \fun{ap}_\epsilon(\fun{seg})\colon
            (
                \epsilon(0_{\op I})
                \eq{A\to B}
                \epsilon(1_{\op I})
            )
            \equiv(f\eq{A\to B}g).
    \]
\end{proof}

\section{Circles \& Spheres}

\begin{lem}
    The circle\/ $\sphere^1$ satisfies $\fun{loop}\ne\fun{refl}_{base}$. In particular, is not equivalent to the unit type\/~$\fun1$.
\end{lem}

\begin{proof}
    We must construct a map
    \[
        \fun{loop}
            \eq{\fun{base}\eq{\sphere^1}\fun{base}}
            \fun{refl}_{\fun{base}}
        \to\fun0.
    \]
    Let $\zeta\colon\fun{loop}\eq{\fun{base}\eq{\sphere^1}\fun{base}}\fun{refl}_{\fun{base}}$. For any $A\colon\univ$, any $a\colon A$, and any $\ell\colon a\eq Aa$, consider the function
    \[
        f_a\defeq
            \fun{rec}_{\sphere^1}(A,a,\ell)
            \colon\sphere^1\to A.
    \]
    By definition, $f_a(\fun{base})\equiv a$ and $\fun{ap}_{f_a}(\fun{loop})\eq{a\eq Aa}\ell$. It follows that
    \[
        \fun{ap}_{\fun{ap}_{f_a}}(\zeta)\colon
            \fun{ap}_{f_a}(\fun{loop})
            \eq{a\eq Aa}
            \fun{ap}_{f_a}(\fun{refl}_{\fun{base}}).
    \]
    Since $\fun{ap}_{f_a}(\fun{refl}_{\fun{base}})\equiv\fun{refl}_{f_a(\fun{base})}\equiv\fun{refl}_a$, we obtain a witness of $\ell\eq{a\eq Aa}\fun{refl}_a$.

    As a consequence, given two paths $p,q\colon a\eq Ax$, because
    \[
        p\ct q^{-1}\colon a\eq Aa,
    \]
    we deduce that $p\ct q^{-1}\eq{a\eq Aa}\fun{refl}_a$, which implies $p\eq{a\eq Ax}q$.

    Thus, $A$ would be a Set. Since the same reasoning can be applied to a higher universe $\univ'$, we would deduce that $\univ\colon\univ'$ is a Set. However, Example~\ref{xmpl:univ-isnt-a-set} yields a contradiction.
\end{proof}

\begin{rem}
    The application of the recursion principle for the circle to a target type $A\colon\univ'$ is justified by \nref{lpar:universe-polymorphism}. Although the elimination rule is commonly stated for a target type in $\univ$, it permits the standard recursor $\fun{rec}_{\sphere^1}$ to eliminate into types residing in any universe above $\univ$, without the need of a separate definition of the circle at each universe level.
\end{rem}

\begin{cor}
    The type\/ $\sphere^1\eq{\univ}\sphere^1$ is not a set.
\end{cor}

\begin{proof}
    By univalence, it suffices to prove that $\sphere^1\simeq\sphere^1$ is not a set. By Definition~\ref{defn:equivalence},
    \[
        \sphere^1\simeq\sphere^1\equiv
            \sum_{f\colon\sphere^1\to\sphere^1}
                \fun{isequiv}(f).
    \]
    Because
    \[
        \dep{
            \id_{\sphere^1},
            \fun{isequiv}(\id_{\sphere^1})
        }
            \colon\sum_{f\colon\sphere^1\to\sphere^1}
                \fun{isequiv}(f),
    \]
    it suffices to show that $\id_{\sphere^1}\eq{\sphere^1\to\sphere^1}\id_{\sphere^1}$ is not a mere proposition.

    By function extensionality, this identity type is equivalent to $\id_{\sphere^1}\sim\id_{\sphere^1}$. Since $\lambda x.\,\fun{refl}_x$ is a homotopy from $\id_{\sphere^1}$ to itself, it suffices to construct another homotopy of this type distinct from this one.

    Let $P\colon\sphere^1\to\univ$ be the type family defined by $P(x)\defeq x\eq{\sphere^1}x$. We define a dependent function
    \[
        \eta\colon\prod_{x\colon\sphere^1}x\eq{\sphere^1}x
    \]
    by circle induction. We choose $\fun{loop}\colon P(\fun{base})$ as the base point. To complete the definition, we need a dependent path $\ell\colon\fun{loop}\eqt{\fun{loop}}P\fun{loop}$. By Proposition~\ref{prop:transport-decomposition-for-paths},
    \[
        \fun{transport}^P(\fun{loop},\fun{loop})
        \eq{} \fun{loop}^{-1}\ct\fun{loop}\ct\fun{loop}
        \eq{} \fun{loop}.
    \]
    Let $\ell$ denote the witness of this equality. We can then define
    \[
        \eta\defeq\fun{ind}_{\sphere^1}
            (
                P,
                \fun{loop},
                \ell
            ).
    \]
    By the computation rule, $\eta(\fun{base})\equiv\fun{loop}$. Since $\fun{loop}\ne\fun{refl}_{\fun{base}}$, we conclude that $\eta\ne\lambda x.\,\fun{refl}_x$, as required.
\end{proof}

\begin{lem}
    Given\/ $f\colon A\to B$,\/ $x,y\colon A$,\/ $p,q\colon x\eq{A}y$, and\/ $\gamma\colon p\eq{x\eq{A}y}q$, we have a path
    \[
        \fun{ap}^2_f(\gamma)\colon\fun{ap}_f(p)\eq{f(x)\eq{B}f(y)}\fun{ap}_f(q).
    \]
\end{lem}

\begin{proof}
    By induction on $\gamma$ it suffices to assume $q\equiv p$ and $\gamma\equiv\fun{refl}_p$. Thus, our goal reduces to $\fun{ap}_f(p)\eq{f(x)\eq Bf(y)}\fun{ap}_f(p)$, which is trivially inhabited by $\fun{ap}^2_f(\fun{refl}_p)\defeq\fun{refl}_{\fun{ap}_f(p)}$.
    
\end{proof}

\begin{lem}
    Given\/ $P\colon A\to\univ$,\/ $x,y\colon A$,\/ $p,q\colon x\eq{A}y$, and\/ $\gamma\colon p\eq{x\eq{A}y}q$, for any\/ $u\colon P(x)$ we have a path
    \[
        \fun{transport}^{2P}(\gamma,u)\colon
        \fun{transport}^P(p,u)
        \eq{P(y)}
        \fun{transport}^P(q,u).
    \]
\end{lem}

\begin{proof}
    By induction on $\gamma$ we obtain our goal and the definitional identity
    \[
        \fun{transport}^{2P}(\fun{refl}_p,u)
        \equiv
        \fun{refl}_{\fun{transport}^P(p,u)}.
    \]
\end{proof}

\begin{defn}
    Let $P\colon A\to\univ$ be a type family, $x,y\colon A$, $p,q\colon x\eq Ay$, and $\gamma\colon p\eq{x\eq Ay}q$. Given points $u\colon P(x)$ and $v\colon P(y)$, along with dependent paths $\rho\colon u\eqt pPv$ and $\sigma\colon u\eqt qPv$, we define the dependent $2$-path over $\gamma$ as
    \[
        \rho\eqt\gamma P\sigma
            \defeq(
                \rho\eq{}\fun{transport}^{2P}(\gamma,u)
                \ct
                \sigma
                ).
    \]
\end{defn}

\begin{rem}
    The elements of the definition above can be illustrated as follows:

    \[

    \]

    \[

\begin{lem}
    Given\/ $P\colon A\to\univ$,\/ $x,y\colon A$,\/ $p,q\colon x\eq{A}y$,\/ $\gamma\colon p\eq{x\eq{A}y}q$, and a function\/ $f\colon\prod_{x\colon A}P(x)$, we have a dependent\/ $2$-path
    \[
        \fun{apd}^2_f(\gamma)\colon
            \fun{apd}_f(p)\eqt\gamma P\fun{apd}_f(q).
    \]
\end{lem}

\begin{proof}
    By path induction on the $2$-path $\gamma$, we only need to address the case where $q\equiv p$ and $\gamma\equiv\fun{refl}_p$. In this case, the target type becomes
    \[
        \fun{apd}_f(p)\eqt{\fun{refl}_p}P\fun{apd}_f(p),
    \]
    which is definitionally equal to
    \[
        \fun{apd}_f(p)\eq{}\fun{apd}_f(p).
    \]
    This identity type is trivially inhabited by $\fun{refl}_{\fun{apd}_f(p)}$.
\end{proof}

\begin{defn}
    The $2$-sphere $\sphere^2$ is the higher inductive type generated by
    \begin{itemize}
        \item $\fun{base}\colon\sphere^2$ and
        \item $\fun{surf}\colon\fun{refl}_{\fun{base}}\eq{\fun{base}\eq{\sphere^2}\fun{base}}\fun{refl}_{\fun{base}}$.
    \end{itemize}
\end{defn}

\begin{rem}
    Think of a $1$-path as a string from one point to another, and a $2$-path as the surface swept out as one string continuously deforms into another. Because $\fun{refl}_{\fun{base}}$ is a constant path of zero length that remains stationary at the base point, a $2$-path from $\fun{refl}_{\fun{base}}$ to itself is a surface whose entire outer boundary is collapsed into that single point.
\end{rem}

The recursion principle for $\sphere^2$ says that given $B$ with
\begin{itemize}
    \item $b\colon B$ and
    \item $s\colon\fun{refl}_b\eq{b\eq{B}b}\fun{refl}_b$,
\end{itemize}
we have
\[
    \fun{rec}_{\sphere^2}(B,b,s)\colon\sphere^2\to B
\]
with
\[
    \fun{rec}_{\sphere^2}(B,b,s)(\fun{base})\equiv b
    \quad\text{and}\quad
    \fun{ap}^2_{\fun{rec}_{\sphere^2}(P,b,s)}(\fun{surf})\eq{}s.
\]
The induction principle for $\sphere^2$ says that, given $P\colon\sphere^2\to\univ$ with
\begin{itemize}
    \item $b\colon P(\fun{base})$ and 
    \item $s\colon\fun{refl}_b\eqt{\fun{surf}}Q\fun{refl}_b$,
\end{itemize}
where $Q\defeq\lambda p.\,b\eqt pPb$, there is a function 
\[
    \fun{ind}_{\sphere^2}(P,b,s)\colon
        \prod_{x\colon\sphere^2}P(x)
\]
such that
\[
    \fun{ind}_{\sphere^2}(P,b,s)(\fun{base})\equiv b
    \quad\text{and}\quad
    \fun{apd}^2_{\fun{ind}_{\sphere^2}(P,b,s)}(\fun{surf})\eq{}s.
\]

\section{Suspensions}

\begin{defn}
    Given a type $A$, the \textsl{suspension} of $A$ is the type inductively defined by
    \begin{itemize}
        \item $\fun N\colon\fun{Susp}(A)$,
        \item $\fun S\colon\fun{Susp}(A)$, and
        \item $\fun{merid}\colon A\to(\fun N\eq{\fun{Susp}(A)}\fun S)$.
    \end{itemize}
\end{defn}

The suspension $\fun{Susp}(A)$ represents the universal method for elevating the points of a type $A$ into paths. Consequently, the $1$-paths of $A$ are promoted to $2$-paths, and so forth across higher dimensions. Geometrically, this construction evokes a globe, featuring a north pole and a south pole connected by a family of meridians parameterized by the elements of $A$.
\[
\begin{tikzpicture}[scale=2,font=\small]

    % Sphere
    \draw (0,0) circle (1);
    
    % Hidden halves of the meridians
    \draw[-latex,thick,name path=merid] (0,1) arc (90:270:0.35 and 1)
        node[pos=.52,right=-1.5pt] {$\fun{merid}(a)$};
    
    \draw[-latex,thick] (0,1) arc (90:270:0.65 and 1);
    \draw[-latex,thick] (0,1) arc (90:270:0.90 and 1);
    
    % Visible halves of the meridians
    \draw[-latex,dashed] (0,1) arc (90:-90:0.35 and 1);
    \draw[-latex,dashed] (0,1) arc (90:-90:0.65 and 1);
    \draw[-latex,dashed] (0,1) arc (90:-90:0.90 and 1);
    
    % Visible half of the equator
    \draw[name path=equator,green!40!black,thick] (-1,0) arc (180:360:1 and 0.25);
    
    % Hidden half of the equator
    \draw[dashed] (-1,0) arc (180:0:1 and 0.25);
    
    % Poles
    \fill (0,1) circle (1.5pt) node[above=3pt] {$\fun N$};

\]
The recursion principle for $\fun{Susp}(A)$ says that given a type $B$ with
\begin{itemize}
    \item $n\colon B$,
    \item $s\colon B$, and
    \item $m\colon A\to(n\eq Bs)$,
\end{itemize}
there is a function
\[
    \fun{rec}_{\fun{Susp}(A)}(B,n,s,m)\colon
        \fun{Susp}(A)\to B
\]
such that
\begin{align*}
    \fun{rec}_{\fun{Susp}(A)}(B,n,s,m)(\fun N)&\equiv n,\\
    \fun{rec}_{\fun{Susp}(A)}(B,n,s,m)(\fun S)&\equiv s,
        \text{ and}\\
    \fun{ap}_{\fun{rec}_{\fun{Susp}(A)}(B,n,s,m)}
        (\fun{merid}(a))
    &\eq{}m(a).
\end{align*}

The induction principle says that given $P\colon\fun{Susp}(A)\to\univ$ together with
\begin{itemize}
    \item $n\colon P(\fun{N})$,
    \item $s\colon P(\fun{S})$, and
    \item for each $a\colon A$, a dependent path $m(a)\colon n\eqt{\fun{merid}(a)}Ps$,
\end{itemize}
there exists a function
\[
    \fun{ind}_{\fun{Susp}(A)}(P,n,s,m)\colon
        \prod_{x\colon\fun{Susp}(A)}P(x)
\]
such that
\[
    \fun{ind}_{\fun{Susp}(A)}(P,n,s,m)(\fun{N})\equiv n
    \quad\text{and}\quad
    \fun{ind}_{\fun{Susp}(A)}(P,n,s,m)(\fun{S})\equiv s,
\]
and for each $a\colon A$ we have
\[
    \fun{apd}_{\fun{ind}_{\fun{Susp}(A)}(P,n,s,m)}
        (\fun{merid}(a))\eq{}m(a).
\]

\begin{lem}\label{lem:Susp-rec-equality}
    Let\/ $A,B\colon\univ$ and\/ $f,g\colon\fun{Susp}(A)\to B$. Suppose we have
    \begin{itemize}
        \item $p\colon f(\fun N)\eq{B}g(\fun N)$,
        \item $q\colon f(\fun S)\eq{B}g(\fun S)$,
        \item $m\colon\prod_{a\colon A}\fun{ap}_f(\fun{merid}(a))^{-1}\ct p\ct\fun{ap}_g(\fun{merid}(a))\eq{}q$.
    \end{itemize}
    Then\/ $f\eq{}g$.
\end{lem}

\begin{proof}
    By function extensionality, it suffices to construct a homotopy
    \[
        \eta\colon\prod_{x\colon\fun{Susp}(A)}f(x)\eq{B}g(x).
    \]
    We define $\eta$ using the induction principle of $\fun{Susp}(A)$ on the type family
    \begin{align*}
        P&\colon\fun{Susp}(A)\to\univ\\
        P(x)&\defeq f(x)\eq{B}g(x).
    \end{align*}
    For the point constructors, we simply provide the given paths $p\colon P(\fun N)$ and $q\colon P(\fun S)$.

    For the path constructors, we must provide, for each $a\colon A$, a dependent path $\mu(a)\colon p\eqt{\fun{merid}(a)}Pq$. By Remark~\ref{rem:eqt-h=k}), there is an equivalence
    \[
        (p\eqt{\fun{merid}(a)}Pq)
        \simeq
        \Bigl(
            \fun{ap}_f(\fun{merid}(a))^{-1}\ct p\ct\fun{ap}_g(\fun{merid}(a))\eq{}q
        \Bigr).
    \]
    Since the right-hand side is the type of $m(a)$, we can define $\mu(a)$ as the path corresponding to $m(a)$ under this equivalence.

    Now the induction principle yields the required homotopy
    \[
        \eta\defeq\fun{ind}_{\fun{Susp}(A)}(P,p,q,\mu).
    \]
\end{proof}

\begin{prop}\label{prop:unpointed-susp-maps}
    Given types $A,B\colon\univ$, the function
    \begin{align*}
        \phi\colon\bigl(\fun{Susp}(A)\to B\bigr)
            &\to\sum_{n,s\colon B}A\to n\eq{B}s\\
        f&\mapsto\dep{
            (f(\fun N),f(\fun S)),
            \fun{ap}_f\circ\fun{merid}
        }
    \end{align*}
    is an equivalence.
\end{prop}

\begin{proof}
    For the backward map, define
    \begin{align*}
        \psi\colon
            \Bigl(
                \sum_{n,s\colon B}A\to n\eq{B}s
            \Bigr)
            &\to
            \bigl(\fun{Susp}(A)\to B\bigr)\\
        \dep{(n,s),m}&\mapsto\fun{rec}_{\fun{Susp}(A)}
            (
                B,n,s,m
            ).
    \end{align*}
    To complete the proof, we have to compute both compositions.
    \begin{align*}
        \phi\circ\psi(\dep{(n,s),m})
            &\equiv\phi
                \bigl(
                    \fun{rec}_{\fun{Susp}(A)}(B,n,s,m)
                \bigr)\\
            &\equiv\dep{(n,s),\fun{ap}_\rho\circ\fun{merid}},
    \end{align*}
    where
    \[
        \rho\defeq\fun{rec}_{\fun{Susp}(A)}(B,n,s,m).
    \]
    Because the first component of the dependent pair is definitionally equal to $(n,s)$ and transport along reflexivity is the identity, the goal reduces, via function extensionality, to showing that $\fun{ap}_\rho(\fun{merid}(a))\eq{}m(a)$, which is precisely the third computation rule of the recursor.

    With regard to the other composition, we have
    \begin{align*}
        \psi\circ\phi(f)
            &\equiv\psi
                (
                    B,
                    f(\fun N),
                    f(\fun S),
                    \fun{ap}_f\circ\fun{merid}
                )\\
            &\equiv\fun{rec}_{\fun{Susp}(A)}
                \bigl(
                    B,
                    f(\fun N),
                    f(\fun S),
                    \fun{ap}_f\circ\fun{merid}
                \bigr).
    \end{align*}
    The conclusion follows from Lemma~\ref{lem:Susp-rec-equality}.
    %
    
\end{proof}

\begin{lem}\label{lem:Susp(S0)=S1}
    $\fun{Susp}(\fun2)\simeq\sphere^1$.
\end{lem}

\begin{proof}
    Define $m\defeq\fun{rec}_{\fun2}(\fun{base}\eq{\sphere^1}\fun{base},\fun{loop},\fun{refl}_{\fun{base}})$, that is,
    \begin{align*}
        m\colon\fun2
            &\to(\fun{base}\eq{\sphere^1}\fun{base})\\
        0_{\fun2}&\mapsto\fun{loop}\\
        1_{\fun2}&\mapsto\fun{refl}_{\fun{base}}.
    \end{align*}
    Then, recursion on $\fun{Susp}(\fun2)$ gives us the forward map
    \[
        f\defeq\fun{rec}_{\fun{Susp}(\fun2)}
            (\sphere^1,\fun{base},\fun{base},m).
    \]
    For the backward map, consider 
    \[
        g\defeq\fun{rec}_{\sphere^1}
            \bigl(
                \fun{Susp}(\fun2),
                \fun N,
                \fun{merid}(0_{\fun2})
                    \ct\fun{merid}(1_{\fun2})^{-1}
            \bigr),
    \]
    which is the function defined by recursion on $\sphere^1$ satisfying
    \begin{align*}
        g\colon\sphere^1&\to\fun{Susp}(\fun2)\\
        g(\fun{base})&\equiv\fun N\\
        \fun{ap}_g(\fun{loop})
            &\eq{}\fun{merid}(0_{\fun2})
            \ct\fun{merid}(1_{\fun2})^{-1}.
    \end{align*}
    To complete the proof, we must show that $f$ and $g$ are quasi-inverses. For the composition $f\circ g$, we have
    \begin{align*}
        f\circ g(\fun{base})
            &\equiv f(\fun N)\\
            &\equiv\fun{base}.\\[2mm]
        \fun{ap}_{f\circ g}(\fun{loop})
            &\eq{}
                \fun{ap}_f(\fun{merid}(0_{\fun2}))
                \ct
                \fun{ap}_f(\fun{merid}(1_{\fun2}))^{-1}\\
            &\eq{}m(0_{\fun2})\ct m(1_{\fun2})^{-1}\\
            &\eq{}\fun{loop}\ct\fun{refl}_{\fun{base}}\\
            &\eq{}\fun{loop}.
    \end{align*}
    Therefore, $f\circ g\sim\id_{\sphere^1}$ by Lemma~\ref{lem:S^1-f=g}.

    For the other composition,
    \begin{align*}
        g\circ f(\fun N)
            &\equiv g(\fun{base})\\
            &\equiv\fun N\\[2mm]
        g\circ f(\fun S)
            &\equiv g(\fun{base})\\
            &\equiv\fun N\\
            &\eq{}\fun S
                &&;\ \fun{merid}(1_{\fun2}).
    \end{align*}
    This means that $n\defeq\fun{refl}_{\fun N}$ and $s\defeq\fun{merid}(1_{\fun2})$ satisfy
    \begin{itemize}
        \item $n\colon g\circ f(\fun N)\eq{}\fun N$,
        \item $s\colon g\circ f(\fun S)\eq{}\fun S$.
    \end{itemize}
    Let $P\colon\fun{Susp}(\fun2)\to\univ$ be defined by $P(x)\defeq g\circ f(x)\eq{\fun{Susp}(\fun2)}x$. The conditions above translate into
    \begin{itemize}
        \item $n\colon P(\fun N)$,
        \item $s\colon P(\fun S)$.
    \end{itemize}
    Therefore, to construct a witness of $g\circ f\sim\id_{\fun{Susp}(\fun2)}$, it suffices to define, for each $b\colon\fun2$, a dependent path $\mathfrak m(b)\colon n\eqt{\fun{merid}(b)}Ps$, and then invoke the induction principle of $\fun{Susp}(\fun2)$.

    The induction principle of $\fun2$ allows us to reduce the construction to the cases $b\equiv0_{\fun2}$ and $b\equiv1_{\fun2}$. By Remark~\ref{rem:eqt-h=k}, the type of the dependent path is equivalent to an equality of paths in $\fun{Susp}(\fun2)$:
    \begin{align*}
        (n\eqt{\fun{merid}(0_{\fun2})}Ps)
            &\simeq
                \Bigl(
                    \fun{ap}_{g\circ f}(\fun{merid}
                        (0_{\fun2}))^{-1}
                    \ct n
                    \ct \fun{ap}_{\id_{\fun{Susp}(\fun2)}}
                        (\fun{merid}(0_{\fun2}))
                    \eq{}s
                \Bigr).
    \end{align*}
    To compute this, observe that
    \begin{align*}
        \fun{ap}_{g\circ f}(\fun{merid}(0_{\fun2}))
            &\eq{}\fun{ap}_g(\fun{ap}_f
                (\fun{merid}(0_{\fun2})))\\
            &\eq{}\fun{ap}_g(m(0_{\fun2}))\\
            &\eq{}\fun{ap}_g(\fun{loop})\\
            &\eq{}\fun{merid}(0_{\fun2})
                \ct\fun{merid}(1_{\fun2})^{-1}.
    \intertext{Hence,}
        \fun{ap}_{g\circ f}(\fun{merid}(0_{\fun2}))^{-1}
            &\eq{}\fun{merid}(1_{\fun2})
                \ct\fun{merid}(0_{\fun2})^{-1}.
    \end{align*}
    Therefore, using that $n\equiv\fun{refl}_{\fun N}$ and $\fun{ap}_{\id_{\fun{Susp}(\fun2)}}(p)\equiv p$, the goal reduces to
    \[
        (n\eqt{\fun{merid}(0_{\fun2})}Ps)
        \simeq
        \fun{merid}(1_{\fun2})\eq{}s,
    \]
    which is trivial since, by definition, $s\equiv\fun{merid}(1_{\fun2})$.
    
    Similarly, for the case $b\equiv1_{\fun2}$, the required dependent path type translates into
    \begin{align*}
        (n\eqt{\fun{merid}(1_{\fun2})}Ps)
            &\simeq
                \Bigl(
                    \fun{ap}_{g\circ f}(\fun{merid}
                        (1_{\fun2}))^{-1}
                    \ct n
                    \ct \fun{ap}_{\id_{\fun{Susp}(\fun2)}}
                        (\fun{merid}(1_{\fun2}))
                    \eq{}s
                \Bigr),
    \end{align*}
    where
    \begin{align*}
        \fun{ap}_{g\circ f}(\fun{merid}(1_{\fun2}))
            &\eq{}\fun{ap}_g(\fun{ap}_f
                (\fun{merid}(1_{\fun2})))\\
            &\eq{}\fun{ap}_g(\fun{refl}_{\fun{base}})\\
            &\eq{}\fun{refl}_{g(\fun{base})}\\
            &\eq{}\fun{refl}_{\fun N}.
    \end{align*}
    Hence,
    \begin{align*}
        \fun{ap}_{g\circ f}(\fun{merid}(1_{\fun2}))^{-1}
            \ct n
            \ct \fun{ap}_{\id_{\fun{Susp}(\fun2)}}(\fun{merid}(1_{\fun2}))
        &\eq{}
            \fun{refl}_{\fun N}^{-1}
            \ct \fun{refl}_{\fun N}
            \ct \fun{merid}(1_{\fun2})\\
        &\eq{}
            \fun{merid}(1_{\fun2})\\
        &\equiv s.
    \end{align*}
    This provides the witnesses $\mathfrak m(0_{\fun2})$ and $\mathfrak m(1_{\fun2})$, completing the proof.
    
\end{proof}

\begin{lem}\label{lem:Susp(S-1)=S0}
    $\fun{Susp}(\fun0)\simeq\fun2$.
\end{lem}

\begin{proof}
    For the forward map, we define
    \begin{align*}
        f&\colon\fun{Susp}(\fun0)\to\fun2\\
        f&\defeq\fun{rec}_{\fun{Susp}(\fun0)}(
            \fun2,
            0_{\fun2},
            1_{\fun2},
            \fun{rec}_{\fun0}
                (0_{\fun2}\eq{\fun2}1_{\fun2})
        ).
    \end{align*}
    For the backward map, let
    \[
        g\defeq\fun{rec}_{\fun2}(
            \fun{Susp}(\fun0),\fun N,\fun S
        ).
    \]
    To verify that $f\circ g\sim\id_{\fun2}$, observe that
    \begin{align*}
        f\circ g(0_{\fun2})&\equiv f(\fun N)\equiv 0_{\fun2},\\
        f\circ g(1_{\fun2})&\equiv f(\fun S)\equiv 1_{\fun2}.
    \end{align*}
    For the other composition, let $P\colon\fun{Susp}(\fun0)\to\univ$ be the type family defined by $P(x)\defeq g\circ f(x)\eq{\fun{Susp}(\fun0)}x$. Since
    \begin{align*}
        g\circ f(\fun N)&\equiv g(0_{\fun2})\equiv\fun N,\\
        g\circ f(\fun S)&\equiv g(1_{\fun2})\equiv\fun S,
    \end{align*}
    we define $n\defeq\fun{refl}_{\fun N}\colon P(\fun N)$ and $s\defeq\fun{refl}_{\fun S}\colon P(\fun S)$. Thus, for any $z\colon\fun0$, we have
    \[
        \fun{ind}_{\fun0}(Q,z)\colon n\eqt{\fun{merid}(z)}Ps,
    \]
    where $Q\colon\fun0\to\univ$ is defined by $Q(z)\defeq n\eqt{\fun{merid}(z)}Ps$. Therefore, we define $m(z)\defeq\fun{ind}_{\fun0}(Q,z)$ and complete the proof by providing
    \[
        \fun{ind}_{\fun{Susp}(\fun0)}(P,n,s,m)
    \]
    as the witness for $g\circ f\sim\id_{\fun{Susp}(\fun0)}$.
    %
    
\end{proof}

\begin{defn}
    Given two pointed types $\dep{A,a_0},\dep{B,b_0}\colon\puniv$, we introduce
    \[
        \fun{Map}_\sbull(\dep{A,a_0},\dep{B,b_0})
            \defeq\sum_{f\colon A\to B}f(a_0)\eq Bb_0.
    \]
    In addition, we define
    \begin{align*}
        (\dummy)_+&\colon\univ\to\puniv\\
        A_+&\defeq\dep{A+\fun1,\fun{inr}(\fstar)}.
    \end{align*}
\end{defn}

\begin{prop}\label{prop:transport-funext-eq}
    Let\/ $A$ and\/ $B$ be types. Given functions\/ $f,g\colon A\to B$ and a path\/ $\gamma\colon f\eq{A\to B}g$, and elements\/ $x_0\colon A$ and\/ $y_0\colon B$, consider the type family
    \begin{align*}
        P&\colon(A\to B)\to\univ\\
        P(h)&\defeq h(x_0)\eq{B}y_0.
    \end{align*}
    Then, for any path\/ $p\colon f(x_0)\eq{B}y_0$, we have
    \[
        \fun{transport}^P(\gamma,p)\eq{}\fun{happly}(\gamma,x_0)^{-1}\ct p.
    \]
\end{prop}

\begin{proof}
    By path induction on $\gamma$, it suffices to consider the case where $g\equiv f$ and $\gamma\equiv\fun{refl}_f$. In this case, the left-hand side evaluates definitionally to
    \[
        \fun{transport}^P(\fun{refl}_f,p) \equiv p.
    \]
    For the right-hand side, since $\fun{happly}(\fun{refl}_f,x_0)\equiv\fun{refl}_{f(x_0)}$, we can evaluate the expression as
    \begin{align*}
        \fun{happly}(\fun{refl}_f,x_0)^{-1} \ct p
            &\equiv \fun{refl}_{f(x_0)}^{-1} \ct p\\
            &\equiv \fun{refl}_{f(x_0)} \ct p\\
            &\equiv p.
    \end{align*}
    This establishes the equality for any path $\gamma$.
\end{proof}

\begin{cor}\label{cor:transport-funext-eq}
    Under the conditions of the propositions, if $\gamma\equiv\fun{funext}(\eta)$ for a homotopy $\eta\colon f\sim g$, then
    \[
        \fun{transport}^P(\fun{funext}(\eta),p)
            \eq{}\eta(x_0)^{-1}\ct p.
    \]
\end{cor}

\begin{proof}
    The result is a direct consequence of the equality
    \[
        \fun{happly}(\fun{funext}(\eta),x_0)
            \eq{}\eta(x_0).
    \]
\end{proof}


\begin{thm}
    Given $A\colon\univ$ and $\dep{B,b_0}\colon\puniv$, we have
    \[
        \fun{Map}_\sbull(A_+,\dep{B,b_0})
        \simeq
        (A\to B).
    \]
\end{thm}

\begin{proof}
    For the forward map, consider
    \begin{align*}
        \phi\colon\fun{Map}_\sbull(A_+,\dep{B,b_0})
            &\to(A\to B)\\
        \dep{f,p}&\mapsto f\circ\fun{inl},
    \end{align*}
    where $f\colon A+\fun1\to B$ with $p\colon f(\fun{inr}(\fstar))\eq Bb_0$, and $\fun{inl}\colon A\to A+\fun1$.

    For the backward map, we have
    \begin{align*}
        \psi\colon(A\to B)
            &\to\fun{Map}_\sbull(A_+,\dep{B,b_0})\\
        g&\mapsto\dep{
            \fun{rec}_{A+\fun1}(
                B,
                g,
                \lambda\fstar.\,b_0
            ),
            \fun{refl}_{b_0}
        }.
    \end{align*}
We can now compute both compositions:
    \begin{align*}
        \phi\circ\psi(g)
            &\equiv\fun{rec}_{A+\fun1}(
                B,
                g,
                \lambda\fstar.\,b_0
            )\circ\fun{inl}\\
            &\equiv g.
    \intertext{The other composition yields}
        \psi\circ\phi(\dep{f,p})
            &\equiv\dep{
                \fun{rec}_{A+\fun1}(
                    B,
                    f\circ\fun{inl},
                    \lambda\fstar.\,b_0
                ),
                \fun{refl}_{b_0}
            }.
    \end{align*}
    By the equality in $\Sigma$-types, constructing a path $\dep{f,p}\eq{}\psi\circ\phi(\dep{f,p})$ translates into providing paths
    \[
        \alpha\colon
            f
            \eq{A+\fun1\to B}
            \fun{rec}_{A+\fun1}(
                B,
                f\circ\fun{inl},
                \lambda\fstar.\,b_0
            )
    \]
    and
    \[
        \beta\colon\fun{transport}^P(\alpha,p)\eq{}\fun{refl}_{b_0},
    \]
    where $P(h)\defeq h(\fun{inr}(\fstar))\eq{B}b_0$.

    Let
    \[
        \rho\defeq\fun{rec}_{A+\fun1}(
            B,
            f\circ\fun{inl},
            \lambda\fstar.\,b_0
        ).
    \]
    To construct $\alpha$, we need a homotopy
    \[
        \eta\colon\prod_{x\colon A+\fun1}
            f(x)\eq{B}\rho(x).
    \]
    We define $\eta$ by induction on the coproduct $A+\fun1$. By the computation rule, we have
    \[
        \rho\circ\fun{inl}\equiv f\circ\fun{inl}.
    \]
    Hence, we can set $\eta\circ\fun{inl}\defeq\fun{refl}_{f\circ\fun{inl}}$. 

    For the right constructor, the computation rule gives $\rho\circ\fun{inr}\equiv\lambda\fstar.\,b_0$, which implies $\rho(\fun{inr}(\fstar))\equiv b_0$.
    
    Because we need a path of type $f(\fun{inr}(\fstar))\eq{B}\rho(\fun{inr}(\fstar))$ and $\rho(\fun{inr}(\fstar))\equiv b_0$, we can define~$\eta(\fun{inr}(\fstar))\defeq p$. 
    
    Applying function extensionality, we obtain $\alpha\defeq\fun{funext}(\eta)$.

    Now we must construct $\beta$. We need to evaluate the transport of $p$ along $\alpha$ within the type family $P(h)\defeq h(\fun{inr}(\fstar))\eq{B}b_0$. By Corollary~\ref{cor:transport-funext-eq}, we obtain
    \begin{align*}
        \fun{transport}^P(\alpha,p)
            &\eq{}\eta(\fun{inr}(\fstar))^{-1}\ct p\\
            &\equiv p^{-1}\ct p\\
            &\eq{}\fun{refl}_{b_0},
    \end{align*}
    which provides the required witness $\beta$.
    
\end{proof}

\begin{lem}\label{lem:sigma-identity-product}
    Given a type\/ $A\colon\univ$, an element\/ $a_0\colon A$, and a type family\/ $P\colon A\to\univ$, there is an equivalence
    \[
        \Bigl(
            \sum_{a\colon A}(a\eq{A}a_0)\times P(a)
        \Bigr)
        \simeq P(a_0).
    \]
\end{lem}

\begin{proof}
    First, we rewrite the cartesian product as a non-dependent $\Sigma$-type:
    \begin{align*}
        \lhs
            &\simeq\sum_{a\colon A}\;\sum_{p\colon a\eq{A}a_0}P(a).
    \intertext{By $\Sigma$-associativity (Proposition~\ref{prop:sigma-associativity}),}
                &\simeq\sum_{\omega\colon\sum_{a\colon A}a\eq{A}a_0}P(\pi_1(\omega)).
    \intertext{By Lemma~\ref{lem:Sigma-identity-is-contractible}, $\sum_{a\colon A}a\eq{A}a_0$ is contractible with center $\dep{a_0,\fun{refl}_{a_0}}$. Hence, Theorem~\ref{thm:contractible-types}~b) reduces the entire sum to}
            &\simeq P(\pi_1(\dep{a_0,\fun{refl}_{a_0}})),
    \end{align*}
    which is definitionally equal to $P(a_0)$.
    
\end{proof}

\begin{lem}\label{lem:map-graph-equivalence}
    Given types\/ $X,Y\colon\univ$ and a map\/ $f\colon X\to Y$, there is an equivalence
    \[
        X\simeq\sum_{x\colon X}\;\sum_{y\colon Y}
            f(x)\eq Yy.
    \]
    In particular, given a type $A\colon\univ$ and $a_0\colon A$, we have
    \[
        (A\to Y)
            \simeq
                \sum_{g\colon A\to Y}\;
                \sum_{y\colon Y}
                g(a_0)\eq Yy.
    \]
\end{lem}

\begin{proof}
    For each $x\colon X$, let $P\colon Y\to\univ$ be the constant family $P(y)\defeq\fun1$. Applying Lemma~\ref{lem:sigma-identity-product} to the type $Y$, the element $f(x)\colon Y$, and $P$, we have
    \begin{align*}
        \Bigl(
            \sum_{y\colon Y}f(x)\eq Yy
        \Bigr)
        &\simeq
        \Bigl(
            \sum_{y\colon Y}(y\eq Yf(x))\times\fun1
        \Bigr)\\
        &\simeq
        \fun1
        &&\text{; Lem.~\ref{lem:sigma-identity-product}}.
    \end{align*}
    Summing over all $x\colon X$, we obtain
    \begin{align*}
        \Bigl(
            \sum_{x\colon X}\;
            \sum_{y\colon Y}f(x)\eq Yy
        \Bigr)
        &\simeq
        \sum_{x\colon X}\fun1\\
        &\simeq X,
    \end{align*}
    which proves the first equivalence. The second equivalence follows from the first by setting $X\defeq A\to Y$ and $f\defeq\ev{a_0}\colon(A\to Y)\to Y$.
    %
    
\end{proof}

\begin{thm}\label{thm:map-susp-loop}
    For pointed types\/ $\dep{A,a_0}$ and\/ $\dep{B,b_0}$ we have
    \[
        \fun{Map}_\sbull(\dep{\fun{Susp}(A),\fun N},\dep{B,b_0})
            \simeq\fun{Map}_\sbull(
                \dep{A,a_0},
                \Omega\dep{B,b_0}
            ).
    \]
\end{thm}

\begin{proof}
    We construct the equivalence between the \lhs and the \rhs via the following chain of equivalences:
    \begin{align*}
        \lhs
            &\equiv\sum_{f\colon\fun{Susp}(A)\to B} f(\fun N)\eq{B}b_0\\
            &\simeq
                \sum_{n,s\colon B}\;
                \sum_{m\colon A\to n\eq{B}s}
                    n\eq{B}b_0
                    &&\text{; Prop.~\ref{prop:unpointed-susp-maps}}\\
            &\simeq
                \sum_{s\colon B}\;
                \sum_{n\colon B}\;
                \sum_{m\colon A\to n\eq{B}s}
                    n\eq{B}b_0
                &&\text{; Prop.~\ref{prop:sigma-commutativity}}\\
            &\simeq
                \sum_{s\colon B}\;
                \sum_{n\colon B}\;
                (A\to n\eq{B}s)\times(n\eq{B}b_0)
                &&\text{; Non-dependent $\Sigma$}\\
            &\simeq
                \sum_{s\colon B}\;
                \sum_{n\colon B}\;
                (n\eq{B}b_0)\times(A\to n\eq{B}s)\\
            &\simeq\sum_{s\colon B}A\to b_0\eq{B}s
                &&\text{; Lem.~\ref{lem:sigma-identity-product}}\\
            &\simeq
                \sum_{s\colon B}\;
                \sum_{g\colon A\to b_0\eq Bs}\;
                \sum_{p\colon b_0\eq Bs}
                    g(a_0)\eq{}p
                &&\text{; Lem.~\ref{lem:map-graph-equivalence}}\\
            &\simeq
                \sum_{s\colon B}\;
                \sum_{p\colon b_0\eq{B}s}\;
                \sum_{g\colon A\to b_0\eq{B}s}
                    g(a_0)\eq{}p
                &&\text{; Prop.~\ref{prop:sigma-commutativity}}\\
            &\simeq
                \sum_{
                    \omega\colon\sum_{s\colon B}b_0\eq{B}s
                }\;
                \sum_{
                    g\colon A\to b_0\eq{B}\pi_1(\omega)
                }
                    g(a_0)\eq{}\pi_2(\omega)
                &&\text{; Prop.~\ref{prop:sigma-associativity}}\\
             &\simeq
                \sum_{g\colon A\to b_0\eq{B}b_0}
                    g(a_0)\eq{}\fun{refl}_{b_0}
                &&\text{; Thm.~\ref{thm:contractible-types}~b)}\\
            &\equiv\rhs
                &&\text{; Defns.\ of $\Omega$ and $\fun{Map}_\sbull$},
    \end{align*}
    where the application of Theorem~\ref{thm:contractible-types}~b) is justified because
    \[
        \sum_{s\colon B}b_0\eq{B}s
    \]
    is contractible with center $\dep{b_0,\fun{refl}_{b_0}}$ by Lemma~\ref{lem:Sigma-identity-is-contractible}.
\end{proof}
